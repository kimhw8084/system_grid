class PackageVerificationError(RuntimeError):
    classification = "INSTALLER_PACKAGE_FAILURE"
    exit_code = 1


def local_sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as src:
        for chunk in iter(lambda: src.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def unittest_count(output: str) -> int:
    matches = re.findall(r"^Ran (\d+) tests? in ", output, flags=re.MULTILINE)
    if len(matches) != 1:
        raise PackageVerificationError(
            f"expected one unittest summary, found {len(matches)}"
        )
    return int(matches[0])

def parse_package_manifest(text: str) -> dict[str, str]:
    result: dict[str, str] = {}
    for line_number, raw in enumerate(text.splitlines(), 1):
        line = raw.strip()
        if not line:
            continue
        parts = line.split("  ", 1)
        if len(parts) != 2 or len(parts[0]) != 64 or any(ch not in "0123456789abcdef" for ch in parts[0]):
            raise PackageVerificationError(f"invalid manifest line {line_number}")
        digest, rel = parts
        if not rel or rel.startswith(("/", "\\")) or "\\" in rel or any(part in {"", ".", ".."} for part in Path(rel).parts):
            raise PackageVerificationError(f"unsafe manifest path at line {line_number}: {rel!r}")
        if rel in result:
            raise PackageVerificationError(f"duplicate manifest path: {rel}")
        result[rel] = digest
    if not result:
        raise PackageVerificationError("package manifest is empty")
    return result


def verify_tree_manifest(root: Path, manifest_path: Path | None = None) -> dict[str, str]:
    manifest_path = manifest_path or (root / "MANIFEST.sha256")
    if not manifest_path.is_file():
        raise PackageVerificationError("MANIFEST.sha256 missing")
    manifest = parse_package_manifest(manifest_path.read_text(encoding="utf-8"))
    actual = sorted(
        path.relative_to(root).as_posix()
        for path in root.rglob("*")
        if path.is_file() and path != manifest_path
    )
    expected = sorted(manifest)
    if expected != actual:
        raise PackageVerificationError(
            f"package manifest coverage mismatch missing={sorted(set(expected)-set(actual))} "
            f"extra={sorted(set(actual)-set(expected))}"
        )
    for rel, expected_digest in manifest.items():
        candidate = root / rel
        if candidate.is_symlink() or not candidate.is_file():
            raise PackageVerificationError(f"manifest member is not a regular file: {rel}")
        got = local_sha256_file(candidate)
        if got != expected_digest:
            raise PackageVerificationError(f"package digest mismatch for {rel}")
    return manifest

def _candidate_package_zips(package_filename: str) -> list[Path]:
    """Return bounded, deterministic candidate locations without requiring launch layout."""
    roots: list[Path] = []
    for candidate in (PACKAGE_ROOT.parent, Path.cwd(), PACKAGE_ROOT.parent.parent):
        with contextlib.suppress(OSError):
            resolved = candidate.resolve()
            if resolved not in roots:
                roots.append(resolved)
    home = Path.home()
    for relative in (
        Path("Library/Mobile Documents/com~apple~CloudDocs/SysGrid-Control/Inbox"),
        Path("Downloads"),
        Path("Desktop"),
    ):
        candidate = home / relative
        if candidate.is_dir():
            resolved = candidate.resolve()
            if resolved not in roots:
                roots.append(resolved)
    candidates: list[Path] = []
    for root in roots:
        exact = root / package_filename
        if exact.is_file() and not exact.is_symlink():
            candidates.append(exact)
        with contextlib.suppress(OSError):
            for item in sorted(root.glob("*.zip")):
                if item.is_file() and not item.is_symlink() and item not in candidates:
                    candidates.append(item)
    return candidates


def verify_issued_package_zip(stage: Path) -> dict[str, object]:
    builder.verify_generated_files(PACKAGE_ROOT)
    package_name = str(RELEASE_DESCRIPTOR["package"]["name"])
    package_filename = f"{package_name}_v{PACKAGE_VERSION}.zip"
    rebuilt = stage / package_filename
    builder.build_zip(PACKAGE_ROOT, rebuilt)
    rebuilt_sha = local_sha256_file(rebuilt)
    rebuilt_bytes = rebuilt.read_bytes()

    issued: Path | None = None
    for candidate in _candidate_package_zips(package_filename):
        with contextlib.suppress(OSError):
            if candidate.read_bytes() == rebuilt_bytes:
                issued = candidate
                break
    package_source = "discovered_exact_archive" if issued else "reconstructed_from_verified_tree"
    archive_path = issued or rebuilt

    prefix = f"{package_name}_v{PACKAGE_VERSION}/"
    expected_files = sorted(
        path.relative_to(PACKAGE_ROOT).as_posix()
        for path in PACKAGE_ROOT.rglob("*")
        if path.is_file() and not path.is_symlink()
    )
    seen: set[str] = set()
    with zipfile.ZipFile(archive_path) as archive:
        for info in archive.infolist():
            if info.is_dir():
                continue
            if info.filename in seen:
                raise sg.SupervisorError("INSTALLER_PACKAGE_FAILURE", f"duplicate ZIP member: {info.filename}", 1)
            seen.add(info.filename)
            if info.flag_bits & 1:
                raise sg.SupervisorError("INSTALLER_PACKAGE_FAILURE", f"encrypted ZIP member: {info.filename}", 1)
            mode = (info.external_attr >> 16) & 0o170000
            if mode == stat.S_IFLNK or mode not in (0, stat.S_IFREG):
                raise sg.SupervisorError("INSTALLER_PACKAGE_FAILURE", f"unsafe ZIP member type: {info.filename}", 1)
            if not info.filename.startswith(prefix):
                raise sg.SupervisorError("INSTALLER_PACKAGE_FAILURE", f"wrong package prefix: {info.filename}", 1)
            rel = info.filename[len(prefix):]
            if not rel or rel.startswith(("/", "\\")) or "\\" in rel or any(part in {"", ".", ".."} for part in Path(rel).parts):
                raise sg.SupervisorError("INSTALLER_PACKAGE_FAILURE", f"unsafe package ZIP path: {rel!r}", 1)
            local = PACKAGE_ROOT / rel
            if not local.is_file() or local.is_symlink() or archive.read(info) != local.read_bytes():
                raise sg.SupervisorError("INSTALLER_PACKAGE_FAILURE", f"ZIP/tree byte mismatch: {rel}", 1)
        corrupt = archive.testzip()
        if corrupt:
            raise sg.SupervisorError("INSTALLER_PACKAGE_FAILURE", f"package CRC failure: {corrupt}", 1)
    actual_files = sorted(name[len(prefix):] for name in seen)
    if actual_files != expected_files:
        raise sg.SupervisorError(
            "INSTALLER_PACKAGE_FAILURE",
            f"package ZIP coverage mismatch missing={sorted(set(expected_files)-set(actual_files))} extra={sorted(set(actual_files)-set(expected_files))}",
            1,
        )
    archive_sha = local_sha256_file(archive_path)
    if archive_path.read_bytes() != rebuilt_bytes:
        raise sg.SupervisorError(
            "INSTALLER_PACKAGE_FAILURE",
            f"deterministic rebuild mismatch archive={archive_sha} rebuilt={rebuilt_sha}",
            1,
        )
    return {
        "path": str(archive_path),
        "source": package_source,
        "sha256": archive_sha,
        "size": archive_path.stat().st_size,
        "member_count": len(expected_files),
        "deterministic_rebuild_sha256": rebuilt_sha,
        "deterministic_rebuild_byte_identical": True,
    }


def wrappers(install_root: Path) -> dict[str, bytes]:
    python = shlex.quote(sys.executable)
    supervisor = shlex.quote(str(install_root / "payload/sysgrid_supervisor.py"))
    sg_wrapper = f'''#!/usr/bin/env bash
set -Eeuo pipefail
case "${{1:-}}" in
  run) shift; exec {python} {supervisor} run "$@" ;;
  status) shift; exec {python} {supervisor} status "$@" ;;
  watch) shift; exec {python} {supervisor} watch "$@" ;;
  self-test) shift; exec {python} {supervisor} self-test "$@" ;;
  notification-test) shift; exec {python} {supervisor} notification-test "$@" ;;
  recover-artifact) shift; exec {python} {supervisor} recover-artifact "$@" ;;
  "") exec {python} {supervisor} run ;;
  *) exec {python} {supervisor} run "$@" ;;
esac
'''.encode()
    sgs_wrapper = f'''#!/usr/bin/env bash
set -Eeuo pipefail
exec {python} {supervisor} status "$@"
'''.encode()
    watch_wrapper = f'''#!/usr/bin/env bash
set -Eeuo pipefail
exec {python} {supervisor} watch "$@"
'''.encode()
    return {"sg": sg_wrapper, "sgs": sgs_wrapper, "sgwatch": watch_wrapper}


def _open_telegram_bot(username: str) -> str:
    url = f"https://t.me/{username}"
    command: list[str] | None = None
    if sys.platform == "darwin":
        command = ["open", url]
    elif shutil.which("xdg-open"):
        command = ["xdg-open", url]
    if command:
        with contextlib.suppress(OSError):
            subprocess.Popen(command, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    return url

def install_atomic(engine: Path, test_log: Path, expected_test_count: int) -> dict[str, object]:
    share = Path.home() / ".local/share/sysgrid-supervisor"
    target = share / f"v{PACKAGE_VERSION}"
    current = share / "current"
    bin_dir = Path.home() / ".local/bin"
    config = sg.config_root() / "config.json"
    stamp = dt.datetime.now(dt.timezone.utc).strftime("%Y%m%dT%H%M%SZ")
    backup = share / "backups" / f"{stamp}-pre-v{PACKAGE_VERSION}"
    share.mkdir(parents=True, exist_ok=True)
    bin_dir.mkdir(parents=True, exist_ok=True)
    backup.mkdir(parents=True, exist_ok=False)

    stage = share / f".v{PACKAGE_VERSION}.stage-{os.getpid()}"
    if stage.exists():
        shutil.rmtree(stage)
    shutil.copytree(PACKAGE_ROOT, stage)
    try:
        verify_tree_manifest(stage)
    except PackageVerificationError as exc:
        raise sg.SupervisorError("INSTALLER_COPY_FAILURE", f"staged package identity failure: {exc}", 1) from exc

    # Test the exact staged bytes before changing any command wrapper.
    env = sg.sanitized_env(); env["PYTHONPATH"] = os.pathsep.join([str(stage), str(stage / "payload")])
    staged_test = subprocess.run(
        [sys.executable, "-m", "unittest", "-v", "test_sysgrid_supervisor", "test_install_and_qualify"],
        cwd=str(stage), env=env, stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, check=False,
    )
    test_log.write_text(test_log.read_text(encoding="utf-8") + "\n# Staged-byte suite\n" + staged_test.stdout, encoding="utf-8")
    print(staged_test.stdout, end="", flush=True)
    if staged_test.returncode:
        raise sg.SupervisorError("INSTALLER_QUALIFICATION_FAILURE", "staged-byte test suite failed", 1)
    staged_test_count = unittest_count(staged_test.stdout)
    if staged_test_count != expected_test_count:
        raise sg.SupervisorError(
            "INSTALLER_QUALIFICATION_FAILURE",
            f"staged-byte test count mismatch: release report {expected_test_count}, observed {staged_test_count}",
            1,
        )
    try:
        verify_tree_manifest(stage)
    except PackageVerificationError as exc:
        raise sg.SupervisorError("INSTALLER_COPY_FAILURE", f"staged package changed during qualification: {exc}", 1) from exc

    previous_current = None
    if current.is_symlink():
        previous_current = os.readlink(current)
    elif current.exists():
        previous_current = str(current.resolve())
    for name in ("sg", "sgs", "sgwatch"):
        path = bin_dir / name
        if path.exists() or path.is_symlink():
            shutil.copy2(path, backup / name, follow_symlinks=False)

    if target.exists():
        try:
            existing_manifest = verify_tree_manifest(target)
        except PackageVerificationError as exc:
            raise sg.SupervisorError("INSTALLER_IDENTITY_COLLISION", f"existing target differs: {target}: {exc}", 1) from exc
        if existing_manifest != VERIFIED_PACKAGE_MANIFEST:
            raise sg.SupervisorError("INSTALLER_IDENTITY_COLLISION", f"existing target manifest differs: {target}", 1)
        shutil.rmtree(stage)
    else:
        os.replace(stage, target)
        sg.fsync_dir(share)
    try:
        installed_manifest = verify_tree_manifest(target)
    except PackageVerificationError as exc:
        raise sg.SupervisorError("INSTALLER_COPY_FAILURE", f"installed package identity failure: {exc}", 1) from exc
    if installed_manifest != VERIFIED_PACKAGE_MANIFEST:
        raise sg.SupervisorError("INSTALLER_COPY_FAILURE", "installed package manifest differs from issued package", 1)

    temp_link = share / f".current-{os.getpid()}"
    with contextlib.suppress(FileNotFoundError):
        temp_link.unlink()
    os.symlink(target.name, temp_link)
    os.replace(temp_link, current)
    sg.fsync_dir(share)

    wrapper_digests: dict[str, str] = {}
    try:
        for name, content in wrappers(target).items():
            sg.atomic_write_bytes(bin_dir / name, content, 0o755)
            os.chmod(bin_dir / name, 0o755)
            wrapper_digests[name] = sg.sha256_file(bin_dir / name)
        sg.atomic_write_json(config, {
            "supervisor_version": PACKAGE_VERSION,
            "supervisor_path": str(target / "payload/sysgrid_supervisor.py"),
            "engine_version": sg.ENGINE_VERSION,
            "engine_path": str(engine),
            "engine_sha256": sg.ENGINE_SHA256,
            "installed_utc": sg.now_utc(),
            "wrapper_sha256": wrapper_digests,
        })
        # Exact installed self-test plus wrapper status. No product execution occurs.
        run([sys.executable, str(target / "payload/sysgrid_supervisor.py"), "self-test"])
        run([sys.executable, str(target / "payload/sysgrid_supervisor.py"), "notification-test"])
        run([str(bin_dir / "sgs")])
    except Exception:
        # Restore command entry points and current pointer. Installed version bytes remain immutable for diagnosis.
        for name in ("sg", "sgs", "sgwatch"):
            saved = backup / name
            destination = bin_dir / name
            if saved.exists() or saved.is_symlink():
                shutil.copy2(saved, destination, follow_symlinks=False)
            else:
                with contextlib.suppress(FileNotFoundError):
                    destination.unlink()
        with contextlib.suppress(FileNotFoundError):
            current.unlink()
        if previous_current:
            os.symlink(previous_current, current)
        raise

    return {
        "target": str(target), "current": str(current), "backup": str(backup),
        "engine_path": str(engine), "engine_sha256": sg.ENGINE_SHA256,
        "wrapper_sha256": wrapper_digests, "config": str(config),
        "installed_manifest_entries": len(installed_manifest),
        "installed_tree_verified": True,
        "staged_test_count": staged_test_count,
    }


def qualification_zip(stage: Path, result: dict[str, object], classification: str, message: str, exit_code: int) -> Path:
    evidence = stage / "qualification"
    evidence.mkdir(parents=True, exist_ok=True)
    record = {
        "result": "success" if exit_code == 0 else "failure",
        "classification": classification,
        "message": message,
        "exit_code": exit_code,
        "project": "SysGrid",
        "issue": "OUT-31",
        "lane": "execution_platform",
        "mode": "qualification",
        "supervisor_version": PACKAGE_VERSION,
        "engine_version": sg.ENGINE_VERSION,
        "engine_sha256": sg.ENGINE_SHA256,
        "finished_utc": sg.now_utc(),
        "product_repository_mutation": "none",
        "product_commit_push": "none",
        "installation": result,
    }
    sg.atomic_write_json(evidence / "RESULT.json", record)
    sg.atomic_write_json(evidence / "QUALIFICATION.json", {
        "contract": list(RELEASE_DESCRIPTOR.get("qualification_contract", [])),
        "package_files": {
            **{
                rel: {"sha256": digest, "size": (PACKAGE_ROOT / rel).stat().st_size}
                for rel, digest in VERIFIED_PACKAGE_MANIFEST.items()
            },
            "MANIFEST.sha256": {
                "sha256": local_sha256_file(PACKAGE_ROOT / "MANIFEST.sha256"),
                "size": (PACKAGE_ROOT / "MANIFEST.sha256").stat().st_size,
            },
        },
        "package_manifest_entries": len(VERIFIED_PACKAGE_MANIFEST),
        "complete_tree_coverage": True,
        "released_test_count": result.get("released_test_count"),
        "staged_test_count": result.get("staged_test_count"),
        "release_descriptor_sha256": local_sha256_file(PACKAGE_ROOT / "RELEASE_DESCRIPTOR.json"),
        "progress_event_count": len(RELEASE_DESCRIPTOR.get("progress_events", [])),
        "resource_policy": RELEASE_DESCRIPTOR.get("resource_policy"),
        "issued_package": result.get("issued_package"),
        "recovery_contract": {
            "verified_output_resume_without_replay": True,
            "exact_live_process_reattachment": True,
            "safe_pre_state_retry_limit": 1,
            "ambiguous_mutation_replay": False,
            "identical_inbox_duplicate_convergence": True,
        },
    })
    test_log = stage / "TESTS.txt"
    if test_log.is_file():
        shutil.copy2(test_log, evidence / "TESTS.txt")
    if exit_code:
        (evidence / "TRACEBACK.txt").write_text(traceback.format_exc(), encoding="utf-8")
    (evidence / "REVIEW_ME_FIRST.md").write_text(
        f"# SysGrid Execution Supervisor v{PACKAGE_VERSION} Qualification\n\n"
        f"- Result: {record['result']}\n- Classification: {classification}\n- Message: {message}\n"
        f"- Exact engine: {sg.ENGINE_VERSION} / {sg.ENGINE_SHA256}\n"
        "- Product repository mutation: none\n- Product commit/push: none\n\n"
        "Review RESULT.json, QUALIFICATION.json, TESTS.txt, and MANIFEST.sha256.\n",
        encoding="utf-8",
    )
    sg.make_manifest(evidence)
    out = stage / sg.CANONICAL_ZIP
    sg.zip_directory(evidence, out)
    sg.verify_archive_manifest(out, allow_source=True)
    return out

    with tempfile.TemporaryDirectory(prefix="sysgrid-supervisor-install-") as td:
        stage = Path(td)
        test_log = stage / "TESTS.txt"
        test_log.write_text("", encoding="utf-8")
        installation: dict[str, object] = {}
        try:
            status("Verifying immutable package manifest.")
            manifest = verify_package_manifest()
            status(f"Package identity accepted: {len(manifest)} files.")
            status("Verifying issued ZIP bytes and deterministic rebuild identity.")
            installation["issued_package"] = verify_issued_package_zip(stage)

            status("Running pre-install released-byte suite.")
            env = sg.sanitized_env(); env["PYTHONPATH"] = os.pathsep.join([str(PACKAGE_ROOT), str(PAYLOAD_ROOT)])
            pre = subprocess.run(
                [sys.executable, "-m", "unittest", "-v", "test_sysgrid_supervisor", "test_install_and_qualify"],
                cwd=str(PACKAGE_ROOT), env=env, stdin=subprocess.DEVNULL,
                stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, check=False,
            )
            print(pre.stdout, end="", flush=True)
            test_log.write_text("# Pre-install released-byte suite\n" + pre.stdout, encoding="utf-8")
            if pre.returncode:
                raise sg.SupervisorError("INSTALLER_QUALIFICATION_FAILURE", "pre-install test suite failed", 1)
            expected_test_count = release_report_test_count()
            observed_test_count = unittest_count(pre.stdout)
            if observed_test_count != expected_test_count:
                raise sg.SupervisorError(
                    "INSTALLER_QUALIFICATION_FAILURE",
                    f"released-byte test count mismatch: release report {expected_test_count}, observed {observed_test_count}",
                    1,
                )
            installation["released_test_count"] = observed_test_count

            status("Locating and self-testing exact accepted Stable Engine v1.0.4.")
            engine = sg.discover_engine(deep=True)
            engine_self_test = sg.verify_engine(engine)
            installation["engine_self_test"] = engine_self_test
            status(f"Exact engine accepted: {engine}")

            status("Proving mandatory Telegram delivery before installation.")
            installation["telegram_probe"] = ensure_telegram_configuration()
            status("Installing versioned supervisor and atomic command wrappers.")
            installation.update(install_atomic(engine, test_log, expected_test_count))
            status("Installation and post-install qualification passed.")
            classification = "PASS"
            message = f"SysGrid Execution Supervisor v{PACKAGE_VERSION} installed and qualified; exact Stable Engine v{sg.ENGINE_VERSION} preserved"
            exit_code = 0
        except Exception as exc:
            classification = getattr(exc, "classification", "INSTALLER_INTERNAL_FAILURE")
            message = str(exc)
            exit_code = int(getattr(exc, "exit_code", 1))
            installation["error"] = f"{type(exc).__name__}: {exc}"
            print(f"FAIL: {classification}: {message}", file=sys.stderr)

        artifact = qualification_zip(stage, installation, classification, message, exit_code)
        control = sg.default_control_root()
        digest = sg.sha256_file(artifact)
        published = sg.publish_artifact(artifact, control, digest)
        print(f"Qualification artifact: {published}")
        print(f"SHA-256: {digest}")
        if exit_code == 0:
            print("PASS: installation complete.")
            print("Future runs: sg run /path/to/capsule.sgcap  (or plain sg with exactly one Inbox capsule)")
            print("Upload: Outbox/UPLOAD_THIS_TO_CHATGPT.zip")
        else:
            print("Upload the qualification artifact for review; no product run was started.", file=sys.stderr)
        return exit_code
