def sha256_file(path: pathlib.Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def run(argv: list[str], *, cwd: pathlib.Path | None = None, timeout: int = 60, check: bool = True, env: dict[str, str] | None = None, input_text: str | None = None) -> subprocess.CompletedProcess[str]:
    cp = subprocess.run(argv, cwd=str(cwd) if cwd else None, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, timeout=timeout, check=False, env=env, input=input_text)
    if check and cp.returncode != 0:
        raise CliError(f"command failed ({cp.returncode}): {' '.join(argv[:4])}: {(cp.stdout or '')[-2000:]}")
    return cp

def graphql(cfg: dict[str, Any], token: str, query: str, variables: dict[str, Any]) -> dict[str, Any]:
    """Use native macOS curl for Linear transport.

    This intentionally avoids inheriting Python/Conda TLS certificate behavior from the
    interactive shell. The credential never enters stdout/stderr or result evidence.
    """
    body = json.dumps({"query": query, "variables": variables}, separators=(",", ":"))
    argv = [
        "/usr/bin/curl", "--silent", "--show-error", "--fail-with-body",
        "--max-time", "20", "--request", "POST",
        "--header", "Content-Type: application/json",
        "--header", f"Authorization: [REDACTED]",
        "--header", "public-file-urls-expire-in: 120",
        "--header", f"User-Agent: SysGridDelivery/{CLI_VERSION}",
        "--data-binary", "@-", str(cfg["linear_endpoint"]),
    ]
    try:
        cp = subprocess.run(argv, input=body, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=25, check=False)
    except (OSError, subprocess.TimeoutExpired) as exc:
        raise CliError(f"Linear native-curl transport failed: {type(exc).__name__}") from exc
    if cp.returncode != 0:
        detail = sg_runner.redact((cp.stderr or cp.stdout or "").strip())[-800:]
        raise CliError(f"Linear native-curl transport failed ({cp.returncode}): {detail or 'no diagnostic'}")
    try:
        payload = json.loads(cp.stdout or "{}")
    except json.JSONDecodeError as exc:
        raise CliError("Linear returned non-JSON content") from exc
    if payload.get("errors"):
        messages = "; ".join(str(x.get("message", "GraphQL error")) for x in payload["errors"][:3])
        raise CliError("Linear GraphQL error: " + messages)
    return payload.get("data") or {}

def parse_marker(body: str, start: str, end: str) -> dict[str, Any] | None:
    if start not in body or end not in body:
        return None
    raw = body.split(start, 1)[1].split(end, 1)[0].strip()
    if raw.startswith("```json"):
        raw = raw[len("```json"):].strip()
        if raw.endswith("```"):
            raw = raw[:-3].strip()
    try:
        value = json.loads(raw)
    except json.JSONDecodeError:
        return None
    return value if isinstance(value, dict) else None


def marker_records(issue: dict[str, Any], start: str, end: str) -> list[tuple[str, dict[str, Any], dict[str, Any]]]:
    rows = []
    for comment in issue.get("comments", {}).get("nodes", []) or []:
        record = parse_marker(comment.get("body") or "", start, end)
        if record is not None:
            rows.append((comment.get("createdAt") or "", record, comment))
    rows.sort(key=lambda x: x[0])
    return rows


def authored_by_principal(comment: dict[str, Any], cfg: dict[str, Any]) -> bool:
    return str((comment.get("user") or {}).get("id") or "") == str(cfg["principal_author_id"])


def find_activation(issue: dict[str, Any], cfg: dict[str, Any]) -> dict[str, Any] | None:
    for _, record, comment in reversed(marker_records(issue, ACTIVATION_START, ACTIVATION_END)):
        if not authored_by_principal(comment, cfg):
            continue
        if (
            record.get("schema_version") == "1.3.0"
            and record.get("classification") == "SYSGRID_VNEXT_ACTIVATION"
            and record.get("status") == "ACTIVE"
            and record.get("project") == cfg["project"]
            and record.get("master") == cfg["master"]
            and record.get("issue") == cfg["issue"]
            and record.get("law_manifest_sha256") == cfg["law_manifest_sha256"]
            and record.get("runner_sha256") == cfg["runner_sha256"]
            and record.get("cli_sha256") == cfg["cli_sha256"]
        ):
            return record
    return None



def activation_active(issue: dict[str, Any], cfg: dict[str, Any]) -> bool:
    activation_file = STATE_ROOT / "ACTIVATION.json"
    records = marker_records(issue, ACTIVATION_START, ACTIVATION_END)
    for _, record, comment in reversed(records):
        if not authored_by_principal(comment, cfg):
            continue
        same_identity = (
            record.get("schema_version") == "1.3.0"
            and record.get("classification") == "SYSGRID_VNEXT_ACTIVATION"
            and record.get("project") == cfg["project"]
            and record.get("master") == cfg["master"]
            and record.get("issue") == cfg["issue"]
            and record.get("law_manifest_sha256") == cfg["law_manifest_sha256"]
            and record.get("runner_sha256") == cfg["runner_sha256"]
            and record.get("cli_sha256") == cfg["cli_sha256"]
        )
        if not same_identity:
            continue
        if record.get("status") == "REVOKED":
            activation_file.unlink(missing_ok=True)
            return False
        if record.get("status") == "ACTIVE":
            STATE_ROOT.mkdir(parents=True, exist_ok=True)
            activation_file.write_text(json.dumps(record, indent=2, sort_keys=True) + "\n", "utf-8")
            os.chmod(activation_file, 0o600)
            return True
    if activation_file.is_file():
        try:
            local = json.loads(activation_file.read_text("utf-8"))
        except Exception:
            return False
        return (
            local.get("status") == "ACTIVE"
            and local.get("law_manifest_sha256") == cfg["law_manifest_sha256"]
            and local.get("runner_sha256") == cfg["runner_sha256"]
            and local.get("cli_sha256") == cfg["cli_sha256"]
        )
    return False

def local_replays() -> set[str]:
    p = STATE_ROOT / "replay.json"
    if not p.is_file():

    try:

    except Exception:
        raise CliError("local replay ledger is invalid")


def find_active_command(issue: dict[str, Any], cfg: dict[str, Any]) -> tuple[dict[str, Any], dict[str, Any]] | None:
    completed = {str(r.get("replay_id")) for _, r, _ in marker_records(issue, RESULT_START, RESULT_END) if r.get("replay_id")}
    completed |= local_replays()
    active = []
    for created, record, comment in marker_records(issue, COMMAND_START, COMMAND_END):
        if not authored_by_principal(comment, cfg):
            continue
        if record.get("status") != "ACTIVE" or record.get("classification") != "SYSGRID_VNEXT_COMMAND" or record.get("schema_version") != "1.3.0":
            continue
        if str(record.get("replay_id")) in completed:
            continue
        active.append((created, record, comment))
    if not active:
        return None
    replay_ids = {str(x[1].get("replay_id")) for x in active}
    if len(replay_ids) > 1:
        raise CliError("multiple active vNext commands exist; fail closed")
    _, record, comment = active[-1]
    return record, comment


def find_approval(issue: dict[str, Any], cfg: dict[str, Any], command: dict[str, Any]) -> dict[str, Any]:
    approval_id = str(command.get("approval_comment_id") or "")
    for _, record, comment in reversed(marker_records(issue, APPROVAL_START, APPROVAL_END)):
        if str(comment.get("id") or "") != approval_id or not authored_by_principal(comment, cfg):
            continue
        keys = ("issue", "product_attempt", "patch_id", "base_sha", "bundle_sha256", "bundle_size", "replay_id", "approval_lock")
        if record.get("status") != "APPROVED" or record.get("classification") != "SYSGRID_VNEXT_APPROVAL" or record.get("schema_version") != "1.3.0":
            break
        if all(str(record.get(k)) == str(command.get(k)) for k in keys):
            return record
        break
    raise CliError("exact principal approval lock is missing or mismatched")

def validate_command_bundle(command: dict[str, Any], path: pathlib.Path) -> None:
    if path.stat().st_size != int(command.get("bundle_size", -1)):
        raise CliError("bundle size mismatch")
    if sha256_file(path) != str(command.get("bundle_sha256")):
        raise CliError("bundle SHA-256 mismatch")
    tmp = pathlib.Path(tempfile.mkdtemp(prefix="sysgrid-command-check-"))
    try:
        bundle = sg_runner.extract_bundle(path, tmp)
        m = bundle.manifest
        for key in ("issue", "product_attempt", "patch_id", "base_sha", "replay_id", "approval_lock"):
            if str(m.get(key)) != str(command.get(key)):
                raise CliError(f"command/bundle mismatch: {key}")
        if m.get("repository") != "kimhw8084/system_grid":
            raise CliError("command bundle repository mismatch")
    finally:
        shutil.rmtree(tmp, ignore_errors=True)


def evidence_markdown(result: dict[str, Any]) -> str:
    blocks: list[str] = []
    total = 0
    for artifact in (result.get("artifacts") or [])[:2]:
        try:
            path = pathlib.Path(str(artifact.get("path") or ""))
            mime = str(artifact.get("mime") or "")
            if mime not in {"image/jpeg", "image/png"} or not path.is_file():
                continue
            size = path.stat().st_size
            if size <= 0 or size > 450_000 or total + size > 700_000:
                continue
            encoded = base64.b64encode(path.read_bytes()).decode("ascii")
            title = re.sub(r"[^A-Za-z0-9 _.-]+", "", str(artifact.get("title") or "SysGrid TDRE"))[:100]
            blocks.append(f"![{title}](data:{mime};base64,{encoded})")
            total += size
        except Exception:
            continue
    return "\n\n".join(blocks)

def github_api(token: str, path: str) -> Any:
    if not path.startswith("/") or "\n" in path or "\r" in path:
        raise CliError("invalid GitHub API path")
    curl_cfg = (
        f'header = "Authorization: Bearer [REDACTED]"\n'
        'header = "Accept: application/vnd.github+json"\n'
        'header = "X-GitHub-Api-Version: 2022-11-28"\n'
    )
    cp = run([
        "/usr/bin/curl", "--config", "-", "--silent", "--show-error", "--fail-with-body",
        "--connect-timeout", "10", "--max-time", "25", f"https://api.github.com{path}",
    ], timeout=30, check=False, input_text=curl_cfg)
    if cp.returncode != 0:
        raise CliError("GITHUB_AUTH_REQUIRED: GitHub OAuth validation failed; run `sg auth`")
    try:
        return json.loads(cp.stdout or "{}")
    except json.JSONDecodeError as exc:
        raise CliError("GitHub API returned non-JSON content") from exc


def verify_github_oauth(token: str) -> str:
    user = github_api(token, "/user")
    login = str(user.get("login") or "")
    if login != GITHUB_OWNER:
        raise CliError(f"GITHUB_AUTH_WRONG_ACCOUNT: expected {GITHUB_OWNER}, received {login or '<unknown>'}")
    repo = github_api(token, f"/repos/{GITHUB_FULL}")
    perms = repo.get("permissions") or {}
    if perms.get("push") is not True:
        raise CliError("GITHUB_AUTH_INSUFFICIENT: authenticated account lacks push permission to SysGrid")
    return login

def configure_repo_credential(cfg: dict[str, Any], repo: pathlib.Path) -> None:
    helper = pathlib.Path(os.path.expanduser(str(cfg["github_credential_helper_path"]))).resolve()
    if not helper.is_file() or not os.access(helper, os.X_OK):
        raise CliError("SysGrid GitHub credential helper is missing or not executable")
    git_cli(repo, "config", "--local", "--unset-all", "credential.https://github.com.helper", check=False)
    git_cli(repo, "config", "--local", "--add", "credential.https://github.com.helper", "")
    git_cli(repo, "config", "--local", "--add", "credential.https://github.com.helper", f"!{helper}")
    git_cli(repo, "config", "--local", "credential.useHttpPath", "true")
    git_cli(repo, "config", "--local", "core.askPass", "/usr/bin/false")
    git_cli(repo, "remote", "set-url", "origin", GITHUB_REMOTE)


def ensure_pinned_gh() -> pathlib.Path:

    os.chmod(tools, 0o700)
    dest = tools / f"gh-{PINNED_GH_VERSION}"
    binary = dest / "gh"
    stamp = dest / "ARCHIVE.sha256"
    if binary.is_file() and stamp.is_file() and stamp.read_text().strip() == PINNED_GH_ARCHIVE_SHA256:
        cp = run([str(binary), "--version"], check=False)
        if cp.returncode == 0 and f"gh version {PINNED_GH_VERSION}" in (cp.stdout or ""):
            return binary

    archive = tools / f"gh_{PINNED_GH_VERSION}_macOS_arm64.zip"
    part = pathlib.Path(str(archive) + ".part")
    part.unlink(missing_ok=True)
    cp = run([
        "/usr/bin/curl", "--fail", "--location", "--silent", "--show-error",
        "--retry", "3", "--retry-delay", "1", "--connect-timeout", "10", "--max-time", "180",
        "--output", str(part), PINNED_GH_URL,
    ], timeout=190, check=False)
    if cp.returncode != 0 or not part.is_file():
        system = shutil.which("gh")
        if system:
            vp = run([system, "--version"], check=False)
            m = re.search(r"gh version (\d+)\.(\d+)\.(\d+)", vp.stdout or "")
            if m and tuple(map(int, m.groups())) >= (2, 92, 0):
                return pathlib.Path(system)
        raise CliError("unable to obtain pinned GitHub CLI bootstrap utility")

    got = sha256_file(part)
    if got != PINNED_GH_ARCHIVE_SHA256:
        part.unlink(missing_ok=True)
        raise CliError("GitHub CLI archive hash mismatch; authentication bootstrap aborted")
    archive.unlink(missing_ok=True)
    part.replace(archive)
    td = pathlib.Path(tempfile.mkdtemp(prefix="sysgrid-gh-", dir=str(tools)))
    try:
        run(["/usr/bin/unzip", "-q", str(archive), "-d", str(td)], timeout=60)
        matches = list(td.glob("gh_*/bin/gh"))
        if len(matches) != 1:
            raise CliError("unexpected GitHub CLI archive layout")
        shutil.rmtree(dest, ignore_errors=True)
        dest.mkdir(mode=0o700)
        shutil.copy2(matches[0], binary)
        os.chmod(binary, 0o755)
        stamp.write_text(PINNED_GH_ARCHIVE_SHA256 + "\n")
        os.chmod(stamp, 0o600)
    finally:
        shutil.rmtree(td, ignore_errors=True)
    cp = run([str(binary), "--version"], check=False)
    if cp.returncode != 0 or f"gh version {PINNED_GH_VERSION}" not in (cp.stdout or ""):
        raise CliError("pinned GitHub CLI bootstrap utility failed version validation")
    return binary

def ensure_github_machine_auth(cfg: dict[str, Any], repo: pathlib.Path, *, push_dry_run: bool = False) -> str:

    token = [REDACTED]

def config_paths(cfg: dict[str, Any]) -> tuple[pathlib.Path, pathlib.Path]:
    repo = pathlib.Path(os.path.expanduser(str(cfg["repository_path"]))).resolve()
    deps = pathlib.Path(os.path.expanduser(str(cfg["dependency_source_path"]))).resolve()
    return repo, deps


def git_snapshot(path: pathlib.Path) -> tuple[str, str]:
    if not path.exists():
        return "<missing>", "<missing>"
    head = git_cli(path, "rev-parse", "HEAD", check=False).stdout.strip()
    status = git_cli(path, "status", "--porcelain=v1", "--untracked-files=all", check=False).stdout
    return head, hashlib.sha256(status.encode()).hexdigest()

def run_active() -> int:
    cfg = load_config()
    token = [REDACTED]
    issue = issue_state(cfg, token)
    if not activation_active(issue, cfg):
        print("SYSGRID v1.3.0 candidate is installed and qualified, but not activated.")
        print("Result: NOT_ACTIVATED")
        return 0
    found = find_active_command(issue, cfg)
    if found is None:
        print("=== SYSGRID RESULT ===")
        print(f"Issue: {cfg['issue']}")
        print("Result: NO_ACTIVE_PATCH")
        print("=======================")
        return 0
    command, _ = found
    required = {"attachment_id", "attachment_title", "bundle_sha256", "bundle_size", "approval_comment_id", "issue", "product_attempt", "patch_id", "base_sha", "replay_id", "approval_lock"}
    if command.get("issue") != cfg["issue"] or not required.issubset(command):
        raise CliError("active command schema mismatch")
    find_approval(issue, cfg, command)
    row = attachment(cfg, token, str(command["attachment_id"]))
    if row.get("title") != command["attachment_title"]:
        raise CliError("attachment title mismatch")
    repo, deps = config_paths(cfg)
    ensure_github_machine_auth(cfg, repo, push_dry_run=False)
    result_file = STATE_ROOT / "last-result.json"
    with tempfile.TemporaryDirectory(prefix="sysgrid-download-") as td:
        bundle_path = pathlib.Path(td) / str(command["attachment_title"])
        download(str(row["url"]), token, bundle_path)
        validate_command_bundle(command, bundle_path)
        rc = sg_runner.execute(bundle_path, repo, STATE_ROOT, result_file, False, deps, emit=False)
    result = json.loads(result_file.read_text("utf-8"))
    posted = post_result(cfg, token, issue["id"], result, command)
    result["return"] = "AUTO_POSTED_TO_LINEAR" if posted else "PASTE_THIS_BLOCK"
    result_file.write_text(json.dumps(result, indent=2, sort_keys=True) + "\n", "utf-8")
    sg_runner.print_result(result)
    return rc

    parser.add_argument("command", nargs="?", default="run", choices=["run", "doctor", "qualify", "auth", "version"])
