        objs={

            'QUESTIONNAIRE.json':{'schema_version':'1.0','campaign_id':cid,'classification':'product_attempt','questions':questions},
            'QUESTIONNAIRE_CLOSURE.json':{'schema_version':'1.0','campaign_id':cid,'closed':True,'unresolved_count':0},
            'CHANGE_SPEC.json':{'schema_version':'1.0','campaign_id':cid,'approved_behavior_ids':['B-1']},
            'RISK_MATRIX.json':{'schema_version':'1.0','campaign_id':cid,'categories':[{'id':c,'status':'EVIDENCE_BACKED_NA','evidence':['not applicable in synthetic qualification']} for c in RISK_CATEGORIES]},

            'SOURCE_IDENTITY.json':{'schema_version':'1.0','campaign_id':cid,'repository':'synthetic','branch':'main','head':'0000000000000000000000000000000000000000','source_fingerprint_sha256':'0'*64},

            'NEXT_PERMITTED_ACTION.json':{'schema_version':'1.0','campaign_id':cid,'action':'SHADOW_QUALIFICATION_THEN_CONTROLLED_APPLY'},
        }
        for name,obj in objs.items(): (root/name).write_text(json.dumps(obj,sort_keys=True)+'\n')
        bound={}
        for name in ('LAW_IDENTITY.json','LINEAR_STATE_EXPECTATION.json','PLATFORM_IDENTITY.json','SOURCE_IDENTITY.json','ACTIVE_ARTIFACT_IDENTITY.json','CURRENT_GATE.json','NEXT_PERMITTED_ACTION.json'):
            bound[name]=hashlib.sha256((root/name).read_bytes()).hexdigest()
        (root/'CHAT_HANDOFF.json').write_text(json.dumps({'schema_version':'1.0','campaign_id':cid,'bound_files_sha256':bound},sort_keys=True)+'\n')
        approval={'schema_version':'1.0','campaign_id':cid}
        for key,name in [('questionnaire_sha256','QUESTIONNAIRE.json'),('change_spec_sha256','CHANGE_SPEC.json'),('risk_matrix_sha256','RISK_MATRIX.json'),('test_plan_sha256','TEST_PLAN.json'),('visual_matrix_sha256','VISUAL_MATRIX.json'),('chat_handoff_sha256','CHAT_HANDOFF.json')]:
            approval[key]=hashlib.sha256((root/name).read_bytes()).hexdigest()
        (root/'APPROVAL_LOCK.json').write_text(json.dumps(approval,sort_keys=True)+'\n')
        lines=[]
        for p in sorted(root.iterdir()):
            if p.is_file() and p.name!='MANIFEST.sha256': lines.append(f'{hashlib.sha256(p.read_bytes()).hexdigest()}  {p.name}')
        (root/'MANIFEST.sha256').write_text('\n'.join(lines)+'\n')
        output=root.parent/'campaign.sgcap'
        with zipfile.ZipFile(output,'w',zipfile.ZIP_DEFLATED) as zf:
            for p in sorted(root.iterdir()): zf.write(p,p.name)
        return output
    def test_full_campaign_validates(self):
        with tempfile.TemporaryDirectory() as td:
            root=Path(td)/'root'; root.mkdir(); package=self.make_package(root)
            result=validate_campaign(package)
            self.assertEqual(result['questions'],64)
            self.assertEqual(result['campaign_id'],'SYSGRID-CAMPAIGN-TEST')
if __name__=='__main__': unittest.main()
