            subprocess.run(["git", "config", "user.email", "test@example.com"], cwd=repo, check=True)
            subprocess.run(["git", "config", "user.name", "Test"], cwd=repo, check=True)
            (repo / "a.txt").write_text("base-a\n")
            (repo / "b.txt").write_text("base-b\n")
            (repo / "protected.txt").write_text("protected\n")
            subprocess.run(["git", "add", "."], cwd=repo, check=True)
            subprocess.run(["git", "commit", "-m", "base"], cwd=repo, check=True, stdout=subprocess.DEVNULL)
            subprocess.run(["git", "remote", "add", "origin", str(remote)], cwd=repo, check=True)

            head = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=repo, text=True).strip()
            (repo / "a.txt").write_text("retained-a\n")
            pre_a = mod.sha256_file(repo / "a.txt")
            pre_b = mod.sha256_file(repo / "b.txt")
            old_b = (repo / "b.txt").read_text()
            (repo / "b.txt").write_text("changed-b\n")
            patch = subprocess.check_output(["git", "diff", "--binary", "--", "b.txt"], cwd=repo)
            post_b = mod.sha256_file(repo / "b.txt")
            (repo / "b.txt").write_text(old_b)
            patch_path = capsule / "PAYLOADS/change.patch"; patch_path.write_bytes(patch)
            (capsule / "STATE/pre-paths.txt").write_text("a.txt\n")
            (capsule / "STATE/post-paths.txt").write_text("a.txt\nb.txt\n")
            (capsule / "STATE/pre.sha256").write_text(f"{pre_a}  a.txt\n{pre_b}  b.txt\n")
            (capsule / "STATE/post.sha256").write_text(f"{pre_a}  a.txt\n{post_b}  b.txt\n")
            (capsule / "STATE/protected.sha256").write_text(mod.sha256_file(repo / "protected.txt") + "  protected.txt\n")
            run = valid_run()
            run.update({"mode":"mutation", "baseline":{"branch":"main","head":head,"upstream":"origin/main"},

            mod.validate_run(run)
            mod.verify_repo_state(repo, run, capsule, "pre")
            mod.apply_patch(repo, capsule, run)
            mod.verify_repo_state(repo, run, capsule, "post")
            self.assertEqual(mod.dirty_paths(repo), ["a.txt", "b.txt"])
