def manifest_bytes(files: dict[str, bytes]) -> bytes:
    return ("\n".join(f"{hashlib.sha256(data).hexdigest()}  {name}" for name, data in sorted(files.items())) + "\n").encode()


def write_zip(path: Path, files: dict[str, bytes], *, modes: dict[str, int] | None = None, add_manifest: bool = True) -> None:
    payload = dict(files)
    if add_manifest:
        payload["MANIFEST.sha256"] = manifest_bytes(files)
    with zipfile.ZipFile(path, "w", compression=zipfile.ZIP_DEFLATED) as zf:
        for name, data in payload.items():
            info = zipfile.ZipInfo(name)
            info.compress_type = zipfile.ZIP_DEFLATED
            mode = (modes or {}).get(name, stat.S_IFREG | 0o600)
            info.external_attr = mode << 16
            zf.writestr(info, data)

        "lane": "read_only",
        "mode": "read_only",
        "engine": {"required_version": sg.ENGINE_VERSION},
        "baseline": {"branch": "main", "head": "0" * 40, "upstream": "origin/main"},
        "state": {},
        "patch": None,
        "recipes": [],
        "acceptance": {},
        "delivery": {"commit_push": False},

    }


def make_capsule(path: Path, run: dict | None = None, extra: dict[str, bytes] | None = None) -> Path:
    files = {"RUN.json": (json.dumps(run or base_run(), sort_keys=True) + "\n").encode()}
    files.update(extra or {})
    write_zip(path, files)
    return path


def make_output(path: Path, capsule_name: str, capsule_sha: str, *, engine_version: str = sg.ENGINE_VERSION, exit_code: int = 0) -> Path:
    result = {
        "result": "success" if exit_code == 0 else "failure",
        "classification": "PASS" if exit_code == 0 else "APP_TEST_FAILURE",
        "message": "done",
        "exit_code": exit_code,
        "capsule_file": capsule_name,
        "capsule_sha256": capsule_sha,
        "engine_version": engine_version,
    }
    write_zip(path, {"RESULT.json": (json.dumps(result) + "\n").encode(), "LOGS/a.txt": b"ok\n"})
    return path

    def test_parallel_checks_collects_all_failures(self):
        calls: list[str] = []
        def fail(name: str):
            calls.append(name)
            raise sg.SupervisorError(name.upper(), name)
        with self.assertRaises(sg.SupervisorError) as caught:
            sg.parallel_checks({"a": lambda: fail("a"), "b": lambda: fail("b")})
        self.assertEqual(set(calls), {"a", "b"})
        self.assertIn("a:", str(caught.exception))
        self.assertIn("b:", str(caught.exception))

    def test_exclusive_lock_rejects_overlap(self):
        with tempfile.TemporaryDirectory() as td, mock.patch.dict(os.environ, {"SYSGRID_SUPERVISOR_STATE": td}):
            entered = threading.Event(); release = threading.Event(); errors: list[Exception] = []
            def holder():
                with sg.exclusive_run_lock():
                    entered.set(); release.wait(5)
            thread = threading.Thread(target=holder); thread.start(); self.assertTrue(entered.wait(2))
            try:
                with self.assertRaises(sg.SupervisorError):
                    with sg.exclusive_run_lock():
                        pass
            finally:
                release.set(); thread.join(5)
            self.assertFalse(errors)

    def init_repo(self, root: Path) -> tuple[Path, str]:
        repo = root / "repo"; repo.mkdir()
        subprocess.run(["git", "init", "-b", "main", str(repo)], check=True, stdout=subprocess.DEVNULL)
        subprocess.run(["git", "-C", str(repo), "config", "user.email", "test@example.com"], check=True)
        subprocess.run(["git", "-C", str(repo), "config", "user.name", "Test"], check=True)
        (repo / "a.txt").write_text("a\n")
        subprocess.run(["git", "-C", str(repo), "add", "a.txt"], check=True)
        subprocess.run(["git", "-C", str(repo), "commit", "-m", "init"], check=True, stdout=subprocess.DEVNULL)
        subprocess.run(["git", "-C", str(repo), "remote", "add", "origin", str(root / "remote.git")], check=True)
        subprocess.run(["git", "init", "--bare", str(root / "remote.git")], check=True, stdout=subprocess.DEVNULL)
        subprocess.run(["git", "-C", str(repo), "push", "-u", "origin", "main"], check=True, stdout=subprocess.DEVNULL)
        head = subprocess.check_output(["git", "-C", str(repo), "rev-parse", "HEAD"], text=True).strip()
        return repo, head

    def write_fake_engine(self, path: Path, count_path: Path, exit_code: int = 0) -> Path:
        script = f"""#!/usr/bin/env python3

rc={exit_code}
result={{'result':'success' if rc==0 else 'failure','classification':'PASS' if rc==0 else 'APP_TEST_FAILURE','message':'fake complete','exit_code':rc,'capsule_file':cap.name,'capsule_sha256':sha,'engine_version':'1.0.4'}}
files={{'RESULT.json':(json.dumps(result,sort_keys=True)+'\\n').encode(),'LOGS/fake.txt':b'fake\\n'}}
files['MANIFEST.sha256']=('\\n'.join(f"{{hashlib.sha256(data).hexdigest()}}  {{name}}" for name,data in sorted(files.items()))+'\\n').encode()
with zipfile.ZipFile(out/'UPLOAD_THIS_TO_CHATGPT.zip','w',compression=zipfile.ZIP_DEFLATED) as z:
    for name,data in files.items():
        info=zipfile.ZipInfo(name); info.compress_type=zipfile.ZIP_DEFLATED; info.external_attr=(stat.S_IFREG|0o600)<<16; z.writestr(info,data)

"""
        path.write_text(script)
        return path

    def run_fake(self, root: Path, exit_code: int) -> tuple[int, Path, Path, sg.CapsuleInfo]:
        repo, head = self.init_repo(root)
        digest = sg.sha256_file(repo / "a.txt")
        cap = self.capsule_for_repo_state(root, repo, head, expected_dirty=[], pre_hashes={"a.txt": digest}, protected_hashes={})
        count = root / "count.txt"
        fake = self.write_fake_engine(root / "fake_engine.py", count, exit_code)
        control = root / "control"
        state = root / "state"
        env = {
            "SYSGRID_CONTROL_ROOT": str(control),
            "SYSGRID_SUPERVISOR_STATE": str(state),
            "SYSGRID_SUPERVISOR_CONFIG": str(root / "config"),
        }
        with mock.patch.dict(os.environ, env, clear=False), \
             mock.patch.object(sg, "discover_engine", return_value=fake), \
             mock.patch.object(sg, "verify_engine", return_value={"engine_version": "1.0.4", "result": "PASS"}):
            rc = sg.run_supervised(cap.path, repo)
        return rc, count, control, cap

    def test_end_to_end_supervised_success_invokes_engine_once(self):
        with tempfile.TemporaryDirectory() as td:
            rc, count, control, cap = self.run_fake(Path(td), 0)
            self.assertEqual(rc, 0)
            self.assertEqual(count.read_text(), "1")
            alias = control / "Outbox" / sg.CANONICAL_ZIP
            self.assertTrue(alias.is_file())
            _, result = sg.verify_archive_manifest(alias, allow_source=True)
            self.assertEqual(result["capsule_file"], cap.path.name)

    def test_end_to_end_app_failure_is_review_ready_without_retry(self):
        with tempfile.TemporaryDirectory() as td:
            root = Path(td)
            rc, count, control, _ = self.run_fake(root, 7)
            self.assertEqual(rc, 7)
            self.assertEqual(count.read_text(), "1")
            latest = json.loads((root / "state/latest.json").read_text())
            self.assertEqual(latest["state"], "FAILED_REVIEW_READY")
            self.assertEqual(latest["classification"], "APP_TEST_FAILURE")
            self.assertTrue((control / "Outbox" / sg.CANONICAL_ZIP).is_file())
