def utc_now() -> str:
    return dt.datetime.now(dt.timezone.utc).replace(microsecond=0).isoformat().replace("+00:00", "Z")


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_file(path: pathlib.Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def run(cmd: list[str], cwd: pathlib.Path | None = None, check: bool = True, text: bool = True) -> subprocess.CompletedProcess:
    return subprocess.run(cmd, cwd=cwd, check=check, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=text)


def git_text(repo: pathlib.Path, *args: str, default: str = "") -> str:
    cp = run(["git", *args], cwd=repo, check=False)
    return cp.stdout.strip() if cp.returncode == 0 else default


def safe_relative_path(raw: str) -> pathlib.PurePosixPath:
    normalized = raw.replace("\\", "/")
    p = pathlib.PurePosixPath(normalized)
    if p.is_absolute() or ".." in p.parts or not p.parts:
        raise ValueError(f"unsafe path: {raw}")
    return p


def zip_is_symlink(info: zipfile.ZipInfo) -> bool:
    mode = (info.external_attr >> 16) & 0o170000
    return mode == stat.S_IFLNK


def validate_zip(path: pathlib.Path, verify_manifest: bool = True) -> dict:
    result = {"path": str(path), "sha256": sha256_file(path), "size": path.stat().st_size}
    with zipfile.ZipFile(path, "r") as z:
        names = z.namelist()
        if len(names) != len(set(names)):
            raise ValueError("duplicate ZIP paths")
        for info in z.infolist():
            safe_relative_path(info.filename.rstrip("/")) if info.filename.rstrip("/") else None
            if zip_is_symlink(info):
                raise ValueError(f"symlink ZIP member: {info.filename}")
        bad = z.testzip()
        if bad:
            raise ValueError(f"ZIP CRC failure: {bad}")
        if verify_manifest and "MANIFEST.sha256" in names:
            for line in z.read("MANIFEST.sha256").decode("utf-8", "replace").splitlines():
                if not line.strip():
                    continue
                digest, name = line.split("  ", 1)
                if name == "MANIFEST.sha256":
                    continue
                if name not in names:
                    raise ValueError(f"manifest missing member: {name}")
                actual = sha256_bytes(z.read(name))
                if actual != digest:
                    raise ValueError(f"manifest mismatch: {name}")
        result["entries"] = len(names)
    return result


def is_safe_source_path(path: str) -> bool:
    p = pathlib.PurePosixPath(path.replace("\\", "/"))
    lowered = [part.lower() for part in p.parts]
    name_lower = p.name.lower()
    if any(part in FORBIDDEN_PARTS for part in lowered):
        return False
    if name_lower.startswith(".env") and name_lower not in ALLOWED_ENV_EXAMPLES:
        return False
    if name_lower in FORBIDDEN_NAMES:
        return False
    if any(name_lower.endswith(ext) for ext in FORBIDDEN_EXTENSIONS):
        return False
    return True


def _looks_like_placeholder(value: str) -> bool:
    normalized = value.strip().strip("'\"").lower()
    if normalized in SECRET_VALUE_PLACEHOLDERS:
        return True
    if normalized.startswith(("${", "$", "process.env", "os.environ", "getenv(")):
        return True
    if any(word in normalized for word in ("example", "sample", "dummy", "fake", "placeholder", "redacted", "changeme")):
        return True
    return False


def source_secret_reason(path: pathlib.Path) -> str | None:
    try:
        if path.stat().st_size > 2_000_000:
            return None
        data = path.read_bytes()
        if b"\x00" in data:
            return None
        text = data.decode("utf-8", "replace")
    except OSError:
        return "unreadable"
    for pattern in SECRET_INLINE_PATTERNS:
        if pattern.search(text):
            return "secret-content"
    for pattern in (SECRET_ASSIGNMENT_RE, SECRET_MAPPING_RE):
        for match in pattern.finditer(text):
            if not _looks_like_placeholder(match.group(1)):
                return "secret-assignment"
    return None



def collect_git_state(repo: pathlib.Path) -> dict:
    head = git_text(repo, "rev-parse", "HEAD", default="unknown")
    branch = git_text(repo, "branch", "--show-current", default="unknown")
    upstream = git_text(repo, "rev-parse", "--abbrev-ref", "--symbolic-full-name", "@{u}", default="unknown")
    remote_sha = git_text(repo, "rev-parse", "@{u}", default="unknown") if upstream != "unknown" else "unknown"
    status_cp = run(["git", "status", "--porcelain=v1", "--untracked-files=all"], cwd=repo, check=False)
    status = status_cp.stdout.rstrip("\n")
    dirty_paths: list[str] = []
    for line in status.splitlines():
        if len(line) >= 4:
            path = line[3:]
            if " -> " in path:
                path = path.split(" -> ", 1)[1]
            dirty_paths.append(path)
    return {
        "head": head,
        "branch": branch,
        "upstream": upstream,
        "remote_sha": remote_sha,
        "status": status,
        "dirty_paths": sorted(set(dirty_paths)),
        "staged_patch": git_text(repo, "diff", "--cached", "--binary"),
        "working_patch": git_text(repo, "diff", "--binary"),
    }


def write_manifest(root: pathlib.Path, output: pathlib.Path) -> None:
    lines: list[str] = []
    for p in sorted(root.rglob("*")):
        if p.is_file() and p != output:
            rel = p.relative_to(root).as_posix()
            lines.append(f"{sha256_file(p)}  {rel}")
    output.write_text("\n".join(lines) + "\n", encoding="utf-8")


def build_safe_source_snapshot(repo: pathlib.Path, output_zip: pathlib.Path) -> dict:
    tracked_raw = run(["git", "ls-files", "-z"], cwd=repo, text=False).stdout
    tracked = [x.decode("utf-8", "surrogateescape") for x in tracked_raw.split(b"\0") if x]
    untracked_raw = run(["git", "ls-files", "--others", "--exclude-standard", "-z"], cwd=repo, text=False).stdout
    untracked = [x.decode("utf-8", "surrogateescape") for x in untracked_raw.split(b"\0") if x]
    selected: list[tuple[str, pathlib.Path]] = []
    excluded: list[dict] = []
    total = 0
    for rel in sorted(set(tracked + untracked)):
        if not is_safe_source_path(rel):
            excluded.append({"path": rel, "reason": "policy"})
            continue
        p = repo / rel
        if not p.is_file() or p.is_symlink():
            excluded.append({"path": rel, "reason": "missing-or-nonregular"})
            continue
        secret_reason = [REDACTED]
        if secret_reason:
            excluded.append({"path": rel, "reason": secret_reason})
            continue
        size = p.stat().st_size
        if size > MAX_SOURCE_FILE or total + size > MAX_SOURCE_TOTAL:
            excluded.append({"path": rel, "reason": "size", "size": size})
            continue
        selected.append((rel, p))
        total += size
    with tempfile.TemporaryDirectory(prefix="sysgrid-source-") as td:
        root = pathlib.Path(td)
        manifest_lines: list[str] = []
        for rel, src in selected:
            dst = root / "SOURCE" / rel
            dst.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(src, dst)
            manifest_lines.append(f"{sha256_file(dst)}  SOURCE/{rel}")
        (root / "EXCLUSIONS.json").write_text(json.dumps(excluded, indent=2) + "\n", encoding="utf-8")
        (root / "SOURCE_MANIFEST.sha256").write_text("\n".join(manifest_lines) + "\n", encoding="utf-8")
        with zipfile.ZipFile(output_zip, "w", compression=zipfile.ZIP_DEFLATED) as z:
            for p in sorted(root.rglob("*")):
                if p.is_file():
                    z.write(p, p.relative_to(root).as_posix())
    return {"files": len(selected), "bytes": total, "excluded": len(excluded), "sha256": sha256_file(output_zip)}

def copy_untracked(repo: pathlib.Path, clone: pathlib.Path) -> None:
    raw = run(["git", "ls-files", "--others", "--exclude-standard", "-z"], cwd=repo, text=False).stdout
    for value in raw.split(b"\0"):
        if not value:
            continue
        rel = value.decode("utf-8", "surrogateescape")
        if not is_safe_source_path(rel):
            continue
        src = repo / rel
        if src.is_file() and not src.is_symlink():
            dst = clone / rel
            dst.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(src, dst)


def run_ship_ai_review(repo: pathlib.Path, pre_head: str, output_zip: pathlib.Path, state: dict) -> dict:

    if not script.is_file():
        create_fallback_ship_review(repo, output_zip, state, "scripts/ship-ai-review.sh missing")
        return {"status": "fallback", "reason": "script-missing", "sha256": sha256_file(output_zip)}
    real_head = state["head"]
    dirty = bool(state["dirty_paths"])
    log_text = ""
    try:
        if dirty:
            with tempfile.TemporaryDirectory(prefix="sysgrid-review-clone-") as td:

                run(["git", "clone", "--no-hardlinks", "--local", str(repo), str(clone)])

                working = run(["git", "diff", "--binary"], cwd=repo).stdout
                if staged:

                    if cp.returncode != 0:
                        raise RuntimeError(f"staged patch apply failed: {cp.stderr}")
                if working:
                    cp = subprocess.run(["git", "apply", "--binary", "-"], cwd=clone, input=working, text=True, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
                    if cp.returncode != 0:
                        raise RuntimeError(f"working patch apply failed: {cp.stderr}")
                copy_untracked(repo, clone)
                run(["git", "config", "user.name", "SysGrid Continuity Snapshot"], cwd=clone)
                run(["git", "config", "user.email", "continuity@sysgrid.local"], cwd=clone)
                run(["git", "add", "-A"], cwd=clone)
                if run(["git", "diff", "--cached", "--quiet"], cwd=clone, check=False).returncode != 0:
                    run(["git", "commit", "-m", "SYSGRID CONTINUITY SNAPSHOT ONLY"], cwd=clone)
                snapshot_head = git_text(clone, "rev-parse", "HEAD")
                clone_script = clone / "scripts" / "ship-ai-review.sh"
                args = [
                    str(clone_script), "--no-commit", "--no-push", "--base", real_head,
                    "--head", snapshot_head, "--mode", "full", "--checks", "none",
                    "--artifact-policy", "block", "--output", str(output_zip),
                    "--goal", "Create a complete continuity review of the exact retained SysGrid working state without mutating the real repository.",

                    raise RuntimeError(f"ship review failed in snapshot clone: exit {cp.returncode}\n{log_text[-4000:]}")
        else:
            base = pre_head if pre_head and pre_head != "unknown" else real_head
            args = [
                str(script), "--no-commit", "--no-push", "--base", base,
                "--head", real_head, "--mode", "full", "--checks", "none",
                "--artifact-policy", "block", "--output", str(output_zip),
                "--goal", "Create the complete SysGrid run-continuity review package for seamless ChatGPT takeover.",
            ]

                args.append("--allow-empty")
            cp = run(args, cwd=repo, check=False)
            log_text = cp.stdout + "\n" + cp.stderr
            if cp.returncode != 0 or not output_zip.is_file():
                raise RuntimeError(f"ship review failed: exit {cp.returncode}\n{log_text[-4000:]}")
        validate_zip(output_zip, verify_manifest=False)
        return {"status": "pass", "sha256": sha256_file(output_zip), "log_tail": log_text[-2000:]}
    except Exception as exc:
        create_fallback_ship_review(repo, output_zip, state, str(exc))
        return {"status": "fallback", "reason": str(exc), "sha256": sha256_file(output_zip), "log_tail": log_text[-2000:]}

        handoff = {
            "schema_version": "1.0",
            "continuity_version": VERSION,
            "created_utc": utc_now(),
            "launch": launch,
            "result": result,
            "provenance": provenance,
            "repository": state,
            "raw_engine_zip": raw_info,
            "ship_ai_review": ship,
            "source_snapshot": source_info,
            "active_law_files": laws,
            "safe_next_action": safe_next_action(result),
            "bundle_sha256": "external-detached",
            "bundle_sha256_location": "SysGrid-Control/Continuity/Current/SHA256.txt",
            "bundle_integrity": "Verify the uploaded ZIP bytes independently; every internal member is covered by MANIFEST.sha256. A ZIP cannot truthfully contain its own final SHA-256.",
        }
        (stage / "CONTINUITY" / "HANDOFF.json").write_text(json.dumps(handoff, indent=2) + "\n", encoding="utf-8")
        (stage / "CONTINUITY" / "git-status.txt").write_text(state.get("status", "") + "\n", encoding="utf-8")
        (stage / "CONTINUITY" / "working-tree.patch").write_text(state.get("working_patch", ""), encoding="utf-8")
        (stage / "CONTINUITY" / "staged.patch").write_text(state.get("staged_patch", ""), encoding="utf-8")
        (stage / "CONTINUITY" / "QUALIFIED_PLATFORM.json").write_text(json.dumps({
            "version": VERSION,
            "core_engine_version": result.get("engine_version", "1.0.3"),
            "continuity_contract": "Every sg run produces an enriched canonical Outbox bundle and iCloud Current/Archive handoff.",
            "linear_sync": "A ready-to-apply Linear update is always produced. Direct local Linear mutation requires separately configured credentials and is not assumed.",
        }, indent=2) + "\n", encoding="utf-8")

        # The final ZIP digest is necessarily detached: embedding a ZIP's own SHA-256
        # changes the ZIP bytes and invalidates that value. Internal continuity files
        # therefore state the detached policy; Current/Archive SHA256.txt carry the
        # exact post-copy digest, while MANIFEST.sha256 covers every internal member.
        (stage / "CONTINUITY" / "BOOTSTRAP.md").write_text(build_bootstrap(result, state, handoff), encoding="utf-8")
        (stage / "CONTINUITY" / "LINEAR_UPDATE_PENDING.md").write_text(build_linear_update(result, state, ship), encoding="utf-8")
        (stage / "CONTINUITY" / "BUNDLE_DIGEST_POLICY.md").write_text(
            "# Canonical bundle digest policy\n\n"
            "A ZIP cannot truthfully contain its own final SHA-256 because adding that value changes the ZIP bytes.\n\n"
            "- Verify all internal members with `MANIFEST.sha256`.\n"
            "- Verify the complete uploaded ZIP by computing SHA-256 from the uploaded bytes.\n"
            "- On the local machine, the detached digest is stored beside the bundle in "
            "`SysGrid-Control/Continuity/Current/SHA256.txt` and the matching archive directory.\n",
            encoding="utf-8",
        )
        review_first = f"""# SysGrid Canonical Continuity Review

- Core result: {result.get('result', 'unknown')}
- Classification: {result.get('classification', 'unknown')}
- Issue: {result.get('issue', 'OUT-31')}
- Iteration / attempt: {result.get('iteration', 'unknown')} / {result.get('attempt', 'unknown')}
- Current HEAD: {state.get('head', 'unknown')}
- Dirty paths: {len(state.get('dirty_paths', []))}
- Ship AI review: {ship.get('status', 'unknown')}

Start with `CONTINUITY/BOOTSTRAP.md` and `CONTINUITY/HANDOFF.json`.
"""
        (stage / "REVIEW_ME_FIRST.md").write_text(review_first, encoding="utf-8")

        manifest = stage / "MANIFEST.sha256"
        write_manifest(stage, manifest)
        final_temp = pathlib.Path(td) / "UPLOAD_THIS_TO_CHATGPT.final.zip"
        with zipfile.ZipFile(final_temp, "w", compression=zipfile.ZIP_DEFLATED) as z:
            for p in sorted(stage.rglob("*")):
                if p.is_file():
                    z.write(p, p.relative_to(stage).as_posix())
        validate_zip(final_temp)
        final_sha = sha256_file(final_temp)

        final_path = outbox / "UPLOAD_THIS_TO_CHATGPT.zip"
        atomic = outbox / f".UPLOAD_THIS_TO_CHATGPT.{os.getpid()}.tmp"
        shutil.copy2(final_temp, atomic)
        if sha256_file(atomic) != final_sha:
            raise RuntimeError("Outbox copy hash mismatch")
        os.replace(atomic, final_path)
        if sha256_file(final_path) != final_sha:
            raise RuntimeError("Outbox post-replace hash mismatch")

        run_id = str(result.get("run_id") or launch.get("launch_id") or dt.datetime.now().strftime("%Y%m%dT%H%M%S"))
        archive_dir = archive_root / run_id
        if archive_dir.exists():
            shutil.rmtree(archive_dir)
        archive_dir.mkdir(parents=True, exist_ok=True)
        if current.exists():
            shutil.rmtree(current)
        current.mkdir(parents=True, exist_ok=True)
        for rel in [
            "CONTINUITY/BOOTSTRAP.md",
            "CONTINUITY/HANDOFF.json",
            "CONTINUITY/LINEAR_UPDATE_PENDING.md",
            "CONTINUITY/QUALIFIED_PLATFORM.json",
            "CONTINUITY/BUNDLE_DIGEST_POLICY.md",
        ]:
            src = stage / rel
            dst = current / pathlib.Path(rel).name
            shutil.copy2(src, dst)
            shutil.copy2(src, archive_dir / pathlib.Path(rel).name)
        shutil.copy2(final_path, current / "UPLOAD_THIS_TO_CHATGPT.zip")
        shutil.copy2(final_path, archive_dir / "UPLOAD_THIS_TO_CHATGPT.zip")
        if sha256_file(current / "UPLOAD_THIS_TO_CHATGPT.zip") != final_sha:
            raise RuntimeError("Current continuity copy hash mismatch")
        if sha256_file(archive_dir / "UPLOAD_THIS_TO_CHATGPT.zip") != final_sha:
            raise RuntimeError("Archive continuity copy hash mismatch")
        (current / "SHA256.txt").write_text(final_sha + "  UPLOAD_THIS_TO_CHATGPT.zip\n", encoding="utf-8")
        (archive_dir / "SHA256.txt").write_text(final_sha + "  UPLOAD_THIS_TO_CHATGPT.zip\n", encoding="utf-8")
        return final_path


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--repo", required=True)
    parser.add_argument("--control-root", required=True)
    parser.add_argument("--raw-zip")
    parser.add_argument("--pre-head", default="unknown")
    parser.add_argument("--launch-json", required=True)
    parser.add_argument("--payload-dir", required=True)
    args = parser.parse_args()
    repo = pathlib.Path(args.repo).resolve()
    control_root = pathlib.Path(args.control_root).expanduser().resolve()
    raw_zip = pathlib.Path(args.raw_zip).resolve() if args.raw_zip else None
    launch = json.loads(pathlib.Path(args.launch_json).read_text(encoding="utf-8"))
    final = build_bundle(repo, control_root, raw_zip, args.pre_head, pathlib.Path(args.payload_dir), launch)
    print(json.dumps({"status": "PASS", "path": str(final), "sha256": sha256_file(final), "version": VERSION}, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
