#!/usr/bin/env bash
set -u
set -o pipefail

PACKAGE_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
VERSION="1.0.0"
TARGET="$HOME/.local/share/sysgrid-campaign-platform/v$VERSION"
BIN="$HOME/.local/bin"
BACKUPS="$HOME/.local/share/sysgrid-campaign-platform/backups"
ICLOUD_OUTBOX="$HOME/Library/Mobile Documents/com~apple~CloudDocs/SysGrid-Control/Outbox"
FALLBACK_OUTBOX="$HOME/SysGrid-Control/Outbox"
ACTIVATION_TMP="$(mktemp -d)"
LOG_FILE="$ACTIVATION_TMP/INSTALL.log"
STAGE_FILE="$ACTIVATION_TMP/STAGE.txt"

mkdir -p "$BIN" "$BACKUPS"
printf '%s\n' "initialization" > "$STAGE_FILE"

mark_stage() {
  printf '%s\n' "$1" > "$STAGE_FILE"
  printf '\n=== %s ===\n' "$1"
}

main() {
  set -euo pipefail

  mark_stage "package-manifest"
  python3 - <<'PY' "$PACKAGE_DIR"
import hashlib, pathlib, sys
root=pathlib.Path(sys.argv[1]); expected={}
for line in (root/'MANIFEST.sha256').read_text().splitlines():
    if line.strip():
        digest,path=line.split(None,1)
        expected[path.lstrip('* ')]=digest
actual=sorted(p.relative_to(root).as_posix() for p in root.rglob('*') if p.is_file() and p.name!='MANIFEST.sha256')
assert sorted(expected)==actual, (sorted(set(expected)-set(actual)),sorted(set(actual)-set(expected)))
for rel,digest in expected.items():
    assert hashlib.sha256((root/rel).read_bytes()).hexdigest()==digest, rel
print('Package manifest: PASS')
PY

  mark_stage "static-compilation"
  PYCACHE="$(mktemp -d)"
  PYTHONPYCACHEPREFIX="$PYCACHE" python3 -m py_compile "$PACKAGE_DIR"/payload/*.py
  rm -rf "$PYCACHE"
  echo "Python compilation: PASS"

  mark_stage "component-tests"
  PYTHONDONTWRITEBYTECODE=1 PYTHONPATH="$PACKAGE_DIR/payload" \
    python3 -m unittest discover -s "$PACKAGE_DIR/tests" -v

  mark_stage "source-engine-self-test"
  PYTHONDONTWRITEBYTECODE=1 "$PACKAGE_DIR/payload/sysgrid_runner.py" self-test

  mark_stage "source-controller-self-test"
  PYTHONDONTWRITEBYTECODE=1 python3 "$PACKAGE_DIR/payload/sysgrid_campaign.py" self-test

  mark_stage "versioned-install"
  STAMP="$(date -u +%Y%m%dT%H%M%SZ)"
  if [[ -e "$HOME/.local/bin/sg" ]]; then
    cp -p "$HOME/.local/bin/sg" "$BACKUPS/sg.$STAMP"
  fi
  rm -rf "$TARGET.tmp"
  mkdir -p "$TARGET.tmp"
  cp -R "$PACKAGE_DIR"/. "$TARGET.tmp"/
  rm -rf "$TARGET"
  mv "$TARGET.tmp" "$TARGET"
  chmod 0755 "$TARGET/payload/sg" "$TARGET/payload/sysgrid_runner.py" \
    "$TARGET/payload/sysgrid_supervisor.py" "$TARGET/payload/sysgrid_campaign.py"
  install -m 0755 "$TARGET/payload/sg" "$BIN/sg"

  mkdir -p "$HOME/.local/share/sysgrid-runner/engine-v1.1.0"
  install -m 0755 "$TARGET/payload/sysgrid_runner.py" \
    "$HOME/.local/share/sysgrid-runner/engine-v1.1.0/sysgrid_runner.py"
  install -m 0644 "$TARGET/payload/sysgrid_state_identity.py" \
    "$HOME/.local/share/sysgrid-runner/engine-v1.1.0/sysgrid_state_identity.py"

  mkdir -p "$HOME/.local/share/sysgrid-supervisor/v2.2.0/payload"
  cp "$TARGET/RELEASE_DESCRIPTOR.json" \
    "$HOME/.local/share/sysgrid-supervisor/v2.2.0/RELEASE_DESCRIPTOR.json"
  cp "$TARGET/payload/sysgrid_supervisor.py" \
    "$TARGET/payload/sysgrid_campaign.py" \
    "$TARGET/payload/sysgrid_state_identity.py" \
    "$HOME/.local/share/sysgrid-supervisor/v2.2.0/payload/"

  if [[ ! -f "$HOME/.zshrc" ]] || ! grep -Fq '$HOME/.local/bin' "$HOME/.zshrc"; then
    printf '\nexport PATH="$HOME/.local/bin:$PATH"\n' >> "$HOME/.zshrc"
  fi
  export PATH="$HOME/.local/bin:$PATH"

  mark_stage "installed-supervisor-self-test"
  PYTHONDONTWRITEBYTECODE=1 sg self-test

  mark_stage "installed-controller-self-test"
  PYTHONDONTWRITEBYTECODE=1 sg campaign-self-test

  mark_stage "activation-complete"
  cat <<'MSG'

SysGrid Campaign Platform v1.0.0 installed and qualified.
Stable Engine: v1.1.0
Supervisor: v2.2.0
Campaign Controller: v1.0.0

Future campaign operation:
  1. Save one approved .sgcap to iCloud Drive/SysGrid-Control/Campaign-Inbox
  2. Run: sg campaign
  3. Check: sg campaign-status
  4. Upload iCloud Drive/SysGrid-Control/Outbox/UPLOAD_THIS_TO_CHATGPT.zip

No product source, Git index, commit, or remote was changed by installation.
MSG
}

set +e
main 2>&1 | tee "$LOG_FILE"
RESULT=${PIPESTATUS[0]}
set -e

if [[ -d "$ICLOUD_OUTBOX" ]] || mkdir -p "$ICLOUD_OUTBOX" 2>/dev/null; then
  OUTBOX="$ICLOUD_OUTBOX"
else
  OUTBOX="$FALLBACK_OUTBOX"
  mkdir -p "$OUTBOX"
fi

if [[ "$RESULT" -eq 0 ]]; then
  REPORT_NAME="UPLOAD_THIS_TO_CHATGPT_PLATFORM_ACTIVATION.zip"
  STATUS="PASS"
else
  REPORT_NAME="UPLOAD_DIAGNOSTIC_TO_CHATGPT_PLATFORM_ACTIVATION.zip"
  STATUS="FAIL"
fi
REPORT_PATH="$OUTBOX/$REPORT_NAME"
STAGE="$(cat "$STAGE_FILE" 2>/dev/null || printf '%s' unknown)"

python3 - <<'PY' "$ACTIVATION_TMP" "$PACKAGE_DIR" "$TARGET" "$STATUS" "$RESULT" "$STAGE" "$REPORT_PATH"
import hashlib
import json
import os
import pathlib
import platform
import subprocess
import sys
import zipfile
from datetime import datetime, timezone

activation = pathlib.Path(sys.argv[1])

target = pathlib.Path(sys.argv[3])
status = sys.argv[4]
exit_code = int(sys.argv[5])
stage = sys.argv[6]
out_path = pathlib.Path(sys.argv[7])
out_path.parent.mkdir(parents=True, exist_ok=True)

def sha(path: pathlib.Path):
    return hashlib.sha256(path.read_bytes()).hexdigest() if path.is_file() else None

def command_version(args):
    try:
        return subprocess.run(args, check=False, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT).stdout.strip()
    except Exception as exc:
        return f"unavailable: {exc}"

installed = {
    "dispatcher": pathlib.Path.home()/".local/bin/sg",
    "engine": pathlib.Path.home()/".local/share/sysgrid-runner/engine-v1.1.0/sysgrid_runner.py",

    "supervisor": pathlib.Path.home()/".local/share/sysgrid-supervisor/v2.2.0/payload/sysgrid_supervisor.py",
    "campaign_controller": pathlib.Path.home()/".local/share/sysgrid-supervisor/v2.2.0/payload/sysgrid_campaign.py",
}
result = {
    "schema_version": "1.0",
    "result": status,
    "exit_code": exit_code,
    "last_stage": stage,
    "platform_version": "1.0.0",
    "engine_version": "1.1.0",
    "supervisor_version": "2.2.0",
    "campaign_controller_version": "1.0.0",
    "captured_utc": datetime.now(timezone.utc).isoformat(),
    "package_manifest_sha256": sha(package/"MANIFEST.sha256"),
    "target_path": str(target),
    "installed_files": {
        name: {"path": str(path), "exists": path.is_file(), "sha256": sha(path)}
        for name, path in installed.items()
    },
    "environment": {
        "platform": platform.platform(),
        "python": sys.version.splitlines()[0],
        "git": command_version(["git", "--version"]),
    },
    "product_repository_mutation": "not_requested",
}
(activation/"ACTIVATION_RESULT.json").write_text(json.dumps(result, indent=2, sort_keys=True)+"\n")
for name in ["RELEASE_DESCRIPTOR.json", "QUALIFICATION_REPORT.json", "IMPLEMENTATION_REPORT.json", "IMPROVEMENT_LEDGER.json"]:
    source = package/name
    if source.is_file():
        (activation/name).write_bytes(source.read_bytes())

print(f"Activation evidence SHA-256: {sha(out_path)}")
PY

rm -rf "$ACTIVATION_TMP"

echo
echo "Platform installation finished with exit code: $RESULT"
echo "Termius remains open."
exit "$RESULT"
