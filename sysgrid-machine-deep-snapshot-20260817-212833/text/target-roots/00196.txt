def manifest_bytes(files: dict[str, bytes]) -> bytes:
    return ("\n".join(f"{hashlib.sha256(data).hexdigest()}  {name}" for name, data in sorted(files.items())) + "\n").encode()


def write_zip(path: Path, files: dict[str, bytes], *, modes: dict[str, int] | None = None, add_manifest: bool = True) -> None:
    payload = dict(files)
    if add_manifest:
        payload["MANIFEST.sha256"] = manifest_bytes(files)
    with zipfile.ZipFile(path, "w", compression=zipfile.ZIP_DEFLATED) as zf:
        for name, data in payload.items():
            info = zipfile.ZipInfo(name)
            info.compress_type = zipfile.ZIP_DEFLATED
            mode = (modes or {}).get(name, stat.S_IFREG | 0o600)
            info.external_attr = mode << 16
            zf.writestr(info, data)

def make_capsule(path: Path, run: dict | None = None, extra: dict[str, bytes] | None = None) -> Path:
    files = {"RUN.json": (json.dumps(run or base_run(), sort_keys=True) + "\n").encode()}
    files.update(extra or {})
    write_zip(path, files)
    return path


def make_output(path: Path, capsule_name: str, capsule_sha: str, *, engine_version: str = sg.ENGINE_VERSION, exit_code: int = 0) -> Path:
    result = {
        "result": "success" if exit_code == 0 else "failure",
        "classification": "PASS" if exit_code == 0 else "APP_TEST_FAILURE",
        "message": "done",
        "exit_code": exit_code,
        "capsule_file": capsule_name,
        "capsule_sha256": capsule_sha,
        "engine_version": engine_version,
    }
    write_zip(path, {"RESULT.json": (json.dumps(result) + "\n").encode(), "LOGS/a.txt": b"ok\n"})
    return path

    def test_exclusive_lock_rejects_overlap(self):
        with tempfile.TemporaryDirectory() as td, mock.patch.dict(os.environ, {"SYSGRID_SUPERVISOR_STATE": td}):
            entered = threading.Event(); release = threading.Event(); errors: list[Exception] = []
            def holder():
                with sg.exclusive_run_lock():
                    entered.set(); release.wait(5)
            thread = threading.Thread(target=holder); thread.start(); self.assertTrue(entered.wait(2))
            try:
                with self.assertRaises(sg.SupervisorError):
                    with sg.exclusive_run_lock():
                        pass
            finally:
                release.set(); thread.join(5)
            self.assertFalse(errors)

    def init_repo(self, root: Path) -> tuple[Path, str]:
        repo = root / "repo"; repo.mkdir()
        subprocess.run(["git", "init", "-b", "main", str(repo)], check=True, stdout=subprocess.DEVNULL)
        subprocess.run(["git", "-C", str(repo), "config", "user.email", "test@example.com"], check=True)
        subprocess.run(["git", "-C", str(repo), "config", "user.name", "Test"], check=True)
        (repo / "a.txt").write_text("a\n")
        subprocess.run(["git", "-C", str(repo), "add", "a.txt"], check=True)
        subprocess.run(["git", "-C", str(repo), "commit", "-m", "init"], check=True, stdout=subprocess.DEVNULL)
        subprocess.run(["git", "-C", str(repo), "remote", "add", "origin", str(root / "remote.git")], check=True)
        subprocess.run(["git", "init", "--bare", str(root / "remote.git")], check=True, stdout=subprocess.DEVNULL)
        subprocess.run(["git", "-C", str(repo), "push", "-u", "origin", "main"], check=True, stdout=subprocess.DEVNULL)
        head = subprocess.check_output(["git", "-C", str(repo), "rev-parse", "HEAD"], text=True).strip()
        return repo, head

    def canonical_capsule_for_repo_state(
        self,
        root: Path,
        repo: Path,
        head: str,
        *,
        expected_dirty: list[str],
        pre_hashes: dict[str, str | None],
        protected_hashes: dict[str, str | None],
    ) -> sg.CapsuleInfo:
        def path_ledger(paths: list[str]) -> bytes:
            return (("\n".join(paths) + "\n") if paths else "").encode()

        def hash_ledger(hashes: dict[str, str | None]) -> bytes:
            lines = [
                f"{digest if digest is not None else 'MISSING'}  {path}"
                for path, digest in sorted(hashes.items())
            ]
            return (("\n".join(lines) + "\n") if lines else "").encode()

        run = base_run()
        run["lane"] = "product"
        run["mode"] = "resume_verify"
        run["baseline"] = {"branch": "main", "head": head, "upstream": "origin/main"}
        run["state"] = {
            "pre_expected_dirty_paths": "STATE/pre_expected_dirty_paths.txt",
            "pre_hashes": "STATE/pre_hashes.sha256",
            "post_expected_dirty_paths": "STATE/post_expected_dirty_paths.txt",
            "post_hashes": "STATE/post_hashes.sha256",
            "protected_hashes": "STATE/protected_hashes.sha256",
        }
        run["recipes"] = [{"id": "full-app-gate", "params": {"timeout": 3000}, "required": True}]
        run["delivery"] = {"branch": "main", "remote": "origin", "commit_push": True}
        cap = make_capsule(root / "canonical-state.sgcap", run, {
            "STATE/pre_expected_dirty_paths.txt": path_ledger(expected_dirty),
            "STATE/pre_hashes.sha256": hash_ledger(pre_hashes),
            "STATE/post_expected_dirty_paths.txt": path_ledger(expected_dirty),
            "STATE/post_hashes.sha256": hash_ledger(pre_hashes),
            "STATE/protected_hashes.sha256": hash_ledger(protected_hashes),
        })
        return sg.inspect_capsule(cap)

    def mutation_capsule_for_repo(self, root: Path, repo: Path, head: str) -> sg.CapsuleInfo:

        run = base_run()
        run["lane"] = "product"
        run["mode"] = "mutation"
        run["baseline"] = {"branch": "main", "head": head, "upstream": "origin/main"}
        run["patch"] = {"path": "PATCH/change.diff", "sha256": hashlib.sha256(patch).hexdigest()}
        run["state"] = {
            "pre_expected_dirty_paths": "STATE/pre-dirty.json",
