+def sha256_file(path: Path) -> str:
+    digest = hashlib.sha256()
+    with path.open("rb") as handle:
+        for chunk in iter(lambda: handle.read(1024 * 1024), b""):
+            digest.update(chunk)
+    return digest.hexdigest()

+def sha256_bytes(data: bytes) -> str:

+def source_role(rel: str) -> str:
+    path = Path(rel)
+    lowered = rel.lower()
+    if "test" in path.name.lower() or any(part.lower() in {"tests", "test", "e2e", "specs"} for part in path.parts):
+        return "test"
+    if "migration" in lowered or "alembic" in lowered:
+        return "migration"
+    if path.name in SOURCE_SPECIAL_NAMES or path.suffix == ".lock":
+        return "dependency_configuration"
+    if any(part in {"scripts", ".github"} for part in path.parts):
+        return "automation"
+    if path.suffix.lower() in {".md", ".rst", ".txt"}:
+        return "documentation"
+    return "product_source"

+def source_allowed(rel: str, path: Path) -> tuple[bool, str]:
+    parts = set(Path(rel).parts)
+    if parts & SOURCE_EXCLUDED_DIRS:
+        return False, "excluded_directory"
+    if path.name in SECRET_BASENAMES or path.name.startswith(".env") and path.name not in {".env.example", ".env.sample"}:
+        return False, "secret_basename"
+    if path.is_symlink() or not path.is_file():
+        return False, "unsupported_path_type"
+    if path.stat().st_size > MAX_SINGLE_SOURCE_BYTES:
+        return False, "file_size_limit"
+    suffix = path.suffix.lower()
+    if suffix not in SOURCE_SUFFIXES and path.name not in SOURCE_SPECIAL_NAMES:
+        return False, "non_source_file"
+    return True, "included"

+def build_maps(repo: Path, included: list[dict[str, Any]]) -> tuple[dict[str, Any], dict[str, Any], dict[str, Any]]:
+    entry_by_path = {entry["path"]: entry for entry in included}
+    paths = sorted(entry_by_path)
+    shared = [
+        p for p in paths
+        if any(token in p.lower() for token in ("shared", "operationalworkspace", "workspace", "api/client", "shell", "monitoringgrid"))
+    ]
+    tests: list[dict[str, Any]] = []
+    routes: list[dict[str, str]] = []
+    apis: list[dict[str, str]] = []
+    imports: list[dict[str, Any]] = []
+    for rel in paths:
+        role = entry_by_path[rel]["role"]
+        if role == "test" or any(marker in Path(rel).name.lower() for marker in TEST_MARKERS):
+            tests.append({"path": rel, "size": entry_by_path[rel]["size"], "sha256": entry_by_path[rel]["sha256"]})
+        text = read_text_bounded(repo / rel)
+        if text is None:
+            continue
+        for match in ROUTE_RE.finditer(text):
+            routes.append({"route": match.group(1), "owner": rel})
+        for match in API_RE.finditer(text):
+            apis.append({"method": match.group(2).upper(), "path": match.group(3), "owner": rel})
+        deps = sorted({next(group for group in match.groups() if group) for match in IMPORT_RE.finditer(text)})
+        if deps:
+            imports.append({"owner": rel, "imports": deps[:500]})
+    ownership = {
+        "shared_high_blast_radius": sorted(shared),
+        "role_counts": {
+            role: sum(1 for entry in included if entry["role"] == role)
+            for role in sorted({entry["role"] for entry in included})
+        },
+        "files": [{"path": entry["path"], "role": entry["role"]} for entry in included],
+    }
+    route_api_test = {"routes": routes[:10000], "api_endpoints": apis[:10000], "tests": tests}
+    dependency_graph = {"imports": imports}
+    return ownership, route_api_test, dependency_graph

+def git_snapshot(repo: Path, target: Path) -> dict[str, Any]:
+    target.mkdir(parents=True, exist_ok=True)
+    commands = {
+        "working.patch": ("diff", "--binary"),
+        "staged.patch": ("diff", "--cached", "--binary"),
+        "name_status.txt": ("diff", "--name-status", "HEAD"),
+        "tracked_index.txt": ("ls-files", "-s"),
+        "recent_commits.txt": ("log", "-n", "30", "--date=iso-strict", "--pretty=format:%H%x09%ad%x09%an%x09%s"),
+        "submodules.txt": ("submodule", "status", "--recursive"),
+    }
+    hashes: dict[str, Any] = {}
+    for name, args in commands.items():
+        try:
+            data = git(repo, *args).stdout.encode("utf-8")
+            (target / name).write_bytes(data)
+            hashes[name] = {"sha256": sha256_bytes(data), "size": len(data)}
+        except CaptureError as exc:
+            if name == "submodules.txt":
+                data = f"unavailable: {exc}\n".encode("utf-8")
+                (target / name).write_bytes(data)
+                hashes[name] = {"sha256": sha256_bytes(data), "size": len(data), "status": "unavailable"}
+            else:
+                raise
+    return hashes

+def source_fingerprint(entries: Iterable[dict[str, Any]], identity: dict[str, Any]) -> str:
+    digest = hashlib.sha256()
+    digest.update(identity["branch"].encode())
+    digest.update(b"\0")
+    digest.update(identity["head"].encode())
+    for path in identity["dirty_paths"]:
+        digest.update(path.encode())
+        digest.update(b"\0")
+    for entry in sorted(entries, key=lambda row: row["path"]):
+        digest.update(entry["path"].encode())
+        digest.update(b"\0")
+        digest.update(entry["sha256"].encode())
+        digest.update(b"\0")
+    return digest.hexdigest()

+            info.external_attr = 0o100644 << 16
+            info.compress_type = zipfile.ZIP_DEFLATED

+def main(argv: list[str] | None = None) -> int:
+    parser = argparse.ArgumentParser(prog="sysgrid-source-authority")
+    parser.add_argument("--repo", type=Path)
+    parser.add_argument("--output", type=Path)
+    parser.add_argument("--self-test", action="store_true")
+    args = parser.parse_args(argv)
+    if args.self_test:
+        return self_test()
+    repo: Path | None = None
+    outbox = outbox_path()
+    try:
+        repo = discover_repo(args.repo)
+        output = args.output.expanduser().resolve() if args.output else outbox / CANONICAL_ZIP
+        report = capture(repo, output)
+        print("SysGrid source authority: PASS")
+        print(f"Repository: {report['repository']}")
+        print(f"Branch/HEAD: {report['branch']} / {report['head']}")
+        print(f"Included files: {report['included_files']}")
+        print(f"Source fingerprint: {report['source_fingerprint']}")
+        print(f"Output: {report['zip_path']}")
+        print(f"SHA-256: {report['zip_sha256']}")
+        return 0
+    except CaptureError as exc:
+        output = outbox / DIAGNOSTIC_ZIP
+        diagnostic(output, repo, exc.classification, str(exc))
+        print(f"SysGrid source authority blocked: {exc.classification}", file=sys.stderr)
+        print(str(exc), file=sys.stderr)
+        print(f"Diagnostic: {output}", file=sys.stderr)
+        return exc.exit_code
+    except Exception as exc:
+        output = outbox / DIAGNOSTIC_ZIP
+        diagnostic(output, repo, "UNEXPECTED_CAPTURE_FAILURE", str(exc))
+        print(f"SysGrid source authority blocked: UNEXPECTED_CAPTURE_FAILURE\n{exc}\nDiagnostic: {output}", file=sys.stderr)
+        return 1
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())
