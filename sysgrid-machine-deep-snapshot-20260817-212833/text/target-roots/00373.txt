SHA256_RE = re.compile(r"^[0-9a-f]{64}$")

def now_utc() -> str:
    return dt.datetime.now(dt.timezone.utc).replace(microsecond=0).isoformat().replace("+00:00", "Z")


def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as handle:
        for chunk in iter(lambda: handle.read(4 * 1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def load_json(path: Path) -> dict[str, Any]:
    raw = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(raw, dict):
        raise RuntimeError(f"JSON object required: {path}")
    return raw


def atomic_json(path: Path, value: dict[str, Any], mode: int = 0o600) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_suffix(path.suffix + f".tmp-{os.getpid()}")
    temp.write_text(json.dumps(value, indent=2, sort_keys=True) + "\n", encoding="utf-8")
    os.chmod(temp, mode)
    os.replace(temp, path)


def run(command: list[str], *, env: dict[str, str] | None = None, check: bool = True) -> subprocess.CompletedProcess[str]:
    result = subprocess.run(command, text=True, capture_output=True, env=env)
    if check and result.returncode:
        raise RuntimeError(
            f"command failed ({result.returncode}): {' '.join(command)}\n"
            f"stdout={result.stdout[-4000:]}\nstderr={result.stderr[-4000:]}"
        )
    return result

@contextmanager
def dispatch_lock(*, blocking: bool) -> Iterator[bool]:
    lock_path = relay_paths()["lock"]
    lock_path.parent.mkdir(parents=True, exist_ok=True)
    handle = lock_path.open("a+")
    flags = fcntl.LOCK_EX | (0 if blocking else fcntl.LOCK_NB)
    try:
        try:
            fcntl.flock(handle.fileno(), flags)
        except BlockingIOError:
            yield False
            return
        yield True
    finally:
        try:
            fcntl.flock(handle.fileno(), fcntl.LOCK_UN)
        finally:
            handle.close()

def send_telegram(message: str) -> dict[str, Any]:
    path = relay_paths()["telegram"]
    if not path.is_file() or path.is_symlink():
        return {"configured": False, "delivered": False}
    config = load_json(path)
    token = [REDACTED]
    chat_id = str(config.get("chat_id") or "").strip()
    if not token or not chat_id:
        return {"configured": False, "delivered": False}
    payload = urllib.parse.urlencode({"chat_id": chat_id, "text": message, "disable_web_page_preview": "true"}).encode()
    request = urllib.request.Request(
        f"https://api.telegram.org/bot{token}/sendMessage",
        data=payload,
        method="POST",
    )
    try:
        with urllib.request.urlopen(request, timeout=12) as response:
            body = json.loads(response.read().decode("utf-8"))
        return {"configured": True, "delivered": bool(body.get("ok")), "message_id": (body.get("result") or {}).get("message_id")}
    except Exception as exc:  # Delivery is reported, never allowed to corrupt product evidence.
        return {"configured": True, "delivered": False, "error": f"{type(exc).__name__}: {exc}"}


def find_history_for_state(state: dict[str, Any]) -> tuple[Path, Path, str, int]:
    run_id = str(state.get("run_id") or "")
    payload_sha = str(state.get("artifact_sha256") or "")
    payload_size = int(state.get("artifact_size") or 0)
    if not RUN_ID_RE.fullmatch(run_id):
        raise RuntimeError(f"invalid supervisor run ID: {run_id!r}")
    if not SHA256_RE.fullmatch(payload_sha):
        raise RuntimeError("terminal supervisor state lacks a valid artifact SHA-256")
    if payload_size <= 0:
        raise RuntimeError("terminal supervisor state lacks a valid artifact size")
    candidates = sorted(relay_paths()["history"].glob(f"{run_id}__*.zip.sha256"))
    for sidecar in candidates:
        line = sidecar.read_text(encoding="utf-8").strip()
        match = re.fullmatch(r"([0-9a-f]{64})  ([^/]+\.zip)", line)
        if not match or match.group(1) != payload_sha:
            continue
        archive = sidecar.parent / match.group(2)
        if not archive.is_file() or archive.is_symlink():
            continue
        if archive.stat().st_size != payload_size:
            continue
        if sha256_file(archive) != payload_sha:
            continue
        return archive, sidecar, payload_sha, payload_size
    raise RuntimeError(f"no checksum-complete immutable History object matches run {run_id}")
