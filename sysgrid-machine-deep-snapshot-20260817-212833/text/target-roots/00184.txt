def canonical_json(value: Any) -> bytes:
    return (json.dumps(value, indent=2, sort_keys=True) + "\n").encode("utf-8")


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_file(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def load_source(root: Path = ROOT) -> dict[str, Any]:
    value = json.loads((root / SOURCE_FILE).read_text(encoding="utf-8"))
    if not isinstance(value, dict) or value.get("schema_version") != "2.1":
        raise RuntimeError("invalid RELEASE_SOURCE.json")
    return value


def count_tests(root: Path = ROOT) -> int:
    count = 0
    names: set[str] = set()
    for rel in TEST_FILES:
        tree = ast.parse((root / rel).read_text(encoding="utf-8"), filename=rel)
        for node in ast.walk(tree):
            if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name.startswith("test_"):
                key = f"{rel}:{node.name}:{node.lineno}"
                if key in names:
                    raise RuntimeError(f"duplicate test identity: {key}")
                names.add(key)
                count += 1
    if count <= 0:
        raise RuntimeError("no tests discovered")
    return count


def descriptor_value(root: Path = ROOT) -> dict[str, Any]:
    source = load_source(root)
    source_hashes = {rel: sha256_file(root / rel) for rel in HASHED_SOURCE_FILES}
    return {
        **source,
        "derived": {
            "generated_by": "BUILD_PACKAGE.py",
            "test_count": count_tests(root),
            "source_sha256": source_hashes,
        },
    }

def generated_files(root: Path = ROOT) -> dict[str, bytes]:
    descriptor = descriptor_value(root)
    return {
        DESCRIPTOR_FILE: canonical_json(descriptor),
        REPORT_FILE: canonical_json(release_report_value(descriptor)),
        "README.md": readme_bytes(descriptor),
        "CHANGELOG.md": changelog_bytes(descriptor),
        "RESEARCH_AND_DESIGN.md": research_bytes(descriptor),
    }


def write_generated_files(root: Path = ROOT) -> None:
    for rel, data in generated_files(root).items():
        (root / rel).write_bytes(data)


def regular_files(root: Path = ROOT) -> list[Path]:
    files = []
    for path in root.rglob("*"):
        if not path.is_file() or path.is_symlink():
            continue
        if "__pycache__" in path.parts or path.suffix == ".pyc":
            raise RuntimeError(f"generated bytecode prohibited: {path}")
        if path.name == MANIFEST_FILE:
            continue
        files.append(path)
    return sorted(files, key=lambda item: item.relative_to(root).as_posix())


def manifest_bytes(root: Path = ROOT) -> bytes:
    return ("\n".join(
        f"{sha256_file(path)}  {path.relative_to(root).as_posix()}" for path in regular_files(root)
    ) + "\n").encode("utf-8")


def write_manifest(root: Path = ROOT) -> None:
    (root / MANIFEST_FILE).write_bytes(manifest_bytes(root))


def verify_generated_files(root: Path = ROOT) -> None:
    expected = generated_files(root)
    mismatches = [rel for rel, data in expected.items() if not (root / rel).is_file() or (root / rel).read_bytes() != data]
    if mismatches:
        raise RuntimeError(f"generated release files differ: {mismatches}")
    if not (root / MANIFEST_FILE).is_file() or (root / MANIFEST_FILE).read_bytes() != manifest_bytes(root):
        raise RuntimeError("MANIFEST.sha256 differs from generated complete-tree manifest")


def deterministic_timestamp(descriptor: dict[str, Any]) -> tuple[int, int, int, int, int, int]:
    date = str(descriptor["package"]["deterministic_timestamp"])
    year, month, day = (int(part) for part in date[:10].split("-"))
    return year, month, day, 0, 0, 0


def build_zip(root: Path, destination: Path) -> None:
    descriptor = json.loads((root / DESCRIPTOR_FILE).read_text(encoding="utf-8"))
    prefix = f"{descriptor['package']['name']}_v{descriptor['supervisor_version']}"
    timestamp = deterministic_timestamp(descriptor)
    destination.parent.mkdir(parents=True, exist_ok=True)
    temp = destination.with_name(f".{destination.name}.tmp-{os.getpid()}")
    if temp.exists():
        temp.unlink()
    with zipfile.ZipFile(temp, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=9, allowZip64=True) as archive:
        for path in sorted((p for p in root.rglob("*") if p.is_file()), key=lambda p: p.relative_to(root).as_posix()):
            rel = path.relative_to(root).as_posix()
            if "__pycache__" in path.parts or path.suffix == ".pyc":
                raise RuntimeError(f"bytecode prohibited: {rel}")
            info = zipfile.ZipInfo(f"{prefix}/{rel}")
            info.date_time = timestamp
            executable = rel in {
                "INSTALL_AND_QUALIFY.command", "INSTALL_AND_QUALIFY.py", "BUILD_PACKAGE.py",
                "payload/sysgrid_supervisor.py",
            }
            info.external_attr = (stat.S_IFREG | (0o755 if executable else 0o644)) << 16
            info.create_system = 3
            info.compress_type = zipfile.ZIP_DEFLATED
            archive.writestr(info, path.read_bytes(), compress_type=zipfile.ZIP_DEFLATED, compresslevel=9)
    os.replace(temp, destination)


def run_tests(root: Path = ROOT) -> tuple[int, str]:
    env = {**os.environ, "PYTHONDONTWRITEBYTECODE": "1", "PYTHONNOUSERSITE": "1"}
    env["PYTHONPATH"] = os.pathsep.join([str(root), str(root / "payload")])
    completed = subprocess.run(
        [sys.executable, "-m", "unittest", "-v", "test_sysgrid_supervisor", "test_install_and_qualify"],
        cwd=root,
        env=env,
        stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        text=True,
        check=False,
    )
    if completed.returncode:
        raise RuntimeError(completed.stdout)
    return completed.returncode, completed.stdout


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--prepare", action="store_true", help="regenerate descriptor/docs/manifest before verification")
    parser.add_argument("--output", type=Path)
    parser.add_argument("--skip-tests", action="store_true")
    args = parser.parse_args()
    if args.prepare:
        write_generated_files(ROOT)
        write_manifest(ROOT)
    verify_generated_files(ROOT)
    expected_count = descriptor_value(ROOT)["derived"]["test_count"]
    if not args.skip_tests:
        _rc, output = run_tests(ROOT)
        import re
        matches = re.findall(r"^Ran (\d+) tests? in ", output, flags=re.MULTILINE)
        if matches != [str(expected_count)]:
            raise RuntimeError(f"test summary mismatch: expected {expected_count}, got {matches}")
        print(output, end="")
    output_path = args.output or ROOT.parent / f"{load_source(ROOT)['package']['name']}_v{load_source(ROOT)['supervisor_version']}.zip"
    build_zip(ROOT, output_path)
    print(f"Built {output_path}")
    print(f"SHA-256: {sha256_file(output_path)}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
