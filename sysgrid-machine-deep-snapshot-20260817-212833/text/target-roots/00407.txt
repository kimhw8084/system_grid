#!/usr/bin/env bash
set -u
set -o pipefail

PACKAGE_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
VERSION="1.1.0"
ENGINE_VERSION="1.1.1"
SUPERVISOR_VERSION="2.2.1"
CONTROLLER_VERSION="1.1.0"
TARGET="$HOME/.local/share/sysgrid-campaign-platform/v$VERSION"
BIN="$HOME/.local/bin"
BACKUPS="$HOME/.local/share/sysgrid-campaign-platform/backups"
ICLOUD_ROOT="$HOME/Library/Mobile Documents/com~apple~CloudDocs/SysGrid-Control"
ICLOUD_OUTBOX="$ICLOUD_ROOT/Outbox"
PROJECT_CONTINUITY_OUT="$ICLOUD_ROOT/Project-Continuity"
FALLBACK_ROOT="$HOME/SysGrid-Control"
ACTIVATION_TMP="$(mktemp -d)"
LOG_FILE="$ACTIVATION_TMP/INSTALL.log"
STAGE_FILE="$ACTIVATION_TMP/STAGE.txt"

mkdir -p "$BIN" "$BACKUPS"
printf '%s\n' "initialization" > "$STAGE_FILE"

mark_stage() {
  printf '%s\n' "$1" > "$STAGE_FILE"
  printf '\n=== %s ===\n' "$1"
}

main() {
  set -euo pipefail

  mark_stage "package-manifest"
  python3 - <<'PY' "$PACKAGE_DIR"
import hashlib, pathlib, sys
root=pathlib.Path(sys.argv[1]); expected={}
for line in (root/'MANIFEST.sha256').read_text().splitlines():
    if line.strip():
        digest,path=line.split(None,1); expected[path.lstrip('* ')]=digest
actual=sorted(p.relative_to(root).as_posix() for p in root.rglob('*') if p.is_file() and p != root/'MANIFEST.sha256')
assert sorted(expected)==actual, (sorted(set(expected)-set(actual)),sorted(set(actual)-set(expected)))
for rel,digest in expected.items(): assert hashlib.sha256((root/rel).read_bytes()).hexdigest()==digest, rel
print(f'Package manifest: PASS ({len(expected)}/{len(expected)})')
PY

  mark_stage "static-compilation"
  PYCACHE="$(mktemp -d)"
  PYTHONPYCACHEPREFIX="$PYCACHE" python3 -m py_compile "$PACKAGE_DIR"/payload/*.py
  rm -rf "$PYCACHE"
  echo "Python compilation: PASS"

  mark_stage "component-tests"
  PYTHONDONTWRITEBYTECODE=1 PYTHONPATH="$PACKAGE_DIR/payload" python3 -m unittest discover -s "$PACKAGE_DIR/tests" -v

  mark_stage "source-self-tests"
  PYTHONDONTWRITEBYTECODE=1 "$PACKAGE_DIR/payload/sysgrid_runner.py" self-test
  PYTHONDONTWRITEBYTECODE=1 python3 "$PACKAGE_DIR/payload/sysgrid_campaign.py" self-test
  PYTHONDONTWRITEBYTECODE=1 python3 "$PACKAGE_DIR/payload/sysgrid_supervisor.py" self-test

  mark_stage "versioned-install"
  STAMP="$(date -u +%Y%m%dT%H%M%SZ)"
  if [[ -e "$HOME/.local/bin/sg" ]]; then cp -p "$HOME/.local/bin/sg" "$BACKUPS/sg.$STAMP"; fi
  rm -rf "$TARGET.tmp"; mkdir -p "$TARGET.tmp"; cp -R "$PACKAGE_DIR"/. "$TARGET.tmp"/
  rm -rf "$TARGET"; mv "$TARGET.tmp" "$TARGET"
  chmod 0755 "$TARGET/payload/sg" "$TARGET/payload/sysgrid_runner.py" "$TARGET/payload/sysgrid_supervisor.py" "$TARGET/payload/sysgrid_campaign.py"
  install -m 0755 "$TARGET/payload/sg" "$BIN/sg"

  mkdir -p "$HOME/.local/share/sysgrid-runner/engine-v$ENGINE_VERSION"
  install -m 0755 "$TARGET/payload/sysgrid_runner.py" "$HOME/.local/share/sysgrid-runner/engine-v$ENGINE_VERSION/sysgrid_runner.py"
  install -m 0644 "$TARGET/payload/sysgrid_state_identity.py" "$HOME/.local/share/sysgrid-runner/engine-v$ENGINE_VERSION/sysgrid_state_identity.py"

  mkdir -p "$HOME/.local/share/sysgrid-supervisor/v$SUPERVISOR_VERSION/payload"
  cp "$TARGET/RELEASE_DESCRIPTOR.json" "$HOME/.local/share/sysgrid-supervisor/v$SUPERVISOR_VERSION/RELEASE_DESCRIPTOR.json"
  cp "$TARGET/payload/sysgrid_supervisor.py" "$TARGET/payload/sysgrid_campaign.py" "$TARGET/payload/sysgrid_state_identity.py" "$HOME/.local/share/sysgrid-supervisor/v$SUPERVISOR_VERSION/payload/"

  if [[ ! -f "$HOME/.zshrc" ]] || ! grep -Fq '$HOME/.local/bin' "$HOME/.zshrc"; then printf '\nexport PATH="$HOME/.local/bin:$PATH"\n' >> "$HOME/.zshrc"; fi
  export PATH="$HOME/.local/bin:$PATH"

  mark_stage "project-continuity-export"
  if mkdir -p "$PROJECT_CONTINUITY_OUT" 2>/dev/null; then
    rm -rf "$PROJECT_CONTINUITY_OUT"/*
    cp "$TARGET/PROJECT_CONTINUITY"/* "$PROJECT_CONTINUITY_OUT"/
  else
    PROJECT_CONTINUITY_OUT="$FALLBACK_ROOT/Project-Continuity"
    mkdir -p "$PROJECT_CONTINUITY_OUT"
    rm -rf "$PROJECT_CONTINUITY_OUT"/*
    cp "$TARGET/PROJECT_CONTINUITY"/* "$PROJECT_CONTINUITY_OUT"/
  fi
  echo "Project continuity files: $PROJECT_CONTINUITY_OUT"

  mark_stage "installed-self-tests"
  PYTHONDONTWRITEBYTECODE=1 sg self-test
  PYTHONDONTWRITEBYTECODE=1 sg campaign-self-test
  TEST_PATH="$(sg continuity-files)"
  [[ "$TEST_PATH" == "$TARGET/PROJECT_CONTINUITY" ]]
  [[ -f "$PROJECT_CONTINUITY_OUT/04_SYSGRID_NEW_CHAT_BOOTSTRAP_PROTOCOL_v1.0.0.md" ]]
  [[ -f "$PROJECT_CONTINUITY_OUT/05_SYSGRID_CONTINUITY_CONFORMANCE_SPEC_v1.0.0.md" ]]

  mark_stage "activation-complete"
  cat <<MSG

SysGrid Campaign Platform v$VERSION installed and qualified.
Stable Engine: v$ENGINE_VERSION
Supervisor: v$SUPERVISOR_VERSION
Campaign Controller: v$CONTROLLER_VERSION

P2 Project files were exported to:
  $PROJECT_CONTINUITY_OUT

No product source, Git index, commit, or remote was changed by installation.
MSG
}

set +e
main 2>&1 | tee "$LOG_FILE"
RESULT=${PIPESTATUS[0]}
set -e

if [[ -d "$ICLOUD_OUTBOX" ]] || mkdir -p "$ICLOUD_OUTBOX" 2>/dev/null; then OUTBOX="$ICLOUD_OUTBOX"; else OUTBOX="$FALLBACK_ROOT/Outbox"; mkdir -p "$OUTBOX"; fi
if [[ "$RESULT" -eq 0 ]]; then REPORT_NAME="UPLOAD_THIS_TO_CHATGPT_P2_CONTINUITY_ACTIVATION.zip"; STATUS="PASS"; else REPORT_NAME="UPLOAD_DIAGNOSTIC_TO_CHATGPT_P2_CONTINUITY_ACTIVATION.zip"; STATUS="FAIL"; fi
REPORT_PATH="$OUTBOX/$REPORT_NAME"
STAGE="$(cat "$STAGE_FILE" 2>/dev/null || printf '%s' unknown)"

python3 - <<'PY' "$ACTIVATION_TMP" "$PACKAGE_DIR" "$TARGET" "$PROJECT_CONTINUITY_OUT" "$STATUS" "$RESULT" "$STAGE" "$REPORT_PATH"
import hashlib,json,pathlib,platform,subprocess,sys,zipfile
from datetime import datetime,timezone
activation=pathlib.Path(sys.argv[1]); package=pathlib.Path(sys.argv[2]); target=pathlib.Path(sys.argv[3]); project_out=pathlib.Path(sys.argv[4]); status=sys.argv[5]; exit_code=int(sys.argv[6]); stage=sys.argv[7]; out_path=pathlib.Path(sys.argv[8]); out_path.parent.mkdir(parents=True,exist_ok=True)
def sha(p): return hashlib.sha256(p.read_bytes()).hexdigest() if p.is_file() else None
def command_version(args):
    try:return subprocess.run(args,check=False,text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT).stdout.strip()
    except Exception as exc:return f'unavailable: {exc}'
installed={

 'engine':pathlib.Path.home()/'.local/share/sysgrid-runner/engine-v1.1.1/sysgrid_runner.py',

 'campaign_controller':pathlib.Path.home()/'.local/share/sysgrid-supervisor/v2.2.1/payload/sysgrid_campaign.py',
}
result={'schema_version':'1.0','result':status,'exit_code':exit_code,'last_stage':stage,'repair':'Platform Verification Repair P2','platform_version':'1.1.0','engine_version':'1.1.1','supervisor_version':'2.2.1','campaign_controller_version':'1.1.0','captured_utc':datetime.now(timezone.utc).isoformat(),'package_manifest_sha256':sha(package/'MANIFEST.sha256'),'target_path':str(target),'project_continuity_path':str(project_out),'installed_files':{name:{'path':str(path),'exists':path.is_file(),'sha256':sha(path)} for name,path in installed.items()},'project_files':{p.name:{'sha256':sha(p),'size':p.stat().st_size} for p in sorted(project_out.glob('*')) if p.is_file()},'environment':{'platform':platform.platform(),'python':sys.version.splitlines()[0],'git':command_version(['git','--version'])},'product_repository_mutation':'not_requested'}
(activation/'ACTIVATION_RESULT.json').write_text(json.dumps(result,indent=2,sort_keys=True)+'\n')
for name in ['RELEASE_DESCRIPTOR.json','QUALIFICATION_REPORT.json','IMPLEMENTATION_REPORT.json','IMPROVEMENT_LEDGER.json']:
    src=package/name
    if src.is_file():(activation/name).write_bytes(src.read_bytes())
project_evidence=activation/'PROJECT_CONTINUITY'; project_evidence.mkdir(exist_ok=True)
for p in sorted(project_out.glob('*')):
    if p.is_file():(project_evidence/p.name).write_bytes(p.read_bytes())

(activation/'MANIFEST.sha256').write_text(''.join(f'{sha(p)}  {p.relative_to(activation).as_posix()}\n' for p in members))
if out_path.exists():out_path.unlink()
with zipfile.ZipFile(out_path,'w',compression=zipfile.ZIP_DEFLATED,compresslevel=9) as zf:
    for p in sorted(activation.rglob('*')):
        if not p.is_file():continue
        info=zipfile.ZipInfo(p.relative_to(activation).as_posix(),(2026,8,2,0,0,0)); info.external_attr=0o100644<<16; info.compress_type=zipfile.ZIP_DEFLATED; zf.writestr(info,p.read_bytes())
print(f'Activation evidence: {out_path}')
print(f'Activation evidence SHA-256: {sha(out_path)}')
PY
rm -rf "$ACTIVATION_TMP"
echo
echo "P2 installation finished with exit code: $RESULT"
echo "Termius remains open."
exit "$RESULT"
