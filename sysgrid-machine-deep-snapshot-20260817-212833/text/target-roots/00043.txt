#!/usr/bin/env bash
set -Eeuo pipefail
backup_root="${1:-}"
if [[ -z "$backup_root" || ! -d "$backup_root" ]]; then
  printf 'Usage: %s <backup-directory>\n' "$0" >&2
  exit 2
fi
for name in sg sgs sgwatch; do
  [[ -f "$backup_root/$name" ]] || { printf 'Missing backup: %s\n' "$backup_root/$name" >&2; exit 2; }
  install -m 0755 "$backup_root/$name" "$HOME/.local/bin/$name"
done
printf 'Restored SysGrid wrappers from %s\n' "$backup_root"
