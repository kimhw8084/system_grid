#!/usr/bin/env python3
"""Canonical SysGrid repository state identity.

This module is standard-library-only and is shared by the supervisor, campaign
controller, and Stable Engine. It returns individual file paths for tracked,
modified, staged, and valid untracked product files. It never represents an
untracked tree as a directory identity.
"""
from __future__ import annotations

import os
import subprocess
from pathlib import Path
from typing import Iterable

MAX_EXPANDED_UNTRACKED_FILES = 100_000

class StateIdentityError(RuntimeError):
    pass


def _split_nul(data: bytes) -> list[str]:
    return [part.decode("utf-8", "surrogateescape") for part in data.split(b"\0") if part]


def _safe_env() -> dict[str, str]:
    allowed = {"PATH", "HOME", "USER", "LOGNAME", "TMPDIR", "LANG", "LC_ALL", "SSH_AUTH_SOCK"}
    env = {key: value for key, value in os.environ.items() if key in allowed}
    env.setdefault("PATH", "/usr/local/bin:/usr/bin:/bin:/usr/sbin:/sbin")
    env["GIT_OPTIONAL_LOCKS"] = "0"
    return env


def _run_git_bytes(repo: Path, *args: str, timeout: float = 30) -> bytes:
    completed = subprocess.run(
        ["git", "-C", str(repo), "-c", "core.quotepath=false", *args],
        stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        timeout=timeout,
        check=False,
        env=_safe_env(),
    )
    if completed.returncode:
        message = completed.stderr.decode("utf-8", "replace")[:1000]
        raise StateIdentityError(f"git {' '.join(args)} failed: {message}")
    return completed.stdout


def _resolve_repo_member(repo: Path, rel: str) -> Path:
    if not rel or rel.startswith("/") or "\\" in rel or "\0" in rel:
        raise StateIdentityError(f"unsafe repository path: {rel!r}")
    candidate = (repo / rel).resolve()
    try:
        candidate.relative_to(repo.resolve())
    except ValueError as exc:
        raise StateIdentityError(f"repository path escapes root: {rel}") from exc
    return candidate


def expand_collapsed_untracked_directory(repo: Path, rel: str) -> tuple[str, ...]:
    normalized = rel.rstrip("/")
    root = _resolve_repo_member(repo, normalized)
    if not root.is_dir() or root.is_symlink():
        return (normalized,)
    canonical_repo = repo.resolve()
    expanded: list[str] = []
    for current, dirnames, filenames in os.walk(root, topdown=True, followlinks=False):
        current_path = Path(current)
        safe_dirs: list[str] = []
        for name in sorted(dirnames):
            candidate = current_path / name
            if name == ".git" or candidate.is_symlink():
                continue
            safe_dirs.append(name)
        dirnames[:] = safe_dirs
        for name in sorted(filenames):
            candidate = current_path / name
            if candidate.is_symlink() or not candidate.is_file():
                raise StateIdentityError(f"unsupported path type inside untracked tree: {candidate}")
            expanded.append(candidate.resolve().relative_to(canonical_repo).as_posix())
            if len(expanded) > MAX_EXPANDED_UNTRACKED_FILES:
                raise StateIdentityError(f"untracked directory exceeds {MAX_EXPANDED_UNTRACKED_FILES} files: {normalized}")
    if not expanded:
        raise StateIdentityError(f"untracked directory contains no regular product files: {normalized}")
    return tuple(expanded)


def canonical_dirty_paths(repo: Path) -> tuple[str, ...]:
    repo = repo.resolve()
    commands: tuple[tuple[str, ...], ...] = (
        ("diff", "--name-only", "-z", "HEAD"),
        ("diff", "--cached", "--name-only", "-z", "HEAD"),
        ("ls-files", "--others", "--exclude-standard", "-z"),
    )
    paths: set[str] = set()
    for command in commands:
        for rel in _split_nul(_run_git_bytes(repo, *command)):
            if rel.endswith("/"):
                paths.update(expand_collapsed_untracked_directory(repo, rel))
            else:
                _resolve_repo_member(repo, rel)
                paths.add(rel)
    return tuple(sorted(paths))


def all_repository_files(repo: Path) -> tuple[str, ...]:
    """Return tracked plus valid untracked files as individual canonical paths."""
    repo = repo.resolve()
    paths: set[str] = set()
    for rel in _split_nul(_run_git_bytes(repo, "ls-files", "--cached", "--others", "--exclude-standard", "-z")):
        if rel.endswith("/"):
            paths.update(expand_collapsed_untracked_directory(repo, rel))
        else:
            candidate = _resolve_repo_member(repo, rel)
            if candidate.is_file() and not candidate.is_symlink():
                paths.add(rel)
    return tuple(sorted(paths))
