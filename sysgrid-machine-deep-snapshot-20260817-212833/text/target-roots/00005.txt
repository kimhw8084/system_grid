from __future__ import annotations

import hashlib
import importlib.util
from importlib.machinery import SourceFileLoader
import json
import os
import sys
from pathlib import Path
from typing import Mapping

import sg_runner

CLI_VERSION = "fresh-text-1.0.0"
FASTPATH_CLI_SHA256 = "377bf5ddc09212c39217f7d0323a5f9c878aae2bd91323ef8c2333de4368e404"
BACKUP_NAME = "sg_cli.py.fastpath.rollback"


class CliBindingError(RuntimeError):
    pass


def _load_host():
    backup = Path(__file__).with_name(BACKUP_NAME)
    if not backup.is_file():
        raise CliBindingError("qualified FastPath CLI backup missing")
    if hashlib.sha256(backup.read_bytes()).hexdigest() != FASTPATH_CLI_SHA256:
        raise CliBindingError("qualified FastPath CLI backup identity mismatch")
    loader = SourceFileLoader("sysgrid_fastpath_cli_host", str(backup))
    spec = importlib.util.spec_from_loader("sysgrid_fastpath_cli_host", loader)
    if spec is None or spec.loader is None:
        raise CliBindingError("cannot load qualified FastPath CLI backup")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    required = ("load_config", "keychain_token", "issue_state", "config_paths", "post_result")
    for name in required:
        if not callable(getattr(module, name, None)):
            raise CliBindingError(f"FastPath CLI host missing {name}")
    return module


def _active_count(snapshot: object) -> int:
    count = 0
    for comment in sg_runner._legacy_collect_comments(snapshot):
        body = comment.get("body")
        if not isinstance(body, str) or sg_runner.COMMAND_START not in body:
            continue
        try:
            obj = sg_runner._extract_json(body, sg_runner.COMMAND_START, sg_runner.COMMAND_END)
        except Exception:
            continue
        if obj.get("classification") == "SYSGRID_VNEXT_COMMAND" and obj.get("status") == "ACTIVE":
            count += 1
    return count


def qualify() -> dict:
    host = _load_host()
    cfg = host.load_config()
    token = [REDACTED]
    issue = host.issue_state(cfg, token)
    if not isinstance(issue, Mapping) or not isinstance(issue.get("id"), str):
        raise CliBindingError("Linear issue snapshot unavailable")
    _control_repo, deps = host.config_paths(cfg)
    deps = Path(deps).expanduser().resolve()
    if not deps.is_dir():
        raise CliBindingError("configured dependency source path missing")
    ident = sg_runner.identity()
    required = ("redact", "extract_bundle", "execute", "print_result", "execute_text_snapshot")
    for name in required:
        if not callable(getattr(sg_runner, name, None)):
            raise CliBindingError(f"fresh runner missing {name}")
    return {
        "status": "QUALIFIED",
        "cli_version": CLI_VERSION,
        "cli_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "fastpath_cli_sha256": FASTPATH_CLI_SHA256,
        "runner_sha256": ident["runner_sha256"],
        "adapter_sha256": ident["text_adapter_sha256"],
        "fastpath_runner_sha256": ident["fastpath_rollback_sha256"],
        "issue": issue["id"],
        "active_command_count": _active_count(issue),
        "dependency_source_path": str(deps),
    }


def run_active() -> int:
    host = _load_host()
    cfg = host.load_config()
    token = [REDACTED]
    issue = host.issue_state(cfg, token)
    if not isinstance(issue, Mapping) or not isinstance(issue.get("id"), str):
        raise CliBindingError("Linear issue snapshot unavailable")
    if _active_count(issue) != 1:
        raise CliBindingError(f"expected exactly one ACTIVE command, found {_active_count(issue)}")
    _control_repo, deps = host.config_paths(cfg)
    result, _command, raw_command = sg_runner.execute_text_snapshot(
        issue_snapshot=issue,
        repo=Path(deps).expanduser().resolve(),
    )
    host.post_result(cfg, token, issue["id"], result, raw_command)
    after = host.issue_state(cfg, token)
    if _active_count(after) != 0:
        raise CliBindingError("result posted but ACTIVE command was not consumed")
    sg_runner.print_result(result)
    return 0


def _delegate(argv: list[str]) -> int:
    backup = Path(__file__).with_name(BACKUP_NAME)
    if not backup.is_file():
        raise CliBindingError("qualified FastPath CLI backup missing")
    os.execv(sys.executable, [sys.executable, str(backup), *argv])
    return 127


def main(argv: list[str] | None = None) -> int:
    args = list(sys.argv[1:] if argv is None else argv)
    if not args:
        return run_active()
    if args == ["qualify"]:
        print(json.dumps(qualify(), sort_keys=True))
        return 0
    return _delegate(args)


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except Exception as exc:
        print("SysGrid delivery failed.", file=sys.stderr)
        print("Cause: " + sg_runner.redact(f"{type(exc).__name__}: {exc}")[:1200].replace("\n", " | "), file=sys.stderr)
        raise SystemExit(1)
