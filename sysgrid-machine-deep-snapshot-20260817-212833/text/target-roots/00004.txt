from __future__ import annotations

import hashlib
import io
import json
import re
import zipfile
from dataclasses import dataclass
from typing import Callable, Iterable, Mapping, Sequence

VERSION = "v0.1"
MEMBER_ORDER = ("MANIFEST.json", "CHANGE.patch", "VERIFY.json", "MANIFEST.sha256")
PAYLOAD_START = "SYSGRID_TEXT_MEMBER_PAYLOAD_START"
PAYLOAD_END = "SYSGRID_TEXT_MEMBER_PAYLOAD_END"
HEADER_RE = re.compile(
    r"\ASYSGRID_TEXT_CAPSULE_MEMBER (?P<version>v\d+\.\d+)\n"
    r"member: (?P<member>[^\n]+)\n"
    r"size: (?P<size>[0-9]+)\n"
    r"sha256: (?P<sha>[0-9a-f]{64})\n"
    r"SYSGRID_TEXT_MEMBER_PAYLOAD_START\n",
    re.ASCII,
)
SHA_LINE_RE = re.compile(r"\A([0-9a-f]{64})  ([^\n]+)\Z", re.ASCII)


class TextCapsuleError(RuntimeError):
    code = "TEXT_CAPSULE_ERROR"


class TransportUnavailable(TextCapsuleError):
    code = "TRANSPORT_UNAVAILABLE"


class TransportAuthenticationError(TextCapsuleError):
    code = "TRANSPORT_AUTHENTICATION_ERROR"


class IntegrityError(TextCapsuleError):
    code = "INTEGRITY_ERROR"


class SchemaError(TextCapsuleError):
    code = "SCHEMA_ERROR"


@dataclass(frozen=True)
class TextMember:
    name: str
    payload: bytes
    sha256: str


@dataclass(frozen=True)
class Capsule:
    members: Mapping[str, TextMember]
    zip_bytes: bytes
    zip_sha256: str
    zip_size: int


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def parse_member_comment(body: str) -> TextMember:
    if "\r" in body:
        raise IntegrityError("CR bytes are forbidden; Linear text members must preserve canonical LF bytes")
    if body.count(PAYLOAD_START) != 1 or body.count(PAYLOAD_END) != 1:
        raise SchemaError("member must contain exactly one start marker and one end marker")
    match = HEADER_RE.match(body)
    if not match:
        raise SchemaError("member header is malformed or reordered")
    if match.group("version") != VERSION:
        raise SchemaError(f"unsupported text capsule version: {match.group('version')}")
    member = match.group("member")
    if member not in MEMBER_ORDER:
        raise SchemaError(f"unknown member: {member}")
    declared_size = int(match.group("size"))
    declared_sha = match.group("sha")
    suffix = "\n" + PAYLOAD_END
    if not body.endswith(suffix):
        raise SchemaError("payload end marker must terminate the comment")
    payload_text = body[match.end(): -len(suffix)]
    payload = payload_text.encode("utf-8")
    if len(payload) != declared_size:
        raise IntegrityError(f"size mismatch for {member}: {len(payload)} != {declared_size}")
    actual_sha = sha256_bytes(payload)
    if actual_sha != declared_sha:
        raise IntegrityError(f"sha256 mismatch for {member}: {actual_sha} != {declared_sha}")
    return TextMember(member, payload, declared_sha)


def collect_members(comments: Iterable[Mapping[str, object]], root_id: str) -> dict[str, TextMember]:
    members: dict[str, TextMember] = {}
    for c in comments:
        if c.get("parentId") != root_id:
            continue
        body = c.get("body")
        if not isinstance(body, str) or not body.startswith("SYSGRID_TEXT_CAPSULE_MEMBER "):
            continue
        member = parse_member_comment(body)
        if member.name in members:
            raise SchemaError(f"duplicate member: {member.name}")
        members[member.name] = member
    missing = [name for name in MEMBER_ORDER if name not in members]
    if missing:
        raise SchemaError(f"missing members: {missing}")
    if set(members) != set(MEMBER_ORDER):
        raise SchemaError("member cardinality mismatch")
    return members


def _parse_checksum_manifest(payload: bytes) -> dict[str, str]:
    try:
        text = payload.decode("ascii")
    except UnicodeDecodeError as exc:
        raise SchemaError("MANIFEST.sha256 must be ASCII") from exc
    if text.endswith("\n"):
        text = text[:-1]
    lines = text.split("\n") if text else []
    if len(lines) != 3:
        raise SchemaError("MANIFEST.sha256 must contain exactly three entries")
    expected_names = MEMBER_ORDER[:3]
    result: dict[str, str] = {}
    for line, expected_name in zip(lines, expected_names):
        m = SHA_LINE_RE.match(line)
        if not m or m.group(2) != expected_name:
            raise SchemaError("MANIFEST.sha256 order/format mismatch")
        result[expected_name] = m.group(1)
    return result


def validate_cross_bindings(members: Mapping[str, TextMember]) -> None:
    checksums = _parse_checksum_manifest(members["MANIFEST.sha256"].payload)
    for name in MEMBER_ORDER[:3]:
        if checksums[name] != members[name].sha256:
            raise IntegrityError(f"checksum manifest disagreement for {name}")

    try:
        manifest = json.loads(members["MANIFEST.json"].payload)
        verify = json.loads(members["VERIFY.json"].payload)
    except (UnicodeDecodeError, json.JSONDecodeError) as exc:
        raise SchemaError("MANIFEST/VERIFY JSON parse failed") from exc

    if not isinstance(manifest, dict) or manifest.get("schema_version") != "1.3.0":
        raise SchemaError("MANIFEST schema_version mismatch")
    required_manifest = {
        "approval_lock", "attempt_branch", "base_sha", "changed_paths", "classification",
        "expected_attempt_sha", "issue", "mode", "patch_file", "patch_id", "patch_sha256",
        "product_attempt", "replay_id", "repository", "schema_version", "verify_file",
    }
    if set(manifest) != required_manifest:
        raise SchemaError("MANIFEST keys differ from accepted v1.3.0 wire shape")
    if manifest["classification"] != "SYSGRID_PRODUCT_PATCH":
        raise SchemaError("MANIFEST classification mismatch")
    if manifest["patch_file"] != "CHANGE.patch" or manifest["verify_file"] != "VERIFY.json":
        raise SchemaError("MANIFEST member pointers mismatch")
    if manifest["patch_sha256"] != members["CHANGE.patch"].sha256:
        raise IntegrityError("MANIFEST patch_sha256 disagreement")
    if manifest["base_sha"] != manifest["expected_attempt_sha"]:
        raise IntegrityError("MANIFEST base/expected_attempt disagreement")

    if not isinstance(verify, dict) or set(verify) != {"actions", "schema_version"}:
        raise SchemaError("VERIFY keys differ from accepted v1.3.0 wire shape")
    if verify["schema_version"] != "1.3.0":
        raise SchemaError("VERIFY schema_version mismatch")
    actions = verify["actions"]
    if not isinstance(actions, list) or not actions:
        raise SchemaError("VERIFY actions must be a non-empty list")
    for action in actions:
        if not isinstance(action, dict) or set(action) != {"adapter", "id", "paths", "timeout_seconds"}:
            raise SchemaError("VERIFY action keys mismatch")
        if not isinstance(action["adapter"], str) or not isinstance(action["id"], str):
            raise SchemaError("VERIFY action strings invalid")
        if not isinstance(action["paths"], list) or not action["paths"] or not all(isinstance(p, str) for p in action["paths"]):
            raise SchemaError("VERIFY paths invalid")
        if not isinstance(action["timeout_seconds"], int) or isinstance(action["timeout_seconds"], bool):
            raise SchemaError("VERIFY timeout invalid")


def build_canonical_zip(members: Mapping[str, TextMember]) -> bytes:
    validate_cross_bindings(members)
    out = io.BytesIO()
    with zipfile.ZipFile(out, mode="w", compression=zipfile.ZIP_DEFLATED, compresslevel=6) as zf:
        for name in MEMBER_ORDER:
            info = zipfile.ZipInfo(name, date_time=(1980, 1, 1, 0, 0, 0))
            info.compress_type = zipfile.ZIP_DEFLATED
            info.create_system = 3
            info.external_attr = 0x81A40000
            zf.writestr(info, members[name].payload, compress_type=zipfile.ZIP_DEFLATED, compresslevel=6)
    return out.getvalue()


def reconstruct_from_comments(
    comments: Sequence[Mapping[str, object]],
    root_id: str,
    *,
    expected_zip_sha256: str,
    expected_zip_size: int,
) -> Capsule:
    members = collect_members(comments, root_id)
    zip_bytes = build_canonical_zip(members)
    actual_size = len(zip_bytes)
    actual_sha = sha256_bytes(zip_bytes)
    if actual_size != expected_zip_size:
        raise IntegrityError(f"canonical ZIP size mismatch: {actual_size} != {expected_zip_size}")
    if actual_sha != expected_zip_sha256:
        raise IntegrityError(f"canonical ZIP sha256 mismatch: {actual_sha} != {expected_zip_sha256}")
    return Capsule(members, zip_bytes, actual_sha, actual_size)


def resolve_exact_capsule(
    primary: Callable[[], bytes],
    mirror: Callable[[], bytes] | None,
    *,
    expected_sha256: str,
    expected_size: int,
) -> bytes:
    """Fallback is permitted only on primary transport unavailability.

    Integrity/auth/schema disagreement from primary is terminal and must never trigger mirror use.
    """
    try:
        data = primary()
    except TransportUnavailable:
        if mirror is None:
            raise
        data = mirror()
    if len(data) != expected_size or sha256_bytes(data) != expected_sha256:
        raise IntegrityError("resolved capsule does not match approved content identity")
    return data
