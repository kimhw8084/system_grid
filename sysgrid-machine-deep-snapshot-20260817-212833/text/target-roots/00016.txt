#!/usr/bin/env python3
from __future__ import annotations

import argparse
import json
import os
import pathlib
import signal
import subprocess
import sys
import time


def detach_process_session() -> None:
    try:
        os.setsid()
    except (OSError, AttributeError):
        pass
    for sig in (signal.SIGHUP, signal.SIGINT, signal.SIGTERM):
        try:
            signal.signal(sig, signal.SIG_DFL)
        except (OSError, ValueError):
            pass


def newest_engine_zip(outbox: pathlib.Path, started_epoch: float) -> pathlib.Path | None:
    candidate = outbox / "UPLOAD_THIS_TO_CHATGPT.zip"
    if candidate.is_file() and candidate.stat().st_mtime >= started_epoch - 2:
        return candidate
    return None


def notify(message: str) -> None:
    notifier = pathlib.Path.home() / ".local" / "bin" / "sgnotify"
    if notifier.is_file() and os.access(notifier, os.X_OK):
        subprocess.run(
            [str(notifier), message],
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            timeout=10,
            check=False,
        )


def initialize_launch(args: argparse.Namespace, launch_path: pathlib.Path) -> dict:
    required = {
        "launch_id": args.launch_id,
        "repo": args.repo,
        "pre_head": args.pre_head,
        "started_epoch": args.started_epoch,
        "control_root": args.control_root,
        "core_sgwatch": args.core_sgwatch,
        "packager": args.packager,
        "payload_dir": args.payload_dir,
    }
    missing = [name for name, value in required.items() if value is None]
    if missing:
        raise ValueError(f"missing initialization fields: {', '.join(missing)}")
    launch = {
        "schema_version": "1.1",
        "continuity_version": "1.0.4",
        "launch_id": args.launch_id,
        "repo": args.repo,
        "pre_head": args.pre_head,
        "started_epoch": float(args.started_epoch),
        "control_root": args.control_root,
        "core_sgwatch": args.core_sgwatch,
        "packager": args.packager,
        "watcher": str(pathlib.Path(__file__).resolve()),
        "payload_dir": args.payload_dir,
        "foreground": args.foreground == "1",
        "argument_count": int(args.argument_count or 0),
        "watcher_start_policy": "prelaunch-independent-session",
    }
    launch_path.parent.mkdir(parents=True, exist_ok=True)
    temporary = launch_path.with_suffix(launch_path.suffix + f".{os.getpid()}.tmp")
    temporary.write_text(json.dumps(launch, indent=2) + "\n", encoding="utf-8")
    os.replace(temporary, launch_path)
    return launch


def main() -> int:
    detach_process_session()
    parser = argparse.ArgumentParser()
    parser.add_argument("launch_json")
    parser.add_argument("--initialize", action="store_true")
    parser.add_argument("--launch-id")
    parser.add_argument("--repo")
    parser.add_argument("--pre-head")
    parser.add_argument("--started-epoch")
    parser.add_argument("--control-root")
    parser.add_argument("--core-sgwatch")
    parser.add_argument("--packager")
    parser.add_argument("--payload-dir")
    parser.add_argument("--foreground")
    parser.add_argument("--argument-count")
    parser.add_argument("--already-finished", action="store_true")
    parser.add_argument("--launch-reason", default="normal")
    args = parser.parse_args()

    launch_path = pathlib.Path(args.launch_json)
    if args.initialize:
        launch = initialize_launch(args, launch_path)
    else:
        launch = json.loads(launch_path.read_text(encoding="utf-8"))

    core_sgwatch = pathlib.Path(launch["core_sgwatch"])
    control_root = pathlib.Path(launch["control_root"])
    outbox = control_root / "Outbox"
    outbox.mkdir(parents=True, exist_ok=True)
    log_path = control_root / "State" / f"continuity-watcher-{launch['launch_id']}.log"
    log_path.parent.mkdir(parents=True, exist_ok=True)

    with log_path.open("a", encoding="utf-8") as log:
        log.write(f"watcher_start={time.time()}\n")
        log.write(f"launch_reason={args.launch_reason}\n")
        if not args.already_finished:
            time.sleep(1.0)
            try:
                cp = subprocess.run(
                    [str(core_sgwatch)],
                    stdout=log,
                    stderr=log,
                    timeout=4 * 60 * 60,
                    check=False,
                )
                log.write(f"core_sgwatch_exit={cp.returncode}\n")
            except subprocess.TimeoutExpired:
                log.write("core_sgwatch_timeout_seconds=14400\n")

        deadline = time.time() + 120
        raw = None
        while time.time() < deadline:
            raw = newest_engine_zip(outbox, float(launch["started_epoch"]))
            if raw:
                break
            time.sleep(1)

        if raw:
            raw_copy = control_root / "State" / f"engine-review-{launch['launch_id']}.zip"
            raw_copy.write_bytes(raw.read_bytes())
            raw_arg = str(raw_copy)
        else:
            raw_arg = ""
            log.write("raw_engine_zip_missing=true\n")

        cmd = [
            sys.executable,
            launch["packager"],
            "--repo", launch["repo"],
            "--control-root", launch["control_root"],
            "--pre-head", launch.get("pre_head", "unknown"),
            "--launch-json", str(launch_path),
            "--payload-dir", launch["payload_dir"],
        ]
        if raw_arg:
            cmd += ["--raw-zip", raw_arg]
        cp = subprocess.run(cmd, stdout=log, stderr=log, check=False)
        log.write(f"packager_exit={cp.returncode}\n")
        if cp.returncode == 0:
            notify(
                "✅ SysGrid continuity bundle ready\n"
                "Outbox: UPLOAD_THIS_TO_CHATGPT.zip\n"
                "Current handoff: SysGrid-Control/Continuity/Current"
            )
        else:
            notify(
                "❌ SysGrid continuity packaging failed\n"
                "Check SysGrid-Control/State continuity watcher log"
            )
        return cp.returncode


if __name__ == "__main__":
    raise SystemExit(main())
