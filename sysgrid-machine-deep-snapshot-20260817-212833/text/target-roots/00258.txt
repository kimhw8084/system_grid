class CampaignError(RuntimeError):

def sha256_bytes(data: bytes) -> str: return hashlib.sha256(data).hexdigest()
def sha256_file(path: Path) -> str:
    h=hashlib.sha256()
    with path.open('rb') as f:
        for chunk in iter(lambda:f.read(1024*1024), b''): h.update(chunk)
    return h.hexdigest()


def safe_member(name: str) -> PurePosixPath:
    if not isinstance(name,str) or not name or name.startswith('/') or name.startswith('./') or '\\' in name or '\0' in name or '//' in name:
        raise CampaignError('CAMPAIGN_PACKAGE_FAILURE', f'unsafe member path: {name!r}')
    p=PurePosixPath(name)
    if any(part in {'','.', '..'} for part in p.parts):
        raise CampaignError('CAMPAIGN_PACKAGE_FAILURE', f'unsafe member path: {name!r}')
    return p

def read_manifest(zf: zipfile.ZipFile) -> dict[str,str]:
    try: raw=zf.read('MANIFEST.sha256').decode('utf-8','strict')
    except Exception as exc: raise CampaignError('CAMPAIGN_MANIFEST_FAILURE', f'invalid MANIFEST.sha256: {exc}') from exc
    expected: dict[str,str]={}
    for line in raw.splitlines():
        if not line.strip(): continue
        parts=line.split(None,1)
        if len(parts)!=2 or len(parts[0])!=64: raise CampaignError('CAMPAIGN_MANIFEST_FAILURE', f'invalid manifest line: {line!r}')
        rel=parts[1].lstrip('* '); safe_member(rel)
        if rel=='MANIFEST.sha256' or rel in expected: raise CampaignError('CAMPAIGN_MANIFEST_FAILURE', f'duplicate/invalid manifest path: {rel}')
        expected[rel]=parts[0]
    actual=sorted(i.filename for i in zf.infolist() if not i.is_dir() and i.filename!='MANIFEST.sha256')
    if sorted(expected)!=actual:
        raise CampaignError('CAMPAIGN_MANIFEST_FAILURE', f'manifest paths differ missing={sorted(set(expected)-set(actual))} extra={sorted(set(actual)-set(expected))}')
    for rel,digest in expected.items():
        got=sha256_bytes(zf.read(rel))
        if got!=digest: raise CampaignError('CAMPAIGN_MANIFEST_FAILURE', f'digest mismatch for {rel}')
    return expected


def load_json(zf: zipfile.ZipFile, name: str) -> dict[str,Any]:
    try: value=json.loads(zf.read(name).decode('utf-8','strict'))
    except Exception as exc: raise CampaignError('CAMPAIGN_SCHEMA_FAILURE', f'{name} invalid: {exc}') from exc
    if not isinstance(value,dict): raise CampaignError('CAMPAIGN_SCHEMA_FAILURE', f'{name} must contain an object')
    return value

def select_campaign(explicit: Path|None) -> Path:
    if explicit: return explicit.expanduser().resolve()
    inbox=control_root()/'Campaign-Inbox'; inbox.mkdir(parents=True,exist_ok=True)
    candidates=sorted([p for p in inbox.iterdir() if p.is_file() and not p.is_symlink() and p.suffix.lower() in {'.sgcap','.zip'} and not p.name.startswith('.')])
    if len(candidates)!=1: raise CampaignError('CAMPAIGN_SELECTION_FAILURE',f'Campaign-Inbox must contain exactly one package; found {len(candidates)}',66)
    return candidates[0]


def install_capsule(path: Path, report: dict[str,Any]) -> Path:
    inbox=control_root()/'Inbox'; inbox.mkdir(parents=True,exist_ok=True)
    for existing in inbox.glob('*.sgcap'): existing.unlink()
    name=path.name if path.suffix.lower()=='.sgcap' else path.stem+'.sgcap'
    destination=inbox/name
    shutil.copy2(path,destination)
    if sha256_file(destination)!=report['sha256']: raise CampaignError('CAMPAIGN_INSTALL_FAILURE','installed capsule checksum mismatch',74)
    return destination


def write_failure(exc: CampaignError) -> Path:
    out=control_root()/'Outbox'; out.mkdir(parents=True,exist_ok=True)
    path=out/'UPLOAD_DIAGNOSTIC_TO_CHATGPT.zip'
    with tempfile.TemporaryDirectory(prefix='sysgrid-campaign-failure-') as td:
        root=Path(td)
        (root/'RESULT.json').write_text(json.dumps({'result':'FAIL','classification':exc.classification,'message':str(exc),'controller_version':VERSION},indent=2,sort_keys=True)+'\n')
        lines=[]
        for p in sorted(root.rglob('*')):
            if p.is_file(): lines.append(f'{sha256_file(p)}  {p.relative_to(root).as_posix()}')
        (root/'MANIFEST.sha256').write_text('\n'.join(lines)+'\n')
        with zipfile.ZipFile(path,'w',zipfile.ZIP_DEFLATED,compresslevel=9) as zf:
            for p in sorted(root.rglob('*')):
                if p.is_file(): zf.write(p,p.relative_to(root).as_posix())
    return path


def cmd_validate(args: argparse.Namespace) -> int:
    try:
        path=select_campaign(args.package); report=validate_campaign(path, platform_preflight=True)
        print(json.dumps({'result':'PASS','controller_version':VERSION,**report},indent=2,sort_keys=True)); return 0
    except CampaignError as exc:
        diagnostic=write_failure(exc); print(json.dumps({'result':'FAIL','classification':exc.classification,'message':str(exc),'diagnostic':str(diagnostic)},indent=2)); return exc.exit_code


def cmd_run(args: argparse.Namespace) -> int:
    try:
        path=select_campaign(args.package); report=validate_campaign(path, platform_preflight=True); installed=install_capsule(path,report)
        command=[sys.executable,str(supervisor_path()),'run',str(installed)]
        completed=subprocess.run(command,check=False)
        return completed.returncode
    except CampaignError as exc:
        diagnostic=write_failure(exc)
        print(f'SysGrid campaign blocked: {exc.classification}\n{exc}\nDiagnostic: {diagnostic}',file=sys.stderr)
        return exc.exit_code


def cmd_status(_args: argparse.Namespace) -> int:
    return subprocess.run([sys.executable,str(supervisor_path()),'status'],check=False).returncode


def cmd_self_test(_args: argparse.Namespace) -> int:
    failures=[]
    try:
        safe_member('QUESTIONNAIRE.json')
        for bad in ('../x','/x','./x','a\\b'):
            try: safe_member(bad); failures.append(f'unsafe path accepted: {bad}')
            except CampaignError: pass
    except Exception as exc: failures.append(f'{type(exc).__name__}: {exc}')
    print(json.dumps({'controller_version':VERSION,'result':'FAIL' if failures else 'PASS','failures':failures},indent=2,sort_keys=True)); return 1 if failures else 0


def main(argv=None) -> int:
    parser=argparse.ArgumentParser(prog='sysgrid-campaign')
    sub=parser.add_subparsers(dest='command',required=True)
    run=sub.add_parser('run'); run.add_argument('package',nargs='?',type=Path)
    validate=sub.add_parser('validate'); validate.add_argument('package',nargs='?',type=Path)
    sub.add_parser('status'); sub.add_parser('self-test')
    args=parser.parse_args(argv)
    return {'run':cmd_run,'validate':cmd_validate,'status':cmd_status,'self-test':cmd_self_test}[args.command](args)

if __name__=='__main__': raise SystemExit(main())
