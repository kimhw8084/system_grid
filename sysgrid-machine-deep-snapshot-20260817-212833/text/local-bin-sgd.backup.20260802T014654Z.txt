def run(
    command: Sequence[str],
    *,
    cwd: Path | None = None,
    check: bool = False,
    text: bool = True,
    capture_output: bool = True,
    env: dict[str, str] | None = None,
    timeout: float | None = None,
) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        list(command),
        cwd=str(cwd) if cwd else None,
        check=check,
        text=text,
        capture_output=capture_output,
        env=env,
        timeout=timeout,
    )


def command_exists(name: str) -> bool:
    return shutil.which(name) is not None


def git_output(repo: Path, *args: str, default: str = "") -> str:
    result = run(["git", *args], cwd=repo)
    if result.returncode != 0:
        return default
    return result.stdout.strip()


def discover_repo(explicit: str | None = None) -> Path:
    candidates: list[Path] = []
    if explicit:
        candidates.append(Path(explicit).expanduser())
    env_repo = os.environ.get("SYSGRID_REPO")
    if env_repo:
        candidates.append(Path(env_repo).expanduser())
    candidates.append(Path.cwd())

    seen: set[Path] = set()
    for candidate in candidates:
        candidate = candidate.resolve()
        for current in [candidate, *candidate.parents]:
            if current in seen:
                continue
            seen.add(current)
            if (current / ".git").exists() and (current / "backend").is_dir() and (current / "frontend").is_dir():
                return current
            result = run(["git", "-C", str(current), "rev-parse", "--show-toplevel"])
            if result.returncode == 0:
                root = Path(result.stdout.strip()).resolve()
                if (root / "backend").is_dir() and (root / "frontend").is_dir():
                    return root
    raise SystemExit(
        "SysGrid repository not found. Run inside the repository or set SYSGRID_REPO=/absolute/path/to/SysGrid."
    )


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def session_exists(name: str) -> bool:
    if not command_exists("tmux"):
        return False
    return run(["tmux", "has-session", "-t", name]).returncode == 0


def kill_session(name: str) -> bool:
    if not session_exists(name):
        return False
    return run(["tmux", "kill-session", "-t", name]).returncode == 0


def listener_pids(port: int) -> list[int]:
    if not command_exists("lsof"):
        return []
    result = run(["lsof", "-tiTCP:%d" % port, "-sTCP:LISTEN"])
    pids: list[int] = []
    if result.returncode not in (0, 1):
        return pids
    for token in result.stdout.split():
        try:
            pids.append(int(token))
        except ValueError:
            continue
    return sorted(set(pids))


def terminate_pid(pid: int, timeout: float = 3.0) -> None:
    try:
        os.kill(pid, signal.SIGTERM)
    except ProcessLookupError:
        return
    deadline = time.monotonic() + timeout
    while time.monotonic() < deadline:
        try:
            os.kill(pid, 0)
        except ProcessLookupError:
            return
        time.sleep(0.1)
    with contextlib.suppress(ProcessLookupError):
        os.kill(pid, signal.SIGKILL)


def stop_runtime_ports(ports: Iterable[int]) -> dict[str, Any]:
    stopped: dict[str, list[int]] = {}
    for port in ports:
        pids = listener_pids(port)
        for pid in pids:
            terminate_pid(pid)
        stopped[str(port)] = pids
    return {"stopped": stopped, "remaining": {str(port): listener_pids(port) for port in ports}}


@dataclass
class ProbeResult:
    name: str
    url: str
    status: int
    ok: bool
    duration_ms: int
    request_id: str | None
    tenant_id: str | None
    content_type: str | None
    body_preview: str
    error: str | None


def http_get(url: str, headers: dict[str, str] | None = None, timeout_seconds: float = 8.0) -> ProbeResult:
    started = time.monotonic()
    request = urllib.request.Request(url, headers=headers or {}, method="GET")
    try:
        with urllib.request.urlopen(request, timeout=timeout_seconds) as response:
            body = response.read(16384)
            status = int(response.status)
            response_headers = response.headers
            error = None
    except urllib.error.HTTPError as exc:
        body = exc.read(16384)
        status = int(exc.code)
        response_headers = exc.headers
        error = f"HTTP {exc.code}"
    except (urllib.error.URLError, TimeoutError, socket.timeout, ConnectionError, OSError) as exc:
        return ProbeResult(
            name="",
            url=url,
            status=0,
            ok=False,
            duration_ms=int((time.monotonic() - started) * 1000),
            request_id=None,
            tenant_id=None,
            content_type=None,
            body_preview="",
            error=f"{type(exc).__name__}: {exc}",
        )
    preview = body.decode("utf-8", errors="replace")[:4000]
    return ProbeResult(
        name="",
        url=url,
        status=status,
        ok=200 <= status < 300,
        duration_ms=int((time.monotonic() - started) * 1000),
        request_id=response_headers.get("x-request-id"),
        tenant_id=response_headers.get("x-sysgrid-tenant-id"),
        content_type=response_headers.get("content-type"),
        body_preview=preview,
        error=error,
    )


def selected_tenant_id(body: str) -> str | None:
    try:
        value = json.loads(body)
    except json.JSONDecodeError:
        return None
    if not isinstance(value, list):
        return None
    selected = [row for row in value if isinstance(row, dict) and row.get("is_selected")]
    candidate = selected[0] if selected else (value[0] if value else None)
    if isinstance(candidate, dict) and candidate.get("id") is not None:
        return str(candidate["id"])
    return None


def classify_smoke(probes: list[ProbeResult]) -> str:
    if not probes:
        return "NO_PROBES"
    if probes[0].status == 0:
        return "BACKEND_UNAVAILABLE"
    if not probes[0].ok:
        return "BACKEND_HEALTH_FAILURE"
    failures = [probe for probe in probes[1:] if not probe.ok]
    if not failures:
        return "PASS"
    if any(probe.status == 0 for probe in failures):
        return "NETWORK_OR_PROCESS_FAILURE"
    if any(probe.status >= 500 for probe in failures):
        return "BACKEND_ENDPOINT_FAILURE"
    if any(probe.status in (401, 403) for probe in failures):
        return "IDENTITY_OR_PERMISSION_FAILURE"
    if any(probe.status == 404 for probe in failures):
        return "API_CONTRACT_ROUTE_MISSING"
    return "API_CONTRACT_FAILURE"

def metadata_path(name: str) -> Path:

def write_metadata(name: str, data: Any) -> None:
    path = metadata_path(name)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json_dump(data), encoding="utf-8")


def read_metadata(name: str) -> dict[str, Any] | None:
    path = metadata_path(name)
    if not path.exists():
        return None
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except (json.JSONDecodeError, OSError):
        return None


def nested_git_dirs(repo: Path) -> list[str]:
    found: list[str] = []
    for root, dirs, _files in os.walk(repo):
        root_path = Path(root)
        if root_path == repo / ".git" or ".git" in root_path.parts[len(repo.parts):]:
            dirs[:] = []
            continue
        if ".git" in dirs:
            nested = root_path / ".git"
            if root_path != repo:
                found.append(str(root_path.relative_to(repo)))
            dirs.remove(".git")
        dirs[:] = [name for name in dirs if name not in EXCLUDED_DIR_NAMES]
    return sorted(set(found))


def repo_identity(repo: Path) -> dict[str, Any]:
    status_porcelain = git_output(repo, "status", "--porcelain=v1", "--untracked-files=all")
    return {
        "schema_version": 1,
        "captured_utc": utc_now().isoformat(),
        "repo": str(repo),
        "git_root": git_output(repo, "rev-parse", "--show-toplevel"),
        "branch": git_output(repo, "branch", "--show-current"),
        "head": git_output(repo, "rev-parse", "HEAD"),
        "remote_origin": git_output(repo, "remote", "get-url", "origin"),
        "upstream": git_output(repo, "rev-parse", "--abbrev-ref", "--symbolic-full-name", "@{u}"),
        "status_lines": status_porcelain.splitlines() if status_porcelain else [],
        "dirty_count": len(status_porcelain.splitlines()) if status_porcelain else 0,
        "nested_git_directories": nested_git_dirs(repo),
        "start_local_sha256": sha256_file(repo / "scripts" / "start-local.sh") if (repo / "scripts" / "start-local.sh").is_file() else None,
        "verify_app_sha256": sha256_file(repo / "scripts" / "verify-app.sh") if (repo / "scripts" / "verify-app.sh").is_file() else None,
    }


def runtime_status(repo: Path | None = None) -> dict[str, Any]:
    health_url = f"http://{DEFAULT_BACKEND_HOST}:{DEFAULT_BACKEND_PORT}/api/v1/health"
    health = http_get(health_url, {"X-User-Id": DEFAULT_USER_ID}, timeout_seconds=2.0)
    return {
        "captured_utc": utc_now().isoformat(),
        "runtime_session": session_exists(RUNTIME_SESSION),
        "verify_session": session_exists(VERIFY_SESSION),
        "listeners": {
            str(DEFAULT_BACKEND_PORT): listener_pids(DEFAULT_BACKEND_PORT),
            str(DEFAULT_FRONTEND_PORT): listener_pids(DEFAULT_FRONTEND_PORT),
        },
        "health": asdict(health),
        "repo": str(repo) if repo else None,
    }


def doctor_report(repo: Path) -> dict[str, Any]:
    checks = {
        "git": command_exists("git"),
        "python3": command_exists("python3"),
        "tmux": command_exists("tmux"),
        "curl": command_exists("curl"),
        "lsof": command_exists("lsof"),
        "npm": command_exists("npm"),
        "start_local": (repo / "scripts" / "start-local.sh").is_file(),
        "verify_app": (repo / "scripts" / "verify-app.sh").is_file(),
        "backend_python": (repo / "backend" / "venv" / "bin" / "python").is_file(),
        "frontend_node_modules": (repo / "frontend" / "node_modules").is_dir(),
    }
    identity = repo_identity(repo)
    required = ["git", "python3", "tmux", "curl", "lsof", "npm", "start_local", "verify_app", "backend_python", "frontend_node_modules"]
    missing = [name for name in required if not checks[name]]
    return {
        "result": "PASS" if not missing else "FAIL",
        "checks": checks,
        "missing": missing,
        "identity": identity,
        "runtime": runtime_status(repo),
    }


def cmd_doctor(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    result = doctor_report(repo)
    write_metadata("doctor-latest", result)
    print(json_dump(result), end="")
    if result["identity"]["nested_git_directories"]:
        print("DELIVERY BLOCKER: nested Git directories detected: " + ", ".join(result["identity"]["nested_git_directories"]), file=sys.stderr)
    return 0 if result["result"] == "PASS" else 2


def cmd_stop(args: argparse.Namespace) -> int:
    _repo = discover_repo(args.repo)
    sessions = {
        RUNTIME_SESSION: kill_session(RUNTIME_SESSION),
        VERIFY_SESSION: kill_session(VERIFY_SESSION) if args.include_verify else False,
    }
    ports = stop_runtime_ports([DEFAULT_BACKEND_PORT, DEFAULT_FRONTEND_PORT])
    result = {"result": "PASS", "sessions_killed": sessions, **ports}
    write_metadata("stop-latest", result)
    print(json_dump(result), end="")
    remaining = any(ports["remaining"].values())
    return 1 if remaining else 0


def wait_for_backend(timeout_seconds: int, log_path: Path) -> ProbeResult:
    deadline = time.monotonic() + timeout_seconds
    url = f"http://{DEFAULT_BACKEND_HOST}:{DEFAULT_BACKEND_PORT}/api/v1/health"
    last = http_get(url, {"X-User-Id": DEFAULT_USER_ID}, timeout_seconds=2.0)
    while time.monotonic() < deadline:
        last = http_get(url, {"X-User-Id": DEFAULT_USER_ID}, timeout_seconds=2.0)
        if last.ok:
            return last
        if not session_exists(RUNTIME_SESSION):
            break
        time.sleep(1)
    if log_path.exists():
        tail = "\n".join(log_path.read_text(encoding="utf-8", errors="replace").splitlines()[-80:])
        last.body_preview = (last.body_preview + "\n--- runtime log tail ---\n" + tail)[-8000:]
    return last


def cmd_start(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    doctor = doctor_report(repo)
    write_metadata("doctor-latest", doctor)
    if doctor["result"] != "PASS":
        print(json_dump(doctor), file=sys.stderr, end="")
        print("Start refused because doctor failed.", file=sys.stderr)
        return 2

    kill_session(RUNTIME_SESSION)
    stop_runtime_ports([DEFAULT_BACKEND_PORT, DEFAULT_FRONTEND_PORT])
    log_path = runtime_log_path()
    log_path.parent.mkdir(parents=True, exist_ok=True)
    rotated = log_path.with_name(f"runtime-{timestamp()}.log")
    if log_path.exists():
        shutil.copy2(log_path, rotated)
    log_path.write_text("", encoding="utf-8")

    launcher = repo / "scripts" / "start-local.sh"
    command = [str(launcher)]
    if args.fast:
        command.append("--skip-typecheck")
    if args.no_seed_data:
        command.append("--no-seed-data")
    tmux_shell = (
        "set -o pipefail; "
        f"cd {shlex.quote(str(repo))}; "
        f"{shlex.join(command)} 2>&1 | tee {shlex.quote(str(log_path))}; "
        "status=${PIPESTATUS[0]}; "
        f"printf '%s\\n' \"$status\" > {shlex.quote(str(state_root() / 'runtime' / 'exit-code'))}; "
        "exit \"$status\""
    )
    started = run(["tmux", "new-session", "-d", "-s", RUNTIME_SESSION, "bash", "-lc", tmux_shell])
    if started.returncode != 0:
        print(started.stderr or "Failed to start tmux runtime session", file=sys.stderr)
        return 2

    health = wait_for_backend(args.timeout, log_path)
    result = {
        "result": "PASS" if health.ok else "FAIL",
        "classification": "RUNTIME_READY" if health.ok else "RUNTIME_START_FAILURE",
        "repo": str(repo),
        "session": RUNTIME_SESSION,
        "log": str(log_path),
        "health": asdict(health),
        "fast": args.fast,
        "seed_data": not args.no_seed_data,
    }
    write_metadata("start-latest", result)
    print(json_dump(result), end="")
    return 0 if health.ok else 1


def smoke_probe_matrix(api_origin: str, user_id: str, tenant_override: str | None = None) -> tuple[list[ProbeResult], str | None]:
    api_base = api_origin.rstrip("/") + "/api/v1"
    probes: list[ProbeResult] = []

    health = http_get(f"{api_base}/health", {"X-User-Id": user_id})
    health.name = "health"
    probes.append(health)
    if not health.ok:
        return probes, tenant_override

    tenants = http_get(f"{api_base}/tenants/me", {"X-User-Id": user_id})
    tenants.name = "tenants"
    probes.append(tenants)
    tenant_id = tenant_override or (selected_tenant_id(tenants.body_preview) if tenants.ok else None)
    headers = {"X-User-Id": user_id}
    if tenant_id:
        headers["X-Tenant-Id"] = tenant_id

    endpoints = [
        ("monitoring", "/monitoring?include_deleted=true"),
        ("assets", "/devices?include_deleted=true"),
        ("services", "/logical-services?include_deleted=true"),
        ("external", "/intelligence/entities?include_deleted=true"),
        ("network", "/networks/connections?include_deleted=true"),
        ("far", "/far/modes"),
        ("research", "/investigations?include_deleted=true"),
        ("vendors", "/vendors?include_deleted=true"),
        ("settings-options", "/settings/options"),
        ("workspace-definitions", "/workspaces/definitions"),
    ]
    for name, path in endpoints:
        probe = http_get(f"{api_base}{path}", headers)
        probe.name = name
        probes.append(probe)
    return probes, tenant_id


def cmd_smoke(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    probes, tenant_id = smoke_probe_matrix(args.api_origin, args.user_id, args.tenant_id)
    classification = classify_smoke(probes)
    result = {
        "schema_version": 1,
        "captured_utc": utc_now().isoformat(),
        "result": "PASS" if classification == "PASS" else "FAIL",
        "classification": classification,
        "repo_identity": repo_identity(repo),
        "api_origin": args.api_origin,
        "user_id": args.user_id,
        "tenant_id": tenant_id,
        "probes": [asdict(probe) for probe in probes],
    }
    path = state_root() / "smoke" / "latest.json"
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json_dump(result), encoding="utf-8")
    write_metadata("smoke-latest", result)
    print(json_dump(result), end="")
    return 0 if classification == "PASS" else 1


def cmd_status(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    probes, tenant_id = smoke_probe_matrix(args.api_origin, args.user_id, args.tenant_id)
    classification = classify_smoke(probes)
    status = {
        "result": "PASS" if classification == "PASS" else "FAIL",
        "classification": classification,
        "repo": repo_identity(repo),
        "runtime": runtime_status(repo),
        "tenant_id": tenant_id,
        "failed_probes": [asdict(probe) for probe in probes if not probe.ok],
        "latest_start": read_metadata("start-latest"),
        "latest_verify": read_metadata("verify-latest"),
    }
    if args.phone:
        print(f"SysGrid: {status['classification']}")
        print(f"HEAD: {status['repo']['head'][:12] if status['repo']['head'] else '<unknown>'} dirty={status['repo']['dirty_count']}")
        print(f"Runtime tmux={status['runtime']['runtime_session']} ports=8000:{bool(status['runtime']['listeners']['8000'])} 5173:{bool(status['runtime']['listeners']['5173'])}")
        for probe in status["failed_probes"][:4]:
            print(f"FAIL {probe['name']}: status={probe['status']} request_id={probe['request_id'] or '-'} {probe['error'] or ''}")
        print(f"Log: {runtime_log_path()}")
    else:
        print(json_dump(status), end="")
    write_metadata("status-latest", status)
    return 0 if classification == "PASS" else 1


def should_skip_source_path(relative: Path) -> tuple[bool, str | None]:
    parts = relative.parts
    if any(part in EXCLUDED_DIR_NAMES for part in parts[:-1]):
        return True, "excluded-directory"
    name = relative.name
    lower_name = name.lower()
    if name in SECRET_BASENAMES or lower_name.startswith(".env."):
        if lower_name.endswith(".example") or lower_name.endswith(".sample") or lower_name.endswith(".template"):
            return False, None
        return True, "secret-or-runtime-env"
    if relative.suffix.lower() in EXCLUDED_FILE_SUFFIXES:
        return True, "binary-or-sensitive-suffix"
    if any(token in lower_name for token in ("secret", "credential", "private_key", "access_token")):
        return True, "sensitive-name"
    return False, None


def is_probably_text(path: Path) -> bool:
    if path.suffix.lower() in SOURCE_EXTENSIONS or path.name in SOURCE_BASENAMES:
        return True
    try:
        sample = path.read_bytes()[:8192]
    except OSError:
        return False
    if b"\x00" in sample:
        return False
    try:
        sample.decode("utf-8")
        return True
    except UnicodeDecodeError:
        return False


def git_candidate_paths(repo: Path) -> list[Path]:
    result = run(["git", "ls-files", "--cached", "--others", "--exclude-standard", "-z"], cwd=repo, text=False)
    if result.returncode != 0:
        raise RuntimeError(result.stderr.decode("utf-8", errors="replace") if isinstance(result.stderr, bytes) else str(result.stderr))
    raw = result.stdout if isinstance(result.stdout, bytes) else result.stdout.encode()
    candidates: list[Path] = []
    for token in raw.split(b"\0"):
        if not token:
            continue
        relative = Path(token.decode("utf-8", errors="surrogateescape"))
        absolute = repo / relative
        if absolute.is_dir():
            for root, dirs, files in os.walk(absolute):
                root_path = Path(root)
                dirs[:] = [name for name in dirs if name not in EXCLUDED_DIR_NAMES]
                for file_name in files:
                    candidates.append((root_path / file_name).relative_to(repo))
        else:
            candidates.append(relative)
    return sorted(set(candidates), key=lambda p: p.as_posix())


def zip_write_bytes(archive: zipfile.ZipFile, name: str, data: bytes, when: dt.datetime) -> None:
    info = zipfile.ZipInfo(name)
    safe_year = min(max(when.year, 1980), 2107)
    info.date_time = (safe_year, when.month, when.day, when.hour, when.minute, when.second)
    info.compress_type = zipfile.ZIP_DEFLATED
    info.external_attr = 0o100644 << 16
    archive.writestr(info, data)


def create_source_snapshot(repo: Path, destination: Path) -> dict[str, Any]:
    captured = utc_now()
    identity = repo_identity(repo)
    candidates = git_candidate_paths(repo)
    included: list[dict[str, Any]] = []
    skipped: list[dict[str, Any]] = []
    manifest_lines: list[str] = []
    destination.parent.mkdir(parents=True, exist_ok=True)
    temp_destination = destination.with_suffix(destination.suffix + ".tmp")
    with zipfile.ZipFile(temp_destination, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
        for relative in candidates:
            absolute = repo / relative
            reason: str | None = None
            if absolute.is_symlink():
                reason = "symlink"
            elif not absolute.is_file():
                reason = "not-regular-file"
            else:
                skip, reason = should_skip_source_path(relative)
                if not skip:
                    try:
                        size = absolute.stat().st_size
                    except OSError:
                        reason = "stat-failed"
                    else:
                        if size > MAX_SOURCE_FILE_BYTES:
                            reason = f"larger-than-{MAX_SOURCE_FILE_BYTES}"
                        elif not is_probably_text(absolute):
                            reason = "non-text"
            if reason:
                skipped.append({"path": relative.as_posix(), "reason": reason})
                continue
            data = absolute.read_bytes()
            digest = hashlib.sha256(data).hexdigest()
            archive_name = f"SOURCE/{relative.as_posix()}"
            zip_write_bytes(archive, archive_name, data, captured)
            included.append({"path": relative.as_posix(), "size": len(data), "sha256": digest})
            manifest_lines.append(f"{digest}  {archive_name}")

        metadata = {
            "schema_version": 1,
            "artifact_type": "SYSGRID_FULL_SOURCE_SNAPSHOT",
            "tool_version": VERSION,
            "captured_utc": captured.isoformat(),
            "identity": identity,
            "included_file_count": len(included),
            "included_bytes": sum(item["size"] for item in included),
            "skipped_file_count": len(skipped),
            "exclusion_policy": {
                "excluded_directories": sorted(EXCLUDED_DIR_NAMES),
                "max_source_file_bytes": MAX_SOURCE_FILE_BYTES,
                "secrets_and_runtime_envs_excluded": True,
                "binaries_and_generated_artifacts_excluded": True,
            },
        }
        supplemental: dict[str, bytes] = {
            "META/SOURCE_IDENTITY.json": json_dump(metadata).encode(),
            "META/INCLUDED_FILES.json": json_dump(included).encode(),
            "META/SKIPPED_FILES.json": json_dump(skipped).encode(),
            "META/FILE_MANIFEST.sha256": ("\n".join(manifest_lines) + "\n").encode(),
            "META/GIT_STATUS.txt": (git_output(repo, "status", "--porcelain=v1", "--untracked-files=all") + "\n").encode(),
            "META/GIT_DIFF.patch": run(["git", "diff", "--binary", "--no-ext-diff"], cwd=repo).stdout.encode(),
            "META/GIT_DIFF_CACHED.patch": run(["git", "diff", "--cached", "--binary", "--no-ext-diff"], cwd=repo).stdout.encode(),
            "META/GIT_LOG.txt": (git_output(repo, "log", "-20", "--date=iso-strict", "--pretty=format:%H%x09%ad%x09%an%x09%s") + "\n").encode(),
            "META/RUNTIME_STATUS.json": json_dump(runtime_status(repo)).encode(),
        }
        for name, data in supplemental.items():
            zip_write_bytes(archive, name, data, captured)
    temp_destination.replace(destination)
    return {
        **metadata,
        "path": str(destination),
        "size": destination.stat().st_size,
        "sha256": sha256_file(destination),
    }


def cmd_snapshot(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    identity = repo_identity(repo)
    short_head = identity["head"][:12] if identity["head"] else "nohead"
    destination = Path(args.output).expanduser() if args.output else outbox_root() / f"SYSGRID_FULL_SOURCE_SNAPSHOT__{timestamp()}__{short_head}.zip"
    result = create_source_snapshot(repo, destination)
    write_metadata("snapshot-latest", result)
    print(json_dump(result), end="")

def cmd_verify(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    verify_script = repo / "scripts" / "verify-app.sh"
    if not verify_script.is_file():
        print(f"Missing verification script: {verify_script}", file=sys.stderr)
        return 2
    kill_session(VERIFY_SESSION)
    log_path = verify_log_path()
    log_path.parent.mkdir(parents=True, exist_ok=True)
    log_path.write_text("", encoding="utf-8")
    exit_path = state_root() / "verify" / "exit-code"
    with contextlib.suppress(FileNotFoundError):
        exit_path.unlink()
    command = (
        "set -o pipefail; "
        f"cd {shlex.quote(str(repo))}; "
        f"./scripts/verify-app.sh 2>&1 | tee {shlex.quote(str(log_path))}; "
        "status=${PIPESTATUS[0]}; "
        f"printf '%s\\n' \"$status\" > {shlex.quote(str(exit_path))}; "
        "exit \"$status\""
    )
    if args.foreground:
        completed = subprocess.run(["bash", "-lc", command])
        result = {"result": "PASS" if completed.returncode == 0 else "FAIL", "exit_code": completed.returncode, "log": str(log_path)}
        write_metadata("verify-latest", result)
        print(json_dump(result), end="")
        return completed.returncode
    started = run(["tmux", "new-session", "-d", "-s", VERIFY_SESSION, "bash", "-lc", command])
    result = {
        "result": "STARTED" if started.returncode == 0 else "FAIL",
        "session": VERIFY_SESSION,
        "log": str(log_path),
        "status_command": "sgd verify-status --phone",
    }
    write_metadata("verify-latest", result)
    print(json_dump(result), end="")
    return 0 if started.returncode == 0 else 2


def cmd_verify_status(args: argparse.Namespace) -> int:
    log_path = verify_log_path()
    exit_path = state_root() / "verify" / "exit-code"
    running = session_exists(VERIFY_SESSION)
    exit_code: int | None = None
    if exit_path.exists():
        with contextlib.suppress(ValueError):
            exit_code = int(exit_path.read_text().strip())
    tail_lines: list[str] = []
    if log_path.exists():
        tail_lines = log_path.read_text(encoding="utf-8", errors="replace").splitlines()[-args.lines:]
    result = {
        "running": running,
        "exit_code": exit_code,
        "result": "RUNNING" if running else ("PASS" if exit_code == 0 else "FAIL" if exit_code is not None else "UNKNOWN"),
        "log": str(log_path),
        "tail": tail_lines,
    }
    if args.phone:
        print(f"Verify: {result['result']} exit={exit_code if exit_code is not None else '-'}")
        for line in tail_lines[-20:]:
            print(line)
        print(f"Log: {log_path}")
    else:
        print(json_dump(result), end="")
    write_metadata("verify-status-latest", result)
    return 0 if result["result"] in ("RUNNING", "PASS") else 1


def add_file_to_zip(archive: zipfile.ZipFile, source: Path, name: str, when: dt.datetime) -> None:
    if source.is_file():
        zip_write_bytes(archive, name, source.read_bytes(), when)


def create_review_package(repo: Path, destination: Path, create_fresh_snapshot: bool = True) -> dict[str, Any]:
    captured = utc_now()
    identity = repo_identity(repo)
    state = state_root()
    temp_dir = Path(tempfile.mkdtemp(prefix="sysgrid-dev-review-"))
    try:
        if create_fresh_snapshot:
            snapshot_name = f"SYSGRID_FULL_SOURCE_SNAPSHOT__{timestamp()}__{identity['head'][:12] if identity['head'] else 'nohead'}.zip"
            snapshot_path = temp_dir / snapshot_name
            snapshot_result = create_source_snapshot(repo, snapshot_path)
        else:
            latest = read_metadata("snapshot-latest")
            if not latest or not Path(latest.get("path", "")).is_file():
                raise RuntimeError("No prior source snapshot is available")
            snapshot_path = Path(latest["path"])
            snapshot_result = latest

        summary = {
            "schema_version": 1,
            "artifact_type": "SYSGRID_DEVELOPMENT_REVIEW",
            "tool_version": VERSION,
            "captured_utc": captured.isoformat(),
            "source_identity": identity,
            "snapshot": snapshot_result,
            "runtime": runtime_status(repo),
            "smoke": read_metadata("smoke-latest"),
            "start": read_metadata("start-latest"),
            "verify": read_metadata("verify-status-latest") or read_metadata("verify-latest"),
            "doctor": read_metadata("doctor-latest"),
            "work_class": "PLATFORM_AND_ENVIRONMENT_OBSERVABILITY",
            "product_attempt_increment": False,
            "product_source_mutation": False,
        }
        destination.parent.mkdir(parents=True, exist_ok=True)
        temp_destination = destination.with_suffix(destination.suffix + ".tmp")
        manifest: list[str] = []
        with zipfile.ZipFile(temp_destination, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
            members: list[tuple[str, bytes]] = [
                ("REVIEW_ME_FIRST.json", json_dump(summary).encode()),
                ("SOURCE_SNAPSHOT.zip", snapshot_path.read_bytes()),
            ]
            metadata_files = sorted(state.glob("*.json"))
            for path in metadata_files:
                members.append((f"STATE/{path.name}", path.read_bytes()))
            for source, name in [
                (runtime_log_path(), "LOGS/runtime.log"),
                (verify_log_path(), "LOGS/verify.log"),
                (repo / "backend" / "test-results" / "verify-backend.log", "EVIDENCE/backend/verify-backend.log"),
                (repo / "frontend" / "test-results" / ".last-run.json", "EVIDENCE/frontend/.last-run.json"),
                (repo / "frontend" / "llm-report.json", "EVIDENCE/frontend/llm-report.json"),
            ]:
                if source.is_file():
                    members.append((name, source.read_bytes()))
            for name, data in sorted(members):
                zip_write_bytes(archive, name, data, captured)
                manifest.append(f"{hashlib.sha256(data).hexdigest()}  {name}")
            manifest_data = ("\n".join(manifest) + "\n").encode()
            zip_write_bytes(archive, "MANIFEST.sha256", manifest_data, captured)
        temp_destination.replace(destination)
        return {
            "path": str(destination),
            "size": destination.stat().st_size,
            "sha256": sha256_file(destination),
            "source_snapshot_sha256": snapshot_result["sha256"],
            "classification": (summary.get("smoke") or {}).get("classification", "NO_SMOKE_RESULT"),
        }
    finally:
        shutil.rmtree(temp_dir, ignore_errors=True)


def cmd_collect(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    destination = Path(args.output).expanduser() if args.output else outbox_root() / f"UPLOAD_THIS_TO_CHATGPT_DEV__{timestamp()}.zip"
    result = create_review_package(repo, destination, create_fresh_snapshot=not args.use_existing_snapshot)
    write_metadata("collect-latest", result)
    print(json_dump(result), end="")
    return 0


def cmd_cycle(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    start_args = argparse.Namespace(
        repo=str(repo), fast=args.fast, no_seed_data=args.no_seed_data, timeout=args.timeout
    )
    start_code = cmd_start(start_args)
    smoke_args = argparse.Namespace(
        repo=str(repo), api_origin=args.api_origin, user_id=args.user_id, tenant_id=args.tenant_id
    )
    if start_code == 0:
        smoke_code = cmd_smoke(smoke_args)
    else:
        smoke_code = 1
        failed_smoke = {
            "schema_version": 1,
            "captured_utc": utc_now().isoformat(),
            "result": "FAIL",
            "classification": "RUNTIME_START_FAILURE",
            "repo_identity": repo_identity(repo),
            "api_origin": args.api_origin,
            "user_id": args.user_id,
            "tenant_id": args.tenant_id,
            "probes": [],
        }
        write_metadata("smoke-latest", failed_smoke)
    destination = outbox_root() / f"UPLOAD_THIS_TO_CHATGPT_DEV__{timestamp()}.zip"
    collect_result = create_review_package(repo, destination, create_fresh_snapshot=True)
    write_metadata("cycle-latest", {
        "result": "PASS" if start_code == 0 and smoke_code == 0 else "FAIL",
        "start_exit_code": start_code,
        "smoke_exit_code": smoke_code,
        "review_package": collect_result,
    })
    print(json_dump({
        "result": "PASS" if start_code == 0 and smoke_code == 0 else "FAIL",
        "start_exit_code": start_code,
        "smoke_exit_code": smoke_code,
        "review_package": collect_result,
        "next_action": "Upload the review package to ChatGPT.",
    }), end="")
    return 0 if start_code == 0 and smoke_code == 0 else 1


def cmd_logs(args: argparse.Namespace) -> int:
    path = verify_log_path() if args.which == "verify" else runtime_log_path()
    if not path.exists():
        print(f"No {args.which} log exists at {path}", file=sys.stderr)
        return 1
    lines = path.read_text(encoding="utf-8", errors="replace").splitlines()
    print("\n".join(lines[-args.lines:]))
    return 0


def cmd_self_test(_args: argparse.Namespace) -> int:
    failures: list[str] = []
    with tempfile.TemporaryDirectory(prefix="sgd-self-test-") as tmp:
        repo = Path(tmp) / "repo"
        (repo / "backend" / "venv" / "bin").mkdir(parents=True)
        (repo / "frontend" / "src").mkdir(parents=True)
        (repo / "frontend" / "node_modules").mkdir(parents=True)
        (repo / "scripts").mkdir(parents=True)
        (repo / "backend" / "venv" / "bin" / "python").write_text("#!/bin/sh\n", encoding="utf-8")
        (repo / "scripts" / "start-local.sh").write_text("#!/bin/sh\n", encoding="utf-8")
        (repo / "scripts" / "verify-app.sh").write_text("#!/bin/sh\n", encoding="utf-8")
        (repo / "frontend" / "src" / "app.ts").write_text("export const value = 1\n", encoding="utf-8")
        (repo / ".env.local").write_text("SECRET=should-not-ship\n", encoding="utf-8")
        run(["git", "init", "-q"], cwd=repo)
        run(["git", "config", "user.email", "selftest@example.invalid"], cwd=repo)
        run(["git", "config", "user.name", "Self Test"], cwd=repo)
        run(["git", "add", "."], cwd=repo)
        run(["git", "commit", "-qm", "fixture"], cwd=repo)
        snapshot = Path(tmp) / "snapshot.zip"
        result = create_source_snapshot(repo, snapshot)
        with zipfile.ZipFile(snapshot) as archive:
            names = set(archive.namelist())
        if "SOURCE/frontend/src/app.ts" not in names:
            failures.append("source file missing from snapshot")
        if "SOURCE/.env.local" in names:
            failures.append("secret runtime env included")
        if result["included_file_count"] < 3:
            failures.append("unexpectedly small snapshot")
    output = {"result": "PASS" if not failures else "FAIL", "failures": failures, "version": VERSION}
    print(json_dump(output), end="")
    return 0 if not failures else 1


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="sgd",
        description="SysGrid development-lane control: source awareness, local runtime, smoke, evidence, and phone continuity.",
    )
    parser.add_argument("--version", action="version", version=f"%(prog)s {VERSION}")
    sub = parser.add_subparsers(dest="command", required=True)

    def add_repo(command: argparse.ArgumentParser) -> None:
        command.add_argument("--repo", help="SysGrid repository root; otherwise discover from cwd or SYSGRID_REPO")

    doctor = sub.add_parser("doctor", help="Validate source identity, prerequisites, nested Git state, and runtime ownership")
    add_repo(doctor)
    doctor.set_defaults(func=cmd_doctor)

    stop = sub.add_parser("stop", help="Stop the managed local runtime and clear ports 8000/5173")
    add_repo(stop)
    stop.add_argument("--include-verify", action="store_true", help="Also stop the persistent verification session")
    stop.set_defaults(func=cmd_stop)

    start = sub.add_parser("start", help="Cleanly start scripts/start-local.sh in a persistent tmux session")
    add_repo(start)
    start.add_argument("--fast", action="store_true", help="Skip the launcher's typecheck for rapid visual iteration")
    start.add_argument("--no-seed-data", action="store_true", help="Start with no domain seed data")
    start.add_argument("--timeout", type=int, default=240, help="Backend readiness timeout in seconds")
    start.set_defaults(func=cmd_start)

    for name, help_text, func in [
        ("status", "Show source/runtime/API status", cmd_status),
        ("smoke", "Probe health and every primary workspace endpoint", cmd_smoke),
    ]:
        command = sub.add_parser(name, help=help_text)
        add_repo(command)
        command.add_argument("--api-origin", default=f"http://{DEFAULT_BACKEND_HOST}:{DEFAULT_BACKEND_PORT}")
        command.add_argument("--user-id", default=DEFAULT_USER_ID)
        command.add_argument("--tenant-id")
        if name == "status":
            command.add_argument("--phone", action="store_true", help="Use compact phone-readable output")
        command.set_defaults(func=func)

    snapshot = sub.add_parser("snapshot", help="Create a full, secret-filtered source snapshot with exact Git identity and hashes")
    add_repo(snapshot)
    snapshot.add_argument("--output")
    snapshot.set_defaults(func=cmd_snapshot)

    collect = sub.add_parser("collect", help="Create one development review ZIP with full source, runtime, smoke, and logs")
    add_repo(collect)
    collect.add_argument("--output")
    collect.add_argument("--use-existing-snapshot", action="store_true")
    collect.set_defaults(func=cmd_collect)

    cycle = sub.add_parser("cycle", help="Clean start, API smoke, full source snapshot, and one upload ZIP")
    add_repo(cycle)
    cycle.add_argument("--fast", action="store_true")
    cycle.add_argument("--no-seed-data", action="store_true")
    cycle.add_argument("--timeout", type=int, default=240)
    cycle.add_argument("--api-origin", default=f"http://{DEFAULT_BACKEND_HOST}:{DEFAULT_BACKEND_PORT}")
    cycle.add_argument("--user-id", default=DEFAULT_USER_ID)
    cycle.add_argument("--tenant-id")
    cycle.set_defaults(func=cmd_cycle)

    verify = sub.add_parser("verify", help="Run the canonical full verification persistently for Mac/phone continuity")
    add_repo(verify)
    verify.add_argument("--foreground", action="store_true")
    verify.set_defaults(func=cmd_verify)

    verify_status = sub.add_parser("verify-status", help="Show persistent verification progress and result")
    verify_status.add_argument("--phone", action="store_true")
    verify_status.add_argument("--lines", type=int, default=80)
    verify_status.set_defaults(func=cmd_verify_status)

    logs = sub.add_parser("logs", help="Tail managed runtime or verification logs")
    logs.add_argument("which", choices=["runtime", "verify"], nargs="?", default="runtime")
    logs.add_argument("--lines", type=int, default=120)
    logs.set_defaults(func=cmd_logs)

    self_test = sub.add_parser("self-test", help="Run immutable package self-tests")
    self_test.set_defaults(func=cmd_self_test)
    return parser


def main(argv: Sequence[str] | None = None) -> int:
    parser = build_parser()
    args = parser.parse_args(argv)
    try:
        return int(args.func(args))
    except KeyboardInterrupt:
        print("Interrupted.", file=sys.stderr)
        return 130
    except subprocess.TimeoutExpired as exc:
        print(f"Timed out: {exc}", file=sys.stderr)
        return 124
    except Exception as exc:  # fail closed with an actionable class
        error = {
            "result": "FAIL",
            "classification": "SGD_INTERNAL_ERROR",
            "error_type": type(exc).__name__,
            "message": str(exc),
        }
        print(json_dump(error), file=sys.stderr, end="")
        return 70


if __name__ == "__main__":
    raise SystemExit(main())
