def json_dump(data: Any) -> str:

def metadata_path(name: str) -> Path:
    return state_root() / f"{name}.json"


def write_metadata(name: str, data: Any) -> None:
    path = metadata_path(name)
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_suffix(path.suffix + ".tmp")
    temp.write_text(json_dump(data), encoding="utf-8")
    temp.replace(path)


def read_metadata(name: str) -> dict[str, Any] | None:
    path = metadata_path(name)
    if not path.is_file():
        return None
    try:
        value = json.loads(path.read_text(encoding="utf-8"))
        return value if isinstance(value, dict) else None
    except (json.JSONDecodeError, OSError):
        return None


def run(
    command: Sequence[str],
    *,
    cwd: Path | None = None,
    check: bool = False,
    text: bool = True,
    capture_output: bool = True,
    env: dict[str, str] | None = None,
    timeout: float | None = None,
) -> subprocess.CompletedProcess:
    return subprocess.run(
        list(command),
        cwd=str(cwd) if cwd else None,
        check=check,
        text=text,
        capture_output=capture_output,
        env=env,
        timeout=timeout,
    )


def command_exists(name: str) -> bool:
    return shutil.which(name) is not None


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def git_output(repo: Path, *args: str, default: str = "") -> str:
    result = run(["git", *args], cwd=repo)
    return result.stdout.strip() if result.returncode == 0 else default


def valid_repo(candidate: Path) -> Path | None:
    try:
        candidate = candidate.expanduser().resolve()
    except OSError:
        return None
    for current in [candidate, *candidate.parents]:
        if (current / "backend").is_dir() and (current / "frontend").is_dir():
            result = run(["git", "-C", str(current), "rev-parse", "--show-toplevel"])
            if result.returncode == 0:
                root = Path(result.stdout.strip()).resolve()
                if (root / "backend").is_dir() and (root / "frontend").is_dir():
                    return root
    return None


def remembered_repo() -> Path | None:
    direct = read_metadata("repo")
    if direct and isinstance(direct.get("repo"), str):
        resolved = valid_repo(Path(direct["repo"]))
        if resolved:
            return resolved
    # Automatic v1.0 migration from existing state.
    for name in ("doctor-latest", "start-latest", "status-latest"):
        data = read_metadata(name) or {}
        candidates = [
            data.get("repo"),
            (data.get("identity") or {}).get("repo") if isinstance(data.get("identity"), dict) else None,
            (data.get("repo") or {}).get("repo") if isinstance(data.get("repo"), dict) else None,
        ]
        for value in candidates:
            if isinstance(value, str):
                resolved = valid_repo(Path(value))
                if resolved:
                    return resolved
    return None


def remember_repo(repo: Path) -> None:
    write_metadata("repo", {
        "schema_version": SCHEMA_VERSION,
        "repo": str(repo.resolve()),
        "recorded_utc": utc_now().isoformat(),
        "head": git_output(repo, "rev-parse", "HEAD"),
    })


def discover_repo(explicit: str | None = None) -> Path:
    candidates: list[Path] = []
    if explicit:
        candidates.append(Path(explicit))
    env_repo = os.environ.get("SYSGRID_REPO")
    if env_repo:
        candidates.append(Path(env_repo))
    candidates.append(Path.cwd())
    remembered = remembered_repo()
    if remembered:
        candidates.append(remembered)
    seen: set[Path] = set()
    for candidate in candidates:
        with contextlib.suppress(OSError):
            resolved_candidate = candidate.expanduser().resolve()
            if resolved_candidate in seen:
                continue
            seen.add(resolved_candidate)
            repo = valid_repo(resolved_candidate)
            if repo:
                remember_repo(repo)
                return repo
    raise SystemExit(
        "SysGrid repository not found. Run `sgd use /absolute/path/to/sysgrid` once, "
        "run inside the repository, or set SYSGRID_REPO."
    )


def session_exists(name: str) -> bool:
    return command_exists("tmux") and run(["tmux", "has-session", "-t", name]).returncode == 0


def kill_session(name: str) -> bool:
    return session_exists(name) and run(["tmux", "kill-session", "-t", name]).returncode == 0


def listener_pids(port: int) -> list[int]:
    if not command_exists("lsof"):
        return []
    result = run(["lsof", f"-tiTCP:{port}", "-sTCP:LISTEN"])
    if result.returncode not in (0, 1):
        return []
    pids: list[int] = []
    for token in result.stdout.split():
        with contextlib.suppress(ValueError):
            pids.append(int(token))
    return sorted(set(pids))


def terminate_pid(pid: int) -> None:
    with contextlib.suppress(ProcessLookupError, PermissionError):
        os.kill(pid, signal.SIGTERM)
    deadline = time.monotonic() + 2.0
    while time.monotonic() < deadline:
        try:
            os.kill(pid, 0)
        except ProcessLookupError:
            return
        except PermissionError:
            return
        time.sleep(0.1)
    with contextlib.suppress(ProcessLookupError, PermissionError):
        os.kill(pid, signal.SIGKILL)


def stop_runtime_ports(ports: Iterable[int]) -> dict[str, Any]:
    stopped: dict[str, list[int]] = {}
    for port in ports:
        pids = listener_pids(port)
        for pid in pids:
            terminate_pid(pid)
        stopped[str(port)] = pids
    return {"stopped": stopped, "remaining": {str(port): listener_pids(port) for port in ports}}

            "duration_ms": self.duration_ms,
            "request_id": self.request_id,

            "record_count": self.record_count,
            "sample_keys": self.sample_keys,
            "error_preview": self.error_preview,
            "error": self.error,

def summarize_json(value: Any) -> tuple[str | None, int | None, list[str]]:
    if isinstance(value, list):
        keys: list[str] = []
        if value and isinstance(value[0], dict):
            keys = sorted(str(key) for key in value[0].keys())[:24]
        return "array", len(value), keys
    if isinstance(value, dict):
        keys = sorted(str(key) for key in value.keys())[:24]
        count: int | None = None
        for key in ("definitions", "items", "results", "data"):
            candidate = value.get(key)
            if isinstance(candidate, list):
                count = len(candidate)
                break
        return "object", count, keys
    if value is None:
        return "null", None, []
    return type(value).__name__, None, []


def http_get(url: str, headers: dict[str, str] | None = None, timeout_seconds: float = 8.0) -> ProbeResult:
    started = time.monotonic()
    request = urllib.request.Request(url, headers=headers or {}, method="GET")
    body = b""
    status = 0
    response_headers: Any = {}
    error: str | None = None
    try:
        with urllib.request.urlopen(request, timeout=timeout_seconds) as response:
            body = response.read(MAX_HTTP_BODY_BYTES + 1)
            status = int(response.status)
            response_headers = response.headers
    except urllib.error.HTTPError as exc:
        body = exc.read(MAX_HTTP_BODY_BYTES + 1)
        status = int(exc.code)
        response_headers = exc.headers
        error = f"HTTP {exc.code}"
    except (urllib.error.URLError, TimeoutError, socket.timeout, ConnectionError, OSError) as exc:
        return ProbeResult(
            name="", url=url, status=0, ok=False,
            duration_ms=int((time.monotonic() - started) * 1000),
            request_id=None, tenant_id=None, content_type=None,
            body_sha256=None, body_bytes=0, body_truncated=False,
            json_type=None, record_count=None, sample_keys=[],
            error_preview="", error=f"{type(exc).__name__}: {exc}", raw_text="",
        )
    truncated = len(body) > MAX_HTTP_BODY_BYTES
    if truncated:
        body = body[:MAX_HTTP_BODY_BYTES]
    text = body.decode("utf-8", errors="replace")
    parsed: Any = None
    json_type: str | None = None
    record_count: int | None = None
    sample_keys: list[str] = []
    try:
        parsed = json.loads(text)
    except json.JSONDecodeError:
        pass
    else:
        json_type, record_count, sample_keys = summarize_json(parsed)
    ok = 200 <= status < 300
    preview = ""
    if not ok:
        preview = re.sub(r"\s+", " ", text).strip()[:1200]
    return ProbeResult(
        name="", url=url, status=status, ok=ok,
        duration_ms=int((time.monotonic() - started) * 1000),
        request_id=response_headers.get("x-request-id"),
        tenant_id=response_headers.get("x-sysgrid-tenant-id"),
        content_type=response_headers.get("content-type"),
        body_sha256=hashlib.sha256(body).hexdigest(),
        body_bytes=len(body),
        body_truncated=truncated,
        json_type=json_type,
        record_count=record_count,
        sample_keys=sample_keys,
        error_preview=preview,
        error=error,
        raw_text=text,
    )


def selected_tenant_id(probe: ProbeResult) -> str | None:
    try:
        value = json.loads(probe.raw_text)
    except json.JSONDecodeError:
        return None
    if not isinstance(value, list):
        return None
    selected = [row for row in value if isinstance(row, dict) and row.get("is_selected")]
    candidate = selected[0] if selected else (value[0] if value else None)
    return str(candidate["id"]) if isinstance(candidate, dict) and candidate.get("id") is not None else None


def classify_smoke(probes: list[ProbeResult]) -> str:
    if not probes:
        return "NO_PROBES"
    if probes[0].status == 0:
        return "BACKEND_UNAVAILABLE"
    if not probes[0].ok:
        return "BACKEND_HEALTH_FAILURE"
    failures = [probe for probe in probes[1:] if not probe.ok]
    if not failures:
        return "PASS"
    if any(probe.status == 0 for probe in failures):
        return "NETWORK_OR_PROCESS_FAILURE"
    if any(probe.status >= 500 for probe in failures):
        return "BACKEND_ENDPOINT_FAILURE"
    if any(probe.status in (401, 403) for probe in failures):
        return "IDENTITY_OR_PERMISSION_FAILURE"
    if any(probe.status == 404 for probe in failures):
        return "API_CONTRACT_ROUTE_MISSING"
    return "API_CONTRACT_FAILURE"

def classify_repo_path(relative: Path) -> str:
    path = relative.as_posix()
    name = relative.name.lower()
    if path.startswith("backend/app/") or path.startswith("backend/alembic/") or path in {"seed.py", "start_backend_fixed.py"}:
        return "product_source_backend"
    if path.startswith("frontend/src/"):
        if re.search(r"(?:^|/)(?:tests?|__tests__)(?:/|$)", path) or re.search(r"\.(?:test|spec)\.[^.]+$", name):
            return "tests"
        return "product_source_frontend"
    if path.startswith("backend/test") or "/test_" in path or path.startswith("frontend/tests/") or re.search(r"\.(?:test|spec)\.[^.]+$", name):
        return "tests"
    if path.startswith("scripts/") or path.startswith("frontend/scripts/") or path.startswith(".github/"):
        return "verification_or_tooling"
    if path.startswith("COMMAND_OUTPUTS/") or path.startswith("RUN_ARTIFACTS/") or path.startswith("REVIEW_ARTIFACTS/"):
        return "historical_evidence"
    if path.startswith("patches/") or path.endswith(".patch") or path.endswith(".diff"):
        return "temporary_artifact"
    if relative.suffix.lower() in {".md", ".rst"} or path.startswith("docs/"):
        return "documentation"
    if relative.name in SOURCE_BASENAMES or relative.suffix.lower() in {".json", ".yaml", ".yml", ".toml", ".ini", ".cfg", ".lock"}:
        return "configuration_or_contract"
    return "repository_support"


def git_status_entries(repo: Path) -> list[dict[str, Any]]:
    result = run(
        ["git", "status", "--porcelain=v1", "-z", "--untracked-files=all"],
        cwd=repo, text=False,
    )
    if result.returncode != 0:
        return []
    raw = result.stdout if isinstance(result.stdout, bytes) else bytes(result.stdout)
    tokens = [REDACTED]
    entries: list[dict[str, Any]] = []
    index = 0
    while index < len(tokens):
        token = [REDACTED]
        index += 1
        if not token:
            continue
        decoded = token.decode("utf-8", errors="surrogateescape")
        if len(decoded) < 4:
            continue
        status = decoded[:2]
        path_text = decoded[3:]
        old_path: str | None = None
        if status[0] in {"R", "C"} or status[1] in {"R", "C"}:
            if index < len(tokens) and tokens[index]:
                old_path = tokens[index].decode("utf-8", errors="surrogateescape")
                index += 1
        relative = Path(path_text)
        entries.append({
            "status": status,
            "path": path_text,
            "old_path": old_path,
            "classification": classify_repo_path(relative),
        })
    return entries


def nested_git_dirs(repo: Path) -> list[str]:
    found: list[str] = []
    for root, dirs, _files in os.walk(repo):
        root_path = Path(root)
        relative_parts = root_path.parts[len(repo.parts):]
        if ".git" in relative_parts:
            dirs[:] = []
            continue
        if ".git" in dirs:
            if root_path != repo:
                found.append(root_path.relative_to(repo).as_posix())
            dirs.remove(".git")
        dirs[:] = [name for name in dirs if name not in EXCLUDED_DIR_NAMES]
    return sorted(set(found))


def repo_identity(repo: Path) -> dict[str, Any]:
    dirty_entries = git_status_entries(repo)
    counts: dict[str, int] = {}
    for entry in dirty_entries:
        role = str(entry["classification"])
        counts[role] = counts.get(role, 0) + 1
    return {
        "schema_version": SCHEMA_VERSION,
        "captured_utc": utc_now().isoformat(),
        "repo": str(repo),
        "git_root": git_output(repo, "rev-parse", "--show-toplevel"),
        "branch": git_output(repo, "branch", "--show-current"),
        "head": git_output(repo, "rev-parse", "HEAD"),
        "remote_origin": git_output(repo, "remote", "get-url", "origin"),
        "upstream": git_output(repo, "rev-parse", "--abbrev-ref", "--symbolic-full-name", "@{u}"),
        "dirty_count": len(dirty_entries),
        "dirty_summary": dict(sorted(counts.items())),
        "dirty_entries": dirty_entries,
        "status_lines": [
            f"{entry['status']} {entry['path']}" + (f" <- {entry['old_path']}" if entry.get("old_path") else "")
            for entry in dirty_entries
        ],
        "nested_git_directories": nested_git_dirs(repo),
        "start_local_sha256": sha256_file(repo / "scripts" / "start-local.sh") if (repo / "scripts" / "start-local.sh").is_file() else None,
        "verify_app_sha256": sha256_file(repo / "scripts" / "verify-app.sh") if (repo / "scripts" / "verify-app.sh").is_file() else None,
    }


def runtime_status(repo: Path | None = None) -> dict[str, Any]:
    health = http_get(
        f"http://{DEFAULT_BACKEND_HOST}:{DEFAULT_BACKEND_PORT}/api/v1/health",
        {"X-User-Id": DEFAULT_USER_ID},
        timeout_seconds=2.0,
    )
    return {
        "captured_utc": utc_now().isoformat(),
        "runtime_session": session_exists(RUNTIME_SESSION),
        "audit_session": session_exists(AUDIT_SESSION),
        "listeners": {
            str(DEFAULT_BACKEND_PORT): listener_pids(DEFAULT_BACKEND_PORT),
            str(DEFAULT_FRONTEND_PORT): listener_pids(DEFAULT_FRONTEND_PORT),
        },
        "health": health.public(),
        "repo": str(repo) if repo else None,
    }


def doctor_report(repo: Path) -> dict[str, Any]:
    checks = {
        "git": command_exists("git"),
        "python3": command_exists("python3"),
        "tmux": command_exists("tmux"),
        "curl": command_exists("curl"),
        "lsof": command_exists("lsof"),
        "npm": command_exists("npm"),
        "start_local": (repo / "scripts" / "start-local.sh").is_file(),
        "verify_app": (repo / "scripts" / "verify-app.sh").is_file(),
        "backend_python": (repo / "backend" / "venv" / "bin" / "python").is_file(),
        "frontend_node_modules": (repo / "frontend" / "node_modules").is_dir(),
        "playwright_runner": (repo / "frontend" / "node_modules" / ".bin" / "playwright").is_file(),
    }
    required = [
        "git", "python3", "tmux", "curl", "lsof", "npm", "start_local",
        "verify_app", "backend_python", "frontend_node_modules", "playwright_runner",
    ]
    missing = [name for name in required if not checks[name]]
    identity = repo_identity(repo)
    return {
        "schema_version": SCHEMA_VERSION,
        "result": "PASS" if not missing else "FAIL",
        "checks": checks,
        "missing": missing,
        "identity": identity,
        "runtime": runtime_status(repo),
        "delivery_blockers": (
            [{"classification": "NESTED_GIT_DIRECTORY", "path": path} for path in identity["nested_git_directories"]]
        ),
    }


def should_skip_source_path(relative: Path) -> tuple[bool, str | None]:
    parts = relative.parts
    if any(part in EXCLUDED_DIR_NAMES for part in parts[:-1]):
        return True, "excluded-directory"
    name = relative.name
    lower_name = name.lower()
    # Exact runtime/secret files are excluded. Source filenames containing
    # "secret" or "credential" are not excluded; those names are common in
    # legitimate schemas, migrations, clients, and tests.
    if name in SECRET_BASENAMES or lower_name.startswith(".env."):
        if lower_name.endswith((".example", ".sample", ".template")):
            return False, None
        return True, "secret-or-runtime-env"
    if relative.suffix.lower() in EXCLUDED_FILE_SUFFIXES:
        return True, "binary-or-sensitive-suffix"
    return False, None


def is_probably_text(path: Path) -> bool:
    if path.suffix.lower() in SOURCE_EXTENSIONS or path.name in SOURCE_BASENAMES:
        return True
    try:
        sample = path.read_bytes()[:8192]
    except OSError:
        return False
    if b"\x00" in sample:
        return False
    try:
        sample.decode("utf-8")
        return True
    except UnicodeDecodeError:
        return False


def git_candidate_paths(repo: Path) -> list[Path]:
    result = run(
        ["git", "ls-files", "--cached", "--others", "--exclude-standard", "-z"],
        cwd=repo, text=False,
    )
    if result.returncode != 0:
        stderr = result.stderr.decode("utf-8", errors="replace") if isinstance(result.stderr, bytes) else str(result.stderr)
        raise RuntimeError(stderr)
    raw = result.stdout if isinstance(result.stdout, bytes) else bytes(result.stdout)
    candidates: list[Path] = []
    for token in raw.split(b"\0"):
        if not token:
            continue
        relative = Path(token.decode("utf-8", errors="surrogateescape"))
        absolute = repo / relative
        if absolute.is_dir():
            for root, dirs, files in os.walk(absolute):
                root_path = Path(root)
                dirs[:] = [name for name in dirs if name not in EXCLUDED_DIR_NAMES]
                for file_name in files:
                    candidates.append((root_path / file_name).relative_to(repo))
        else:
            candidates.append(relative)
    return sorted(set(candidates), key=lambda path: path.as_posix())


def source_inventory(repo: Path) -> dict[str, Any]:
    included: list[dict[str, Any]] = []
    skipped: list[dict[str, Any]] = []
    role_counts: dict[str, int] = {}
    fingerprint_lines: list[str] = []
    for relative in git_candidate_paths(repo):
        absolute = repo / relative
        reason: str | None = None
        if absolute.is_symlink():
            reason = "symlink"
        elif not absolute.is_file():
            reason = "not-regular-file"
        else:
            skip, skip_reason = should_skip_source_path(relative)
            if skip:
                reason = skip_reason
            else:
                try:
                    size = absolute.stat().st_size
                except OSError:
                    reason = "stat-failed"
                else:
                    if size > MAX_SOURCE_FILE_BYTES:
                        reason = f"larger-than-{MAX_SOURCE_FILE_BYTES}"
                    elif not is_probably_text(absolute):
                        reason = "non-text"
        role = classify_repo_path(relative)
        if reason:
            skipped.append({"path": relative.as_posix(), "classification": role, "reason": reason})
            continue
        data = absolute.read_bytes()
        digest = hashlib.sha256(data).hexdigest()
        entry = {
            "path": relative.as_posix(),
            "classification": role,
            "size": len(data),
            "sha256": digest,
        }
        included.append(entry)
        role_counts[role] = role_counts.get(role, 0) + 1
        fingerprint_lines.append(f"{digest}  {relative.as_posix()}")
    fingerprint = hashlib.sha256(("\n".join(sorted(fingerprint_lines)) + "\n").encode()).hexdigest()
    return {
        "included": included,
        "skipped": skipped,
        "role_counts": dict(sorted(role_counts.items())),
        "source_fingerprint": fingerprint,
        "included_bytes": sum(int(item["size"]) for item in included),
    }


def zip_write_bytes(archive: zipfile.ZipFile, name: str, data: bytes, when: dt.datetime) -> None:
    info = zipfile.ZipInfo(name)
    safe_year = min(max(when.year, 1980), 2107)
    info.date_time = (safe_year, when.month, when.day, when.hour, when.minute, when.second)
    info.compress_type = zipfile.ZIP_DEFLATED
    info.external_attr = 0o100644 << 16
    archive.writestr(info, data)


def create_source_snapshot(repo: Path, destination: Path) -> dict[str, Any]:
    captured = utc_now()
    identity = repo_identity(repo)
    inventory = source_inventory(repo)
    included = inventory["included"]
    skipped = inventory["skipped"]
    destination.parent.mkdir(parents=True, exist_ok=True)
    temp_destination = destination.with_suffix(destination.suffix + ".tmp")
    manifest_lines: list[str] = []
    authoritative_count = 0
    auxiliary_count = 0
    with zipfile.ZipFile(temp_destination, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
        for entry in included:
            relative = Path(str(entry["path"]))
            data = (repo / relative).read_bytes()
            role = str(entry["classification"])
            if role in AUXILIARY_ROLES:
                archive_name = f"AUXILIARY/{relative.as_posix()}"
                auxiliary_count += 1
            else:
                archive_name = f"SOURCE/{relative.as_posix()}"
                authoritative_count += 1
            zip_write_bytes(archive, archive_name, data, captured)
            manifest_lines.append(f"{entry['sha256']}  {archive_name}")
        metadata = {
            "schema_version": SCHEMA_VERSION,
            "artifact_type": "SYSGRID_AUTHORITATIVE_SOURCE_SNAPSHOT",
            "tool_version": VERSION,
            "captured_utc": captured.isoformat(),
            "identity": identity,
            "source_fingerprint": inventory["source_fingerprint"],
            "included_file_count": len(included),
            "authoritative_code_file_count": authoritative_count,
            "auxiliary_file_count": auxiliary_count,
            "included_bytes": inventory["included_bytes"],
            "skipped_file_count": len(skipped),
            "classification_counts": inventory["role_counts"],
            "exclusion_policy": {
                "excluded_directories": sorted(EXCLUDED_DIR_NAMES),
                "max_source_file_bytes": MAX_SOURCE_FILE_BYTES,
                "exact_runtime_envs_and_key_material_excluded": True,
                "legitimate_source_names_containing_secret_or_credential_included": True,
                "binaries_and_generated_artifacts_excluded": True,
            },
        }
        supplemental = {
            "META/SOURCE_IDENTITY.json": json_dump(metadata).encode(),
            "META/INCLUDED_FILES.json": json_dump(included).encode(),
            "META/SKIPPED_FILES.json": json_dump(skipped).encode(),
            "META/WORKTREE_CLASSIFICATION.json": json_dump({
                "dirty_entries": identity["dirty_entries"],
                "dirty_summary": identity["dirty_summary"],
                "source_classification_counts": inventory["role_counts"],
            }).encode(),
            "META/FILE_MANIFEST.sha256": ("\n".join(manifest_lines) + "\n").encode(),
            "META/GIT_STATUS.txt": (
                "\n".join(identity["status_lines"]) + ("\n" if identity["status_lines"] else "")
            ).encode(),
            "META/GIT_DIFF.patch": run(["git", "diff", "--binary", "--no-ext-diff"], cwd=repo).stdout.encode(),
            "META/GIT_DIFF_CACHED.patch": run(["git", "diff", "--cached", "--binary", "--no-ext-diff"], cwd=repo).stdout.encode(),
            "META/GIT_LOG.txt": (
                git_output(repo, "log", "-20", "--date=iso-strict", "--pretty=format:%H%x09%ad%x09%an%x09%s") + "\n"
            ).encode(),
            "META/RUNTIME_STATUS.json": json_dump(runtime_status(repo)).encode(),
        }
        for name, data in supplemental.items():
            zip_write_bytes(archive, name, data, captured)
    temp_destination.replace(destination)
    result = {
        **metadata,
        "path": str(destination),
        "size": destination.stat().st_size,
        "sha256": sha256_file(destination),
    }
    return result

def wait_for_backend(timeout_seconds: int, log_path: Path) -> ProbeResult:
    deadline = time.monotonic() + timeout_seconds
    url = f"http://{DEFAULT_BACKEND_HOST}:{DEFAULT_BACKEND_PORT}/api/v1/health"
    last = http_get(url, {"X-User-Id": DEFAULT_USER_ID}, timeout_seconds=2.0)
    while time.monotonic() < deadline:
        last = http_get(url, {"X-User-Id": DEFAULT_USER_ID}, timeout_seconds=2.0)
        if last.ok:
            return last
        if not session_exists(RUNTIME_SESSION):
            break
        time.sleep(1)
    if log_path.exists():
        tail = "\n".join(log_path.read_text(encoding="utf-8", errors="replace").splitlines()[-80:])
        last.error_preview = (last.error_preview + "\n--- runtime log tail ---\n" + tail)[-8000:]
    return last


def start_runtime(repo: Path, *, fast: bool, seed_data: bool, timeout: int) -> dict[str, Any]:
    doctor = doctor_report(repo)
    write_metadata("doctor-latest", doctor)
    if doctor["result"] != "PASS":
        return {
            "result": "FAIL",
            "classification": "DOCTOR_FAILURE",
            "doctor": doctor,
            "repo": str(repo),
        }
    kill_session(RUNTIME_SESSION)
    stop_runtime_ports([DEFAULT_BACKEND_PORT, DEFAULT_FRONTEND_PORT])
    log_path = runtime_log_path()
    log_path.parent.mkdir(parents=True, exist_ok=True)
    if log_path.exists():
        shutil.copy2(log_path, log_path.with_name(f"runtime-{timestamp()}.log"))
    log_path.write_text("", encoding="utf-8")
    exit_path = state_root() / "runtime" / "exit-code"
    exit_path.parent.mkdir(parents=True, exist_ok=True)
    with contextlib.suppress(FileNotFoundError):
        exit_path.unlink()

    launcher = repo / "scripts" / "start-local.sh"
    command = [str(launcher)]
    if fast:
        command.append("--skip-typecheck")
    if not seed_data:
        command.append("--no-seed-data")
    tmux_shell = (
        "set -o pipefail; "
        f"cd {shlex.quote(str(repo))}; "
        f"{shlex.join(command)} 2>&1 | tee {shlex.quote(str(log_path))}; "
        "status=${PIPESTATUS[0]}; "
        f"printf '%s\\n' \"$status\" > {shlex.quote(str(exit_path))}; "
        "exit \"$status\""
    )
    started = run(["tmux", "new-session", "-d", "-s", RUNTIME_SESSION, "bash", "-lc", tmux_shell])
    if started.returncode != 0:
        return {
            "result": "FAIL",
            "classification": "TMUX_START_FAILURE",
            "message": started.stderr or "Failed to start tmux runtime session",
            "repo": str(repo),
        }
    health = wait_for_backend(timeout, log_path)
    result = {
        "schema_version": SCHEMA_VERSION,
        "captured_utc": utc_now().isoformat(),
        "result": "PASS" if health.ok else "FAIL",
        "classification": "RUNTIME_READY" if health.ok else "RUNTIME_START_FAILURE",
        "repo": str(repo),
        "session": RUNTIME_SESSION,
        "log": str(log_path),
        "health": health.public(),
        "fast": fast,
        "seed_data": seed_data,
    }
    write_metadata("start-latest", result)
    return result


def cmd_start(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    result = start_runtime(repo, fast=args.fast, seed_data=not args.no_seed_data, timeout=args.timeout)
    if args.json:
        print(json_dump(result), end="")
    else:
        print(f"Runtime: {result['result']} ({result['classification']})")
        print(f"Repo: {repo}")
        if result.get("health"):
            health = result["health"]
            print(f"Backend: status={health['status']} request_id={health['request_id'] or '-'}")
        print(f"Log: {result.get('log', runtime_log_path())}")
    return 0 if result["result"] == "PASS" else 1


def smoke_probe_matrix(
    api_origin: str,
    user_id: str,
    tenant_override: str | None = None,
) -> tuple[list[ProbeResult], str | None]:
    api_base = api_origin.rstrip("/") + "/api/v1"
    probes: list[ProbeResult] = []
    health = http_get(f"{api_base}/health", {"X-User-Id": user_id})
    health.name = "health"
    probes.append(health)
    if not health.ok:
        return probes, tenant_override
    tenants = http_get(f"{api_base}/tenants/me", {"X-User-Id": user_id})
    tenants.name = "tenants"
    probes.append(tenants)
    tenant_id = tenant_override or (selected_tenant_id(tenants) if tenants.ok else None)
    headers = {"X-User-Id": user_id}
    if tenant_id:
        headers["X-Tenant-Id"] = tenant_id
    endpoints = [
        ("monitoring", "/monitoring?include_deleted=true"),
        ("assets", "/devices?include_deleted=true"),
        ("services", "/logical-services?include_deleted=true"),
        ("external", "/intelligence/entities?include_deleted=true"),
        ("network", "/networks/connections?include_deleted=true"),
        ("far", "/far/modes"),
        ("research", "/investigations?include_deleted=true"),
        ("vendors", "/vendors?include_deleted=true"),
        ("settings-options", "/settings/options"),
        ("workspace-definitions", "/workspaces/definitions"),
    ]
    for name, path in endpoints:
        probe = http_get(f"{api_base}{path}", headers)
        probe.name = name
        probes.append(probe)
    return probes, tenant_id


def build_smoke_result(
    repo: Path,
    api_origin: str,
    user_id: str,
    tenant_id: str | None,
) -> dict[str, Any]:
    probes, resolved_tenant = smoke_probe_matrix(api_origin, user_id, tenant_id)
    classification = classify_smoke(probes)
    inventory = source_inventory(repo)
    result = {
        "schema_version": SCHEMA_VERSION,
        "captured_utc": utc_now().isoformat(),
        "result": "PASS" if classification == "PASS" else "FAIL",
        "classification": classification,
        "repo_identity": repo_identity(repo),
        "source_fingerprint": inventory["source_fingerprint"],
        "api_origin": api_origin,
        "user_id": user_id,
        "tenant_id": resolved_tenant,
        "probe_count": len(probes),
        "passed_probe_count": sum(1 for probe in probes if probe.ok),
        "probes": [probe.public() for probe in probes],
    }
    write_metadata("smoke-latest", result)
    smoke_path = state_root() / "smoke" / "latest.json"
    smoke_path.parent.mkdir(parents=True, exist_ok=True)
    smoke_path.write_text(json_dump(result), encoding="utf-8")
    return result


def print_smoke_compact(result: dict[str, Any]) -> None:
    print(
        f"API smoke: {result['result']} ({result['classification']}) "
        f"{result['passed_probe_count']}/{result['probe_count']} tenant={result.get('tenant_id') or '-'}"
    )
    for probe in result["probes"]:
        shape = probe.get("json_type") or "-"
        count = probe.get("record_count")
        shape_text = f"{shape}" + (f"[{count}]" if count is not None else "")
        marker = "PASS" if probe["ok"] else "FAIL"
        line = (
            f"{marker:4} {probe['name']:<22} status={probe['status']:<3} "
            f"{probe['duration_ms']:>4}ms {shape_text}"
        )
        if not probe["ok"]:
            line += f" request_id={probe.get('request_id') or '-'} {probe.get('error') or ''}"
        print(line.rstrip())
        if not probe["ok"] and probe.get("error_preview"):
            print(f"     {probe['error_preview'][:500]}")

def parse_verify_specs(verify_script: Path) -> list[str]:
    if not verify_script.is_file():
        return []
    text = verify_script.read_text(encoding="utf-8", errors="replace")
    match = re.search(r"PLAYWRIGHT_SPECS=\(\s*(.*?)\s*\)", text, re.DOTALL)
    if not match:
        return []
    specs: list[str] = []
    for line in match.group(1).splitlines():
        value = line.strip()
        if not value or value.startswith("#"):
            continue
        value = value.split("#", 1)[0].strip().strip("'\"")
        if value:
            specs.append(value)
    return specs


def parse_route_matrix(route_matrix: Path) -> dict[str, list[dict[str, str]]]:
    result: dict[str, list[dict[str, str]]] = {"golden": [], "all": []}
    if not route_matrix.is_file():
        return result
    text = route_matrix.read_text(encoding="utf-8", errors="replace")
    golden_block = re.search(r"goldenWorkspaceRouteMatrix\s*=\s*\[(.*?)\]\s*as const", text, re.DOTALL)
    app_block = re.search(r"appRouteMatrix\s*=\s*\[(.*?)\]\s*as const", text, re.DOTALL)
    if golden_block:
        for key, path, name in re.findall(
            r"\{\s*key:\s*'([^']+)'\s*,\s*path:\s*'([^']+)'\s*,\s*name:\s*'([^']+)'",
            golden_block.group(1),
        ):
            result["golden"].append({"key": key, "path": path, "name": name})
    if app_block:
        for path, name in re.findall(
            r"\{\s*path:\s*'([^']+)'\s*,\s*name:\s*'([^']+)'",
            app_block.group(1),
        ):
            result["all"].append({"path": path, "name": name})
    return result


def coverage_inventory(repo: Path) -> dict[str, Any]:
    frontend = repo / "frontend"
    verify_script = repo / "scripts" / "verify-app.sh"
    canonical_specs = parse_verify_specs(verify_script)
    all_specs = sorted(
        path.relative_to(frontend).as_posix()
        for path in (frontend / "tests").glob("*.spec.ts")
        if path.is_file()
    )
    canonical_set = set(canonical_specs)
    noncanonical = [spec for spec in all_specs if spec not in canonical_set]
    route_matrix = parse_route_matrix(frontend / "tests" / "helpers" / "routeMatrix.ts")
    spec_text: dict[str, str] = {}
    for spec in canonical_specs:
        path = frontend / spec
        if path.is_file():
            spec_text[spec] = path.read_text(encoding="utf-8", errors="replace")
    routes: list[dict[str, Any]] = []
    for route in route_matrix["all"]:
        refs: list[str] = []
        for spec, text in spec_text.items():
            if route["path"] in text or route["name"] in text:
                refs.append(spec)
            elif "appRouteMatrix" in text:
                refs.append(spec)
        routes.append({**route, "canonical_references": sorted(set(refs))})
    dedicated_gaps: list[dict[str, str]] = []
    for golden in route_matrix["golden"]:
        candidates = [
            spec for spec in all_specs
            if golden["key"] in Path(spec).name.lower() and spec not in canonical_set
        ]
        for spec in candidates:
            dedicated_gaps.append({
                "workspace": golden["key"],
                "spec": spec,
                "reason": "dedicated-spec-not-in-promotion-playwright-list",
            })
    return {
        "schema_version": SCHEMA_VERSION,
        "captured_utc": utc_now().isoformat(),
        "verification_script": str(verify_script),
        "verification_script_sha256": sha256_file(verify_script) if verify_script.is_file() else None,
        "canonical_playwright_specs": canonical_specs,
        "all_top_level_playwright_specs": all_specs,
        "noncanonical_top_level_specs": noncanonical,
        "routes": routes,
        "golden_workspaces": route_matrix["golden"],
        "dedicated_coverage_gaps": dedicated_gaps,
        "interpretation": (
            "Static inventory only. A referenced route or spec is not proof that every control, "
            "state, breakpoint, or destructive/recovery path is covered."
        ),
    }


def cmd_coverage(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    result = coverage_inventory(repo)
    write_metadata("coverage-latest", result)
    if args.json:
        print(json_dump(result), end="")
    else:
        print(f"Canonical Playwright specs: {len(result['canonical_playwright_specs'])}")
        print(f"Top-level specs outside canonical list: {len(result['noncanonical_top_level_specs'])}")
        if result["dedicated_coverage_gaps"]:
            print("Dedicated coverage gaps:")
            for gap in result["dedicated_coverage_gaps"]:
                print(f"  {gap['workspace']}: {gap['spec']}")
        routes_without_refs = [row for row in result["routes"] if not row["canonical_references"]]
        print(f"Routes without static canonical references: {len(routes_without_refs)}")
        for route in routes_without_refs:
            print(f"  {route['path']} ({route['name']})")
    return 0


def evidence_roots(repo: Path) -> list[tuple[Path, str]]:
    return [
        (repo / "backend" / "test-results", "backend/test-results"),
        (repo / "frontend" / "test-results", "frontend/test-results"),
        (repo / "frontend" / "playwright-report", "frontend/playwright-report"),
        (repo / "frontend" / "blob-report", "frontend/blob-report"),
    ]


def fresh_evidence_files(repo: Path, since_epoch: float) -> list[tuple[Path, str]]:
    rows: list[tuple[Path, str]] = []
    threshold = since_epoch - 5.0
    for root, archive_root in evidence_roots(repo):
        if not root.is_dir():
            continue
        for path in sorted(root.rglob("*")):
            if path.is_symlink() or not path.is_file():
                continue
            with contextlib.suppress(OSError):
                stat = path.stat()
                if stat.st_mtime < threshold or stat.st_size > MAX_EVIDENCE_FILE_BYTES:
                    continue
                relative = path.relative_to(root).as_posix()
                rows.append((path, f"{archive_root}/{relative}"))
    for path, archive_name in [
        (repo / "frontend" / "llm-report.json", "frontend/llm-report.json"),
        (audit_log_path(), "controller/audit.log"),
    ]:
        if path.is_file():
            with contextlib.suppress(OSError):
                stat = path.stat()
                if stat.st_mtime >= threshold and stat.st_size <= MAX_EVIDENCE_FILE_BYTES:
                    rows.append((path, archive_name))
    dedup: dict[str, Path] = {}
    for path, name in rows:
        dedup[name] = path
    return [(path, name) for name, path in sorted(dedup.items())]


def parse_llm_failures(repo: Path, since_epoch: float) -> list[dict[str, Any]]:
    path = repo / "frontend" / "llm-report.json"
    if not path.is_file():
        return []
    try:
        if path.stat().st_mtime < since_epoch - 5.0:
            return []
        value = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, json.JSONDecodeError):
        return []
    failures = value.get("failures") if isinstance(value, dict) else None
    return failures if isinstance(failures, list) else []


def create_evidence_archive(repo: Path, destination: Path, since_epoch: float, run_id: str) -> dict[str, Any]:
    captured = utc_now()
    files = fresh_evidence_files(repo, since_epoch)
    destination.parent.mkdir(parents=True, exist_ok=True)
    temp = destination.with_suffix(destination.suffix + ".tmp")
    manifest: list[dict[str, Any]] = []
    with zipfile.ZipFile(temp, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
        for path, name in files:
            data = path.read_bytes()
            digest = hashlib.sha256(data).hexdigest()
            zip_write_bytes(archive, f"EVIDENCE/{name}", data, captured)
            manifest.append({
                "path": f"EVIDENCE/{name}",
                "sha256": digest,
                "size": len(data),
                "mtime": path.stat().st_mtime,
            })
        metadata = {
            "schema_version": SCHEMA_VERSION,
            "artifact_type": "SYSGRID_CURRENT_AUDIT_EVIDENCE",
            "run_id": run_id,
            "captured_utc": captured.isoformat(),
            "freshness_threshold_epoch": since_epoch - 5.0,
            "file_count": len(manifest),
            "files": manifest,
        }
        zip_write_bytes(archive, "EVIDENCE_MANIFEST.json", json_dump(metadata).encode(), captured)
    temp.replace(destination)
    return {
        "path": str(destination),
        "sha256": sha256_file(destination),
        "size": destination.stat().st_size,
        "file_count": len(manifest),
        "run_id": run_id,
    }


def current_audit_for_source(repo: Path, source_fingerprint: str) -> tuple[dict[str, Any] | None, str]:
    audit = read_metadata("audit-latest")
    if not audit:
        return None, "NOT_RUN"
    if audit.get("post_source_fingerprint") != source_fingerprint:
        return audit, "SOURCE_MISMATCH"
    evidence = audit.get("evidence_archive")
    if not isinstance(evidence, dict) or not Path(str(evidence.get("path", ""))).is_file():
        return audit, "EVIDENCE_MISSING"
    return audit, "CURRENT"


def readiness_report(repo: Path) -> dict[str, Any]:
    inventory = source_inventory(repo)
    fingerprint = inventory["source_fingerprint"]
    smoke = read_metadata("smoke-latest")
    coverage = coverage_inventory(repo)
    audit, audit_status = current_audit_for_source(repo, fingerprint)
    identity = repo_identity(repo)
    reasons: list[str] = []
    if not smoke or smoke.get("source_fingerprint") != fingerprint:
        reasons.append("API smoke is missing or belongs to different source bytes.")
    elif smoke.get("result") != "PASS":
        reasons.append(f"API smoke failed: {smoke.get('classification')}.")
    if audit_status != "CURRENT":
        reasons.append(f"Current-source visual/programmatic audit unavailable: {audit_status}.")
    elif audit and audit.get("result") != "PASS":
        reasons.append(f"Current-source audit failed: {audit.get('classification')}.")
    product_dirty = (
        identity["dirty_summary"].get("product_source_backend", 0)
        + identity["dirty_summary"].get("product_source_frontend", 0)
    )
    if product_dirty:
        reasons.append(f"{product_dirty} dirty product-source paths remain unaccepted.")
    if coverage["dedicated_coverage_gaps"]:
        reasons.append(
            f"{len(coverage['dedicated_coverage_gaps'])} dedicated Playwright specs are outside the promotion list."
        )
    reasons.append("Independent product acceptance and Linear lock remain manual qualified-delivery gates.")
    automated_pass = (
        smoke is not None
        and smoke.get("source_fingerprint") == fingerprint
        and smoke.get("result") == "PASS"
        and audit_status == "CURRENT"
        and audit is not None
        and audit.get("result") == "PASS"
    )
    return {
        "schema_version": SCHEMA_VERSION,
        "captured_utc": utc_now().isoformat(),
        "result": "AUTOMATED_GATES_PASS_MANUAL_ACCEPTANCE_REQUIRED" if automated_pass else "NOT_READY",
        "source_fingerprint": fingerprint,
        "source_identity": identity,
        "smoke": {
            "status": "CURRENT" if smoke and smoke.get("source_fingerprint") == fingerprint else "STALE_OR_MISSING",
            "result": smoke.get("result") if smoke else None,
            "classification": smoke.get("classification") if smoke else None,
        },
        "audit": {
            "status": audit_status,
            "result": audit.get("result") if audit else None,
            "classification": audit.get("classification") if audit else None,
            "run_id": audit.get("run_id") if audit else None,
        },
        "coverage": {
            "canonical_spec_count": len(coverage["canonical_playwright_specs"]),
            "noncanonical_spec_count": len(coverage["noncanonical_top_level_specs"]),
            "dedicated_gap_count": len(coverage["dedicated_coverage_gaps"]),
        },
        "reasons": reasons,
        "claim_boundary": (
            "This report never claims unknown defects are impossible. Production acceptance requires "
            "current source, passing runtime/API and rendered/programmatic evidence, risk closure, "
            "independent review, and qualified delivery."
        ),
    }


def cmd_readiness(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    result = readiness_report(repo)
    write_metadata("readiness-latest", result)
    if args.json:
        print(json_dump(result), end="")
    else:
        print(f"Readiness: {result['result']}")
        print(f"Source: {result['source_fingerprint'][:16]}")
        print(
            f"Smoke: {result['smoke']['status']} {result['smoke']['result'] or '-'} "
            f"Audit: {result['audit']['status']} {result['audit']['result'] or '-'}"
        )
        for reason in result["reasons"]:
            print(f"- {reason}")
    return 0 if result["result"].startswith("AUTOMATED_GATES_PASS") else 1


def add_file_to_zip(
    archive: zipfile.ZipFile,
    source: Path,
    name: str,
    when: dt.datetime,
    manifest: list[str],
) -> None:
    if not source.is_file():
        return
    data = source.read_bytes()
    zip_write_bytes(archive, name, data, when)
    manifest.append(f"{hashlib.sha256(data).hexdigest()}  {name}")


def create_review_package(
    repo: Path,
    destination: Path,
    *,
    create_fresh_snapshot: bool = True,
    explicit_audit: dict[str, Any] | None = None,
) -> dict[str, Any]:
    captured = utc_now()
    identity = repo_identity(repo)
    temp_dir = Path(tempfile.mkdtemp(prefix="sysgrid-dev-review-"))
    try:
        if create_fresh_snapshot:
            snapshot_path = temp_dir / f"SYSGRID_AUTHORITATIVE_SOURCE__{timestamp()}__{identity['head'][:12] if identity['head'] else 'nohead'}.zip"
            snapshot = create_source_snapshot(repo, snapshot_path)
        else:
            latest = read_metadata("snapshot-latest")
            if not latest or not Path(str(latest.get("path", ""))).is_file():
                raise RuntimeError("No prior source snapshot is available")
            snapshot = latest
            snapshot_path = Path(str(latest["path"]))
        coverage = coverage_inventory(repo)
        readiness = readiness_report(repo)
        audit = explicit_audit
        evidence_status = "NOT_RUN"
        evidence_path: Path | None = None
        if audit is None:
            audit, evidence_status = current_audit_for_source(repo, snapshot["source_fingerprint"])
        else:
            if audit.get("post_source_fingerprint") == snapshot["source_fingerprint"]:
                evidence = audit.get("evidence_archive")
                if isinstance(evidence, dict) and Path(str(evidence.get("path", ""))).is_file():
                    evidence_status = "CURRENT"
                    evidence_path = Path(str(evidence["path"]))
                else:
                    evidence_status = "EVIDENCE_MISSING"
            else:
                evidence_status = "SOURCE_MISMATCH"
        if evidence_path is None and evidence_status == "CURRENT" and audit:
            evidence = audit.get("evidence_archive")
            if isinstance(evidence, dict):
                candidate = Path(str(evidence.get("path", "")))
                if candidate.is_file():
                    evidence_path = candidate
        summary = {
            "schema_version": SCHEMA_VERSION,
            "artifact_type": "SYSGRID_DEVELOPMENT_REVIEW",
            "tool_version": VERSION,
            "captured_utc": captured.isoformat(),
            "source_identity": identity,
            "snapshot": snapshot,
            "runtime": runtime_status(repo),
            "smoke": read_metadata("smoke-latest"),
            "start": read_metadata("start-latest"),
            "audit": audit,
            "audit_evidence_status": evidence_status,
            "coverage": coverage,
            "readiness": readiness,
            "doctor": read_metadata("doctor-latest"),
            "work_class": "PLATFORM_AND_ENVIRONMENT_OBSERVABILITY",
            "product_attempt_increment": False,
            "product_source_mutation": False,
            "qualified_delivery_replacement": False,
        }
        destination.parent.mkdir(parents=True, exist_ok=True)
        temp_destination = destination.with_suffix(destination.suffix + ".tmp")
        manifest: list[str] = []
        with zipfile.ZipFile(temp_destination, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
            members = [
                ("REVIEW_ME_FIRST.json", json_dump(summary).encode()),
                ("SOURCE_SNAPSHOT.zip", snapshot_path.read_bytes()),
                ("COVERAGE_INVENTORY.json", json_dump(coverage).encode()),
                ("READINESS.json", json_dump(readiness).encode()),
            ]
            if evidence_path:
                members.append(("CURRENT_AUDIT_EVIDENCE.zip", evidence_path.read_bytes()))
            for state_file in sorted(state_root().glob("*.json")):
                members.append((f"STATE/{state_file.name}", state_file.read_bytes()))
            for name, data in sorted(members):
                zip_write_bytes(archive, name, data, captured)
                manifest.append(f"{hashlib.sha256(data).hexdigest()}  {name}")
            add_file_to_zip(archive, runtime_log_path(), "LOGS/runtime.log", captured, manifest)
            add_file_to_zip(archive, audit_log_path(), "LOGS/audit.log", captured, manifest)
            zip_write_bytes(archive, "MANIFEST.sha256", ("\n".join(sorted(manifest)) + "\n").encode(), captured)
        temp_destination.replace(destination)
        return {
            "path": str(destination),
            "size": destination.stat().st_size,
            "sha256": sha256_file(destination),
            "source_snapshot_sha256": snapshot["sha256"],
            "source_fingerprint": snapshot["source_fingerprint"],
            "audit_evidence_status": evidence_status,
            "classification": (
                audit.get("classification")
                if audit and evidence_status == "CURRENT"
                else (summary.get("smoke") or {}).get("classification", "NO_SMOKE_RESULT")
            ),
        }
    finally:
        shutil.rmtree(temp_dir, ignore_errors=True)

def audit_stage_metadata_name(run_id: str, stage: str) -> str:
    safe_run = re.sub(r"[^A-Za-z0-9_.-]+", "-", run_id)
    safe_stage = re.sub(r"[^A-Za-z0-9_.-]+", "-", stage)

def create_combined_evidence_archive(
    destination: Path,
    run_id: str,
    stages: list[dict[str, Any]],
) -> dict[str, Any]:
    captured = utc_now()
    destination.parent.mkdir(parents=True, exist_ok=True)
    temp = destination.with_suffix(destination.suffix + ".tmp")
    manifest: list[dict[str, Any]] = []
    with zipfile.ZipFile(temp, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=9) as archive:
        for stage in stages:
            evidence = stage.get("evidence_archive")
            if not isinstance(evidence, dict):
                continue
            path = Path(str(evidence.get("path", "")))
            if not path.is_file():
                continue
            data = path.read_bytes()
            name = f"STAGES/{stage.get('stage', 'unknown')}.zip"
            zip_write_bytes(archive, name, data, captured)
            manifest.append({
                "path": name,
                "sha256": hashlib.sha256(data).hexdigest(),
                "size": len(data),
                "stage_exit_code": stage.get("exit_code"),
            })
        summary = {
            "schema_version": SCHEMA_VERSION,
            "artifact_type": "SYSGRID_MULTI_STAGE_AUDIT_EVIDENCE",
            "run_id": run_id,
            "captured_utc": captured.isoformat(),
            "stages": stages,
            "stage_archive_manifest": manifest,
        }
        zip_write_bytes(archive, "AUDIT_STAGE_SUMMARY.json", json_dump(summary).encode(), captured)
    temp.replace(destination)
    return {
        "path": str(destination),
        "sha256": sha256_file(destination),
        "size": destination.stat().st_size,
        "file_count": sum(int((stage.get("evidence_archive") or {}).get("file_count", 0)) for stage in stages),
        "stage_count": len(stages),
        "run_id": run_id,
    }


def cmd_capture_stage_evidence(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    current = read_metadata("audit-current") or {}
    if current.get("run_id") != args.run_id:
        print(
            f"Audit run mismatch: expected {current.get('run_id')}, got {args.run_id}",
            file=sys.stderr,
        )
        return 2
    destination = state_root() / "audit" / f"{args.run_id}-{args.stage}-evidence.zip"
    evidence = create_evidence_archive(
        repo,
        destination,
        float(args.since_epoch),
        f"{args.run_id}:{args.stage}",
    )
    failures = parse_llm_failures(repo, float(args.since_epoch))
    stage = {
        "schema_version": SCHEMA_VERSION,
        "run_id": args.run_id,
        "stage": args.stage,
        "exit_code": args.stage_exit_code,
        "result": "PASS" if args.stage_exit_code == 0 else "FAIL",
        "captured_utc": utc_now().isoformat(),
        "since_epoch": float(args.since_epoch),
        "evidence_archive": evidence,
        "failure_count": len(failures),
        "failures": failures[:100],
    }
    write_metadata(audit_stage_metadata_name(args.run_id, args.stage), stage)
    print(json_dump(stage), end="")
    return 0

def start_audit(repo: Path, profile: str) -> dict[str, Any]:
    doctor = doctor_report(repo)
    write_metadata("doctor-latest", doctor)
    if doctor["result"] != "PASS":
        return {
            "result": "FAIL",
            "classification": "DOCTOR_FAILURE",
            "doctor": doctor,
            "repo": str(repo),
        }
    kill_session(AUDIT_SESSION)
    log_path = audit_log_path()
    log_path.parent.mkdir(parents=True, exist_ok=True)
    if log_path.exists():
        shutil.copy2(log_path, log_path.with_name(f"audit-{timestamp()}.log"))
    log_path.write_text("", encoding="utf-8")
    with contextlib.suppress(FileNotFoundError):
        audit_exit_path().unlink()

    inventory = source_inventory(repo)
    coverage = coverage_inventory(repo)
    run_id = f"audit-{timestamp()}-{inventory['source_fingerprint'][:12]}"
    started = utc_now()
    expected_stages = ["extended", "canonical"] if profile == "full" else [profile]
    current = {
        "schema_version": SCHEMA_VERSION,
        "run_id": run_id,
        "result": "RUNNING",
        "classification": "AUDIT_RUNNING",
        "profile": profile,
        "expected_stages": expected_stages,
        "extended_spec_count": len(coverage["noncanonical_top_level_specs"]) if profile == "full" else 0,
        "repo": str(repo),
        "started_utc": started.isoformat(),
        "started_epoch": started.timestamp(),
        "pre_source_fingerprint": inventory["source_fingerprint"],
        "source_identity": repo_identity(repo),
        "coverage_inventory": coverage,
        "log": str(log_path),
    }
    write_metadata("audit-current", current)

    sgd_executable = Path(__file__).resolve()
    verify = repo / "scripts" / "verify-app.sh"
    finalize = [
        str(sgd_executable),
        "_finalize-audit",
        "--repo",
        str(repo),
        "--run-id",
        run_id,
    ]

    def capture_command(stage: str, status_var: str, started_var: str) -> str:
        command = [
            str(sgd_executable),
            "_capture-stage-evidence",
            "--repo",
            str(repo),
            "--run-id",
            run_id,
            "--stage",
            stage,
            "--stage-exit-code",
            f"${status_var}",
            "--since-epoch",
            f"${started_var}",
        ]
        # Preserve shell variables while quoting every fixed argument.
        rendered: list[str] = []
        for token in command:
            if token.startswith("$"):
                rendered.append(f'"{token}"')
            else:
                rendered.append(shlex.quote(token))
        return " ".join(rendered)

    shell_parts = [
        "set -o pipefail",
        f"cd {shlex.quote(str(repo))}",
        f"printf '%s\\n' {shlex.quote('SysGrid audit run ' + run_id + ' profile=' + profile)} | tee -a {shlex.quote(str(log_path))}",
    ]

    if profile == "full":
        extended_specs = list(coverage["noncanonical_top_level_specs"])
        start_command = [
            str(sgd_executable), "start", "--repo", str(repo), "--fast", "--json"
        ]
        runner = repo / "frontend" / "node_modules" / ".bin" / "playwright"
        extended_env = [
            "env",
            "SYSGRID_CANONICAL_GATE=0",
            f"PLAYWRIGHT_BASE_URL=http://{DEFAULT_FRONTEND_HOST}:{DEFAULT_FRONTEND_PORT}",
            f"PW_API_BASE=http://{DEFAULT_BACKEND_HOST}:{DEFAULT_BACKEND_PORT}/api/v1",
            "PW_TENANT_ID=1",
            f"USER_ID={DEFAULT_USER_ID}",
        ]
        extended_command = extended_env + [
            str(runner), "test", *extended_specs, "--workers=1",
        ]
        shell_parts.extend([
            "extended_started=$(date +%s)",
            f"{shlex.join(start_command)} >> {shlex.quote(str(log_path))} 2>&1",
            "runtime_start_status=$?",
            "if [ \"$runtime_start_status\" -eq 0 ]; then "
            f"(cd {shlex.quote(str(repo / 'frontend'))} && {shlex.join(extended_command)}) "
            f"2>&1 | tee -a {shlex.quote(str(log_path))}; "
            "extended_status=${PIPESTATUS[0]}; "
            "else extended_status=90; "
            f"printf '%s\\n' 'Extended audit runtime failed to start.' | tee -a {shlex.quote(str(log_path))}; "
            "fi",
            capture_command("extended", "extended_status", "extended_started") + f" >> {shlex.quote(str(log_path))} 2>&1 || true",
        ])

        canonical_env = audit_profile_env("canonical")
        canonical_command = [
            "env",
            *[f"{key}={value}" for key, value in canonical_env.items()],
            str(verify),
        ]
        shell_parts.extend([
            "canonical_started=$(date +%s)",
            f"{shlex.join(canonical_command)} 2>&1 | tee -a {shlex.quote(str(log_path))}",
            "canonical_status=${PIPESTATUS[0]}",
            capture_command("canonical", "canonical_status", "canonical_started") + f" >> {shlex.quote(str(log_path))} 2>&1 || true",
            "if [ \"$extended_status\" -eq 0 ] && [ \"$canonical_status\" -eq 0 ]; then status=0; else status=1; fi",
        ])
    else:
        env_assignments = audit_profile_env(profile)
        stage_command = [
            "env",
            *[f"{key}={value}" for key, value in env_assignments.items()],
            str(verify),
        ]
        shell_parts.extend([
            "stage_started=$(date +%s)",
            f"{shlex.join(stage_command)} 2>&1 | tee -a {shlex.quote(str(log_path))}",
            "stage_status=${PIPESTATUS[0]}",
            capture_command(profile, "stage_status", "stage_started") + f" >> {shlex.quote(str(log_path))} 2>&1 || true",
            "status=$stage_status",
        ])

    shell_parts.extend([
        f"printf '%s\\n' \"$status\" > {shlex.quote(str(audit_exit_path()))}",
        f"{shlex.join(finalize)} --exit-code \"$status\" >> {shlex.quote(str(log_path))} 2>&1 || true",
        "exit \"$status\"",
    ])
    tmux_shell = "; ".join(shell_parts)

    started_process = run(["tmux", "new-session", "-d", "-s", AUDIT_SESSION, "bash", "-lc", tmux_shell])
    if started_process.returncode != 0:
        failure = {
            **current,
            "result": "FAIL",
            "classification": "TMUX_AUDIT_START_FAILURE",
            "message": started_process.stderr or "Failed to start audit tmux session",
        }
        write_metadata("audit-latest", failure)
        return failure
    return {
        **current,
        "session": AUDIT_SESSION,
        "status_command": "sgd audit-status --phone",
        "result": "STARTED",
    }

def cmd_audit(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    result = start_audit(repo, args.profile)
    if args.json:
        print(json_dump(result), end="")
    else:
        print(f"Audit: {result['result']} profile={args.profile}")
        print(f"Run: {result.get('run_id', '-')}")
        print(f"Source: {result.get('pre_source_fingerprint', '-')}")
        print(f"Log: {result.get('log', audit_log_path())}")
        if result["result"] == "STARTED":
            print("Reconnect-safe status: sgd audit-status --phone")
    return 0 if result["result"] == "STARTED" else 1


def cmd_finalize_audit(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    current = read_metadata("audit-current") or {}
    if current.get("run_id") != args.run_id:
        print(
            f"Audit run mismatch: expected {current.get('run_id')}, got {args.run_id}",
            file=sys.stderr,
        )
        return 2

    pre_fingerprint = str(current.get("pre_source_fingerprint", ""))
    post_inventory = source_inventory(repo)
    post_fingerprint = post_inventory["source_fingerprint"]
    expected_stages = [str(stage) for stage in current.get("expected_stages", [])]
    stages: list[dict[str, Any]] = []
    missing_stages: list[str] = []
    for stage_name in expected_stages:
        stage = read_metadata(audit_stage_metadata_name(args.run_id, stage_name))
        if not stage:
            missing_stages.append(stage_name)
            continue
        stages.append(stage)

    if not stages:
        # Fail-safe fallback: preserve whatever current evidence exists, but mark
        # that the stage capture contract was not satisfied.
        started_epoch = float(current.get("started_epoch", time.time()))
        fallback_path = state_root() / "audit" / f"{args.run_id}-fallback-evidence.zip"
        fallback = create_evidence_archive(repo, fallback_path, started_epoch, f"{args.run_id}:fallback")
        stages.append({
            "schema_version": SCHEMA_VERSION,
            "run_id": args.run_id,
            "stage": "fallback",
            "exit_code": args.exit_code,
            "result": "PASS" if args.exit_code == 0 else "FAIL",
            "captured_utc": utc_now().isoformat(),
            "since_epoch": started_epoch,
            "evidence_archive": fallback,
            "failure_count": 0,
            "failures": [],
        })

    combined_path = state_root() / "audit" / f"{args.run_id}-evidence.zip"
    evidence = create_combined_evidence_archive(combined_path, args.run_id, stages)
    failures: list[dict[str, Any]] = []
    for stage in stages:
        for failure in stage.get("failures") or []:
            row = dict(failure) if isinstance(failure, dict) else {"error": str(failure)}
            row["stage"] = stage.get("stage")
            failures.append(row)

    if pre_fingerprint != post_fingerprint:
        result = "FAIL"
        classification = "SOURCE_CHANGED_DURING_AUDIT"
    elif missing_stages:
        result = "FAIL"
        classification = "EVIDENCE_CAPTURE_FAILURE"
    elif args.exit_code == 0 and all(stage.get("result") == "PASS" for stage in stages):
        result = "PASS"
        classification = "PASS"
    else:
        result = "FAIL"
        classification = "VERIFICATION_FAILURE"

    audit = {
        **current,
        "completed_utc": utc_now().isoformat(),
        "exit_code": args.exit_code,
        "result": result,
        "classification": classification,
        "post_source_fingerprint": post_fingerprint,
        "source_changed": pre_fingerprint != post_fingerprint,
        "missing_stages": missing_stages,
        "stages": stages,
        "evidence_archive": evidence,
        "failure_count": len(failures),
        "failures": failures[:100],
    }

    destination = outbox_root() / f"UPLOAD_THIS_TO_CHATGPT_AUDIT__{args.run_id}.zip"
    package = create_review_package(
        repo,
        destination,
        create_fresh_snapshot=True,
        explicit_audit=audit,
    )
    audit["review_package"] = package
    write_metadata("audit-latest", audit)
    write_metadata("verify-latest", audit)
    write_metadata("verify-status-latest", audit)
    print(json_dump(audit), end="")
    return 0 if result == "PASS" else 1

def audit_status_result() -> dict[str, Any]:
    current = read_metadata("audit-current")
    latest = read_metadata("audit-latest")
    running = session_exists(AUDIT_SESSION)
    if running and current:
        return {**current, "running": True}
    if latest:
        return {**latest, "running": False}
    return {
        "schema_version": SCHEMA_VERSION,
        "result": "UNKNOWN",
        "classification": "NOT_RUN",
        "running": False,
    }


def cmd_audit_status(args: argparse.Namespace) -> int:
    result = audit_status_result()
    if args.phone:
        print(
            f"Audit: {result.get('result', 'UNKNOWN')} "
            f"{result.get('classification', '')} profile={result.get('profile', '-')}"
        )
        print(f"Run: {result.get('run_id', '-')}")
        print(f"Source: {result.get('post_source_fingerprint') or result.get('pre_source_fingerprint') or '-'}")
        if result.get("failure_count"):
            print(f"Failures: {result['failure_count']}")
            for failure in (result.get("failures") or [])[:5]:
                print(f"- {failure.get('test', '<unknown>')}: {failure.get('error', '')[:180]}")
        package = result.get("review_package")
        if isinstance(package, dict):
            print(f"Package: {package.get('path', '-')}")
        print(f"Log: {result.get('log', audit_log_path())}")
    elif args.json:
        print(json_dump(result), end="")
    else:
        print(f"Audit: {result.get('result')} ({result.get('classification')})")
        if result.get("review_package"):
            print(f"Package: {result['review_package'].get('path')}")
    if result.get("running"):
        return 0
    return 0 if result.get("result") == "PASS" else 1


def cmd_cycle(args: argparse.Namespace) -> int:
    repo = discover_repo(args.repo)
    start = start_runtime(
        repo,
        fast=args.fast,
        seed_data=not args.no_seed_data,
        timeout=args.timeout,
    )
    if start["result"] == "PASS":
        smoke = build_smoke_result(repo, args.api_origin, args.user_id, args.tenant_id)
    else:
        inventory = source_inventory(repo)
        smoke = {
            "schema_version": SCHEMA_VERSION,
            "captured_utc": utc_now().isoformat(),
            "result": "FAIL",
            "classification": "RUNTIME_START_FAILURE",
            "source_fingerprint": inventory["source_fingerprint"],
            "repo_identity": repo_identity(repo),
            "probes": [],
            "probe_count": 0,
            "passed_probe_count": 0,
        }
        write_metadata("smoke-latest", smoke)
    destination = outbox_root() / f"UPLOAD_THIS_TO_CHATGPT_DEV__{timestamp()}.zip"
    package = create_review_package(repo, destination, create_fresh_snapshot=True)
    result = {
        "schema_version": SCHEMA_VERSION,
        "result": "PASS" if start["result"] == "PASS" and smoke["result"] == "PASS" else "FAIL",
        "start": start,
        "smoke": smoke,
        "review_package": package,
        "next_action": "Upload the review package to ChatGPT or start `sgd audit --profile full`.",
    }
    write_metadata("cycle-latest", result)
    if args.json:
        print(json_dump(result), end="")
    else:
        print(f"Runtime: {start['result']} ({start['classification']})")
        print_smoke_compact(smoke)
        print(f"Source fingerprint: {package['source_fingerprint']}")
        print(f"Review package: {package['path']}")
        print("Full current-source audit: sgd audit --profile full")
    return 0 if result["result"] == "PASS" else 1


def cmd_logs(args: argparse.Namespace) -> int:
    path = audit_log_path() if args.which in {"audit", "verify"} else runtime_log_path()
    if not path.exists():
        print(f"No {args.which} log exists at {path}", file=sys.stderr)
        return 1
    lines = path.read_text(encoding="utf-8", errors="replace").splitlines()
    print("\n".join(lines[-args.lines:]))
    return 0


def cmd_self_test(_args: argparse.Namespace) -> int:
    failures: list[str] = []
    with tempfile.TemporaryDirectory(prefix="sgd-self-test-") as tmp:
        temp = Path(tmp)
        repo = temp / "repo"
        (repo / "backend" / "venv" / "bin").mkdir(parents=True)
        (repo / "backend" / "alembic" / "versions").mkdir(parents=True)
        (repo / "frontend" / "src").mkdir(parents=True)
        (repo / "frontend" / "tests" / "helpers").mkdir(parents=True)
        (repo / "frontend" / "node_modules" / ".bin").mkdir(parents=True)
        (repo / "scripts").mkdir(parents=True)
        (repo / "patches").mkdir(parents=True)
        files = {
            "backend/venv/bin/python": "#!/bin/sh\n",
            "backend/alembic/versions/001_add_external_entity_secrets.py": "def upgrade():\n    pass\n",
            "frontend/src/app.ts": "export const value = 1\n",
            "frontend/tests/far-golden-workspace.spec.ts": "test('far', () => {})\n",
            "frontend/tests/helpers/routeMatrix.ts": (
                "export const goldenWorkspaceRouteMatrix = ["
                "{ key: 'far', path: '/far', name: 'FAR', archetype: 'analytical', definitionArchetype: 'investigation' }"
                "] as const\n"
                "export const appRouteMatrix = [{ path: '/far', name: 'FAR' }] as const\n"
            ),
            "frontend/node_modules/.bin/playwright": "#!/bin/sh\n",
            "scripts/start-local.sh": "#!/bin/sh\n",
            "scripts/verify-app.sh": (
                "#!/bin/sh\nPLAYWRIGHT_SPECS=(\n  tests/sentinel_comprehensive.spec.ts\n)\n"
            ),
            "patches/old.patch": "diff --git a/a b/a\n",
            ".env.local": "SECRET=should-not-ship\n",
        }
        for relative, content in files.items():
            path = repo / relative
            path.parent.mkdir(parents=True, exist_ok=True)
            path.write_text(content, encoding="utf-8")
        for executable in [
            repo / "backend" / "venv" / "bin" / "python",
            repo / "frontend" / "node_modules" / ".bin" / "playwright",
            repo / "scripts" / "start-local.sh",
            repo / "scripts" / "verify-app.sh",
        ]:
            executable.chmod(0o755)
        run(["git", "init", "-q"], cwd=repo)
        run(["git", "config", "user.email", "selftest@example.invalid"], cwd=repo)
        run(["git", "config", "user.name", "Self Test"], cwd=repo)
        run(["git", "add", "."], cwd=repo)
        run(["git", "commit", "-qm", "fixture"], cwd=repo)
        snapshot = temp / "snapshot.zip"
        result = create_source_snapshot(repo, snapshot)
        with zipfile.ZipFile(snapshot) as archive:
            names = set(archive.namelist())
        if "SOURCE/backend/alembic/versions/001_add_external_entity_secrets.py" not in names:
            failures.append("legitimate migration containing 'secrets' was omitted")
        if "SOURCE/.env.local" in names or "AUXILIARY/.env.local" in names:
            failures.append("runtime env included")
        if "AUXILIARY/patches/old.patch" not in names:
            failures.append("temporary artifact was not separated into AUXILIARY")
        if result["source_fingerprint"] != source_inventory(repo)["source_fingerprint"]:
            failures.append("source fingerprint mismatch")
        coverage = coverage_inventory(repo)
        if not any(gap.get("workspace") == "far" for gap in coverage["dedicated_coverage_gaps"]):
            failures.append("dedicated FAR coverage gap not detected")
        # Repository memory must work outside the repository.
        remember_repo(repo)
        if remembered_repo() != repo.resolve():
            failures.append("repository memory failed")
    output = {"result": "PASS" if not failures else "FAIL", "failures": failures, "version": VERSION}
    print(json_dump(output), end="")
    return 0 if not failures else 1

def main(argv: Sequence[str] | None = None) -> int:
    parser = build_parser()
    args = parser.parse_args(argv)
    try:
        return int(args.func(args))
    except KeyboardInterrupt:
        print("Interrupted.", file=sys.stderr)
        return 130
    except subprocess.TimeoutExpired as exc:
        print(f"Timed out: {exc}", file=sys.stderr)
        return 124
    except Exception as exc:
        error = {
            "result": "FAIL",
            "classification": "SGD_INTERNAL_ERROR",
            "error_type": type(exc).__name__,
            "message": str(exc),
        }
        print(json_dump(error), file=sys.stderr, end="")
        return 70


if __name__ == "__main__":
    raise SystemExit(main())
